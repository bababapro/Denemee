local function handler()
    local ok, result = pcall(function()
        return HttpService:JSONEncode(jsonencodeConfig)
    end)

    if ok and result then
        local assetPath = ""
        local parts = string.split("Decode by ItzYumi/Save System/Steal an Egg/config.json", "/")

        for i = 1, #parts - 1 do
            local secondaryI = i
            local secondaryAssetPath = assetPath == ""

            if secondaryAssetPath then
                secondaryAssetPath = parts[secondaryI]
            end

            if not secondaryAssetPath then
                secondaryAssetPath = assetPath .. "/" .. parts[secondaryI]
            end

            assetPath = secondaryAssetPath

            if not dataCallback(assetPath) then
                secondaryDataCallback(assetPath)
            end
        end

        alternateCallback("Decode by ItzYumi/Save System/Steal an Egg/config.json", result)
    end
end

local function handleResult()
    local Eggs = game:GetService("ReplicatedStorage")
        :WaitForChild("Assets")
        :WaitForChild("Models")
        :WaitForChild("Eggs")

    if #Eggs:GetChildren() == 0 then
        Eggs.ChildAdded:Wait()
        task.wait(0.5)
    end

    local children = Eggs:GetChildren()
    local vNames = {}

    for _, item in ipairs(children) do
        if item:IsA("Model") or item:IsA("Folder") then
            table.insert(vNames, item.Name)
        end
    end

    return vNames
end
local function secondaryHandleResult()
    local Configs = game:GetService("ReplicatedStorage")
        :WaitForChild("Data")
        :WaitForChild("Rarity")
        :WaitForChild("Configs")

    if #Configs:GetChildren() == 0 then
        Configs.ChildAdded:Wait()
        task.wait(0.5)
    end

    local children = Configs:GetChildren()
    local vNames = {}

    for _, item in ipairs(children) do
        local resultCondition = item:IsA("ModuleScript")

        if not resultCondition then
            resultCondition = item:IsA("Folder") or item:IsA("Model")
        end

        if resultCondition then
            table.insert(vNames, item.Name)
        end
    end

    return vNames
end

function uiLibrary.Notify(_, displayValue, text, textLabelColor)
    for i = #frames, 1, -1 do
        local secondaryI = i
        local input = frames[secondaryI]

        if secondaryI >= 3 then
            table.remove(frames, secondaryI)

            local tween = TweenService:Create(
                input,
                TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
                {
                    Position = UDim2.new(1.5, 0, 0, input.Position.Y.Offset),
                    BackgroundTransparency = 1,
                }
            )

            tween:Play()
            tween.Completed:Connect(function()
                input:Destroy()
            end)
        else
            frames[secondaryI + 1] = input

            local product = secondaryI * 68

            TweenService
                :Create(
                    input,
                    TweenInfo.new(0.2, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out),
                    {
                        Position = UDim2.new(0, 0, 0, product),
                    }
                )
                :Play()
        end
    end

    local Frame = Instance.new("Frame")

    Frame.Size = UDim2.new(1, 0, 0, 56)
    Frame.Position = UDim2.new(1.4, 0, 0, 0)
    Frame.BackgroundColor3 = uiLibrary.Theme.SidebarBackground
    Frame.BackgroundTransparency = 0.1
    Frame.BorderSizePixel = 0
    Frame.ZIndex = 10000
    Frame.Parent = vData.NotifyContainer

    local UICorner = Instance.new("UICorner")

    UICorner.CornerRadius = UDim.new(0, 6)
    UICorner.Parent = Frame

    local parent = Instance.new("Frame")

    parent.Size = UDim2.new(0, 4, 1, 0)
    parent.Position = UDim2.new(0, 0, 0, 0)

    local backgroundColor3 = textLabelColor and Color3.fromRGB(239, 68, 68)

    if not backgroundColor3 then
        backgroundColor3 = uiLibrary.Theme.AccentColor
    end

    parent.BackgroundColor3 = backgroundColor3
    parent.BorderSizePixel = 0
    parent.ZIndex = 10001
    parent.Parent = Frame

    local uiCorner = Instance.new("UICorner")

    uiCorner.CornerRadius = UDim.new(0, 4)
    uiCorner.Parent = parent

    local TextLabel = Instance.new("TextLabel")

    TextLabel.Size = UDim2.new(1, -20, 0, 20)
    TextLabel.Position = UDim2.new(0, 14, 0, 8)
    TextLabel.BackgroundTransparency = 1
    TextLabel.Text = tostring(displayValue):upper()

    if textLabelColor then
        textLabelColor = Color3.fromRGB(252, 165, 165)
    end

    if not textLabelColor then
        textLabelColor = uiLibrary.Theme.TextMain
    end

    TextLabel.TextColor3 = textLabelColor
    TextLabel.TextSize = 11
    TextLabel.Font = Enum.Font.SourceSansBold
    TextLabel.TextXAlignment = Enum.TextXAlignment.Left
    TextLabel.ZIndex = 10002
    TextLabel.Parent = Frame

    local textLabel = Instance.new("TextLabel")

    textLabel.Size = UDim2.new(1, -20, 0, 24)
    textLabel.Position = UDim2.new(0, 14, 0, 24)
    textLabel.BackgroundTransparency = 1
    textLabel.Text = text
    textLabel.TextColor3 = uiLibrary.Theme.TextMuted
    textLabel.TextSize = 13
    textLabel.Font = Enum.Font.SourceSans
    textLabel.TextXAlignment = Enum.TextXAlignment.Left
    textLabel.TextWrapped = true
    textLabel.ZIndex = 10002
    textLabel.Parent = Frame
    table.insert(frames, 1, Frame)
    TweenService
        :Create(Frame, TweenInfo.new(0.25, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out), {
            Position = UDim2.new(0, 0, 0, 0),
        })
        :Play()
    task.delay(3, function()
        if Frame and Frame.Parent then
            local find = table.find
            local searchQuery = Frame
            local condition = find(frames, searchQuery)

            if condition then
                table.remove(frames, condition)
            end

            local tween = TweenService:Create(
                Frame,
                TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
                {
                    Position = UDim2.new(1.4, 0, 0, Frame.Position.Y.Offset),
                    BackgroundTransparency = 1,
                }
            )

            tween:Play()
            tween.Completed:Connect(function()
                Frame:Destroy()
            end)
        end
    end)
end

                    table.insert(vs, item)

                table.sort(vs)

                local selectBtnText = ""

                for i, item in ipairs(vs) do
                    selectBtnText ..= (i ~= 1 and ", " or "") .. item

            fontData.SelectBtn.TextSize = not (string.len((tostring(textSizeData))) > 22) and 12
                or 10
        else
            fontData.SelectBtn.Text = "Select Options"
            fontData.SelectBtn.TextColor3 = uiLibrary.Theme.TextMuted
            fontData.SelectBtn.TextSize = 12
        end

        for _, child in pairs(fontData.Scroll:GetChildren()) do
            if child:IsA("TextButton") then
                local gsubResult = string.gsub(child.Text, "^%s+", "")
                local SelectBtnText = fontData.SelectBtn.Text
                local flag = false

                if SelectBtnText == gsubResult then
                    flag = true
                else
                    for match in string.gmatch(SelectBtnText, "([^,]+)") do
                        if gsubResult == string.gsub(match, "^%s+", "") then
                            flag = true

                            break
                        end
                    end
                end

                if flag then
                    child.BackgroundTransparency = 0.4
                else
                    child.BackgroundTransparency = 1
                end
            end
        end
    end
    function fontData.Populate(_, data)
        for _, child in pairs(fontData.Scroll:GetChildren()) do
            if child:IsA("TextButton") then
                child:Destroy()
            end
        end

        local vs = {}

        if additionalArgument == true then
            table.insert(vs, "All")
        end

        if secondaryAdditionalArgument == true then
            table.insert(vs, "Normal")
        end

        for _, item in ipairs(data) do
            table.insert(vs, item)
        end

        for _, item in ipairs(vs) do
            local capturedV = item
            local TextButton = Instance.new("TextButton")

            TextButton.Size = UDim2.new(1, -5, 0, 28)
            TextButton.BackgroundColor3 = uiLibrary.Theme.AccentColor
            TextButton.BackgroundTransparency = 1
            TextButton.Text = "  " .. capturedV
            TextButton.TextColor3 = uiLibrary.Theme.TextMain
            TextButton.TextSize = 13

            TextButton.TextXAlignment = Enum.TextXAlignment.Left
            TextButton.ZIndex = 52
            TextButton.Parent = fontData.Scroll

            local UICorner = Instance.new("UICorner")

            UICorner.CornerRadius = UDim.new(0, 4)
            UICorner.Parent = TextButton

            local SelectBtnText = fontData.SelectBtn.Text
            local flag = false

            if SelectBtnText == capturedV then
                flag = true
            else
                for match in string.gmatch(SelectBtnText, "([^,]+)") do
                    if capturedV == string.gsub(match, "^%s+", "") then
                        flag = true

                        break
                    end
                end
            end

            if flag then
                TextButton.BackgroundTransparency = 0.4
            end

            TextButton.MouseButton1Click:Connect(function()
                if callback then
                    callback(capturedV, fontData)

                    return
                end

                if fontData.SelectBtn.Text == capturedV then
                    fontData:UpdateSelectionText("")

                    return
                end

                fontData:UpdateSelectionText(capturedV)
            end)
        end
    end

    if condition then

    end

                input.UserInputType == Enum.UserInputType.MouseButton1
                or input.UserInputType == Enum.UserInputType.Touch
            ) and fontData.Popup.Visible
        then
            local inputPosition = input.Position
            local AbsolutePosition = fontData.Popup.AbsolutePosition
            local AbsoluteSize = fontData.Popup.AbsoluteSize
            local absolutePosition = fontData.SelectBtn.AbsolutePosition
            local absoluteSize = fontData.SelectBtn.AbsoluteSize
            local isAbsoluteSizeX = inputPosition.X >= AbsolutePosition.X

            if isAbsoluteSizeX then

                if isAbsoluteSizeX then
                    isAbsoluteSizeX = inputPosition.Y >= AbsolutePosition.Y
                        and inputPosition.Y <= AbsolutePosition.Y + AbsoluteSize.Y
                end
            end

            local isX = inputPosition.X >= absolutePosition.X

                if isX then
                    isX = inputPosition.Y >= absolutePosition.Y
                        and inputPosition.Y <= absolutePosition.Y + absoluteSize.Y
                end
            end

            if not isAbsoluteSizeX and not isX then
                fontData.Popup.Visible = false
            end
        end
    end)

local function handleData()
    local conditions = {}

    if
        secondaryDataCallback
        and not dataCallback("Decode by ItzYumi/Save System/Steal an Egg/Saved Configs/")
    then
        secondaryDataCallback("Decode by ItzYumi/Save System/Steal an Egg/Saved Configs/")
    end

    pcall(function()
        local data = listfiles("Decode by ItzYumi/Save System/Steal an Egg/Saved Configs/")

        for _, item in ipairs(data) do
            local dataCondition = item:match("([^/\\]+)%.json$")

            if dataCondition then
                table.insert(conditions, dataCondition)
            end
        end
    end)

    return conditions
end
