local function fsReady()
	return type(writefile) == "function" and type(readfile) == "function"
end

local function loadStore()
	local data = { configs = {}, autoload = "" }
	if not fsReady() then
		return data
	end
	local ok, raw = pcall(function()
		if type(isfile) == "function" and not isfile(STORE_FILE) then
			return nil
		end
		return readfile(STORE_FILE)
	end)
	if not ok or type(raw) ~= "string" or raw == "" then
		return data
	end
	local parsed = nil
	pcall(function()
		parsed = HttpService:JSONDecode(raw)
	end)
	if type(parsed) == "table" then
		parsed.configs = parsed.configs or {}
		parsed.autoload = parsed.autoload or ""
		return parsed
	end
	return data
end

local function saveStore(data)
	if not fsReady() then
		return
	end
	pcall(function()
		if type(makefolder) == "function" and type(isfolder) == "function" and not isfolder(STORE_FOLDER) then
			makefolder(STORE_FOLDER)
		end
		writefile(STORE_FILE, HttpService:JSONEncode(data))
	end)
end

local function serverHop()
	notify("Looking for a quiet server")
	local ok = pcall(function()
		local raw = game:HttpGet("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100")
		local data = HttpService:JSONDecode(raw)
		local best
		for _, server in data.data or {} do
			if server.id ~= game.JobId and type(server.playing) == "number" and server.playing < (server.maxPlayers or 1) then
				if not best or server.playing < best.playing then
					best = server
				end
			end
		end
		if not best then
			error("none")
		end
		notify("Hopping · " .. tostring(best.playing) .. " players")
		TeleportService:TeleportToPlaceInstance(game.PlaceId, best.id, LocalPlayer)
	end)
	if not ok then
		notify("Server hop failed")
	end
end

local function partOf(inst)
	if not inst then
		return nil
	end
	if inst:IsA("BasePart") then
		return inst
	end
	if inst:IsA("Model") then
		return inst:FindFirstChild("Head") or inst:FindFirstChild("HumanoidRootPart") or inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart")
	end
	return inst:FindFirstAncestorWhichIsA("BasePart")
end

local function tag(inst, text, color)
	local host = partOf(inst)
	if not host then
		return
	end
	local bb = visuals[inst]
	if not bb or not bb.Parent then
		bb = Instance.new("BillboardGui")
		bb.Name = "BunnyTag"
		bb.AlwaysOnTop = true
		bb.Size = UDim2.fromOffset(180, 36)
		bb.StudsOffset = Vector3.new(0, 3, 0)
		bb.MaxDistance = 900
		bb.Parent = ensureFolder()
		local label = Instance.new("TextLabel", bb)
		label.Name = "Label"
		label.BackgroundTransparency = 1
		label.Size = UDim2.fromScale(1, 1)
		label.Font = Enum.Font.GothamBold
		label.TextSize = 14
		label.TextStrokeTransparency = 0.4
		visuals[inst] = bb
	end
	bb.Adornee = host
	local label = bb:FindFirstChild("Label")
	if label then
		label.Text = text
		label.TextColor3 = color
	end
end

local SKIP_NAMES = { "dmitri", "vendor", "shop", "cashier", "clerk", "dummy", "lobby" }

local function isMonkey(model)
	if not model or not model:IsA("Model") then
		return false
	end
	if Players:GetPlayerFromCharacter(model) then
		return false
	end
	local views = workspace:FindFirstChild("UnitViews")
	if not views or not model:IsDescendantOf(views) then
		return false
	end
	local unitPart = model:FindFirstChild("Unit")
	if not unitPart or unitPart:GetAttribute("UnitId") == nil then
		return false
	end
	local hum = model:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then
		return false
	end
	local name = string.lower(model.Name .. " " .. tostring(hum.DisplayName or ""))
	for _, word in SKIP_NAMES do
		if string.find(name, word, 1, true) then
			return false
		end
	end
	return true
end

local function consider(inst)
	local model = nil
	if inst:IsA("Model") then
		model = inst
	elseif inst:IsA("Humanoid") and inst.Parent and inst.Parent:IsA("Model") then
		model = inst.Parent
	end
	if not model then
		return
	end
	local views = workspace:FindFirstChild("UnitViews")
	if not views or not model:IsDescendantOf(views) then
		return
	end
	if model:FindFirstChildOfClass("Humanoid") and isMonkey(model) then
		monkeys[model] = true
	end
end

local function scanWorld()
	if not running then
		return
	end
	local root = rootOf()
	local origin = root and root.Position or Vector3.zero
	local wanted = {}
	for model in monkeys do
		if not isMonkey(model) then
			monkeys[model] = nil
		else
			local part = partOf(model)
			local hum = model:FindFirstChildOfClass("Humanoid")
			if part and hum and Config.EspMonkeys then
				local dist = math.floor((part.Position - origin).Magnitude)
				highlight(model, Color3.fromRGB(255, 90, 120))
				tag(model, string.format("%s  %d  %d", model.Name, math.floor(hum.Health), dist), Color3.fromRGB(255, 150, 170))
				wanted[model] = true
			end
		end
	end
	if Config.EspPlayers and root then
		for _, player in Players:GetPlayers() do
			if player ~= LocalPlayer and player.Character then
				local part = partOf(player.Character)
				if part then
					local dist = math.floor((part.Position - origin).Magnitude)
					highlight(player.Character, Color3.fromRGB(126, 224, 164))
					tag(player.Character, player.DisplayName .. "  " .. dist, Color3.fromRGB(126, 224, 164))
					wanted[player.Character] = true
				end
			end
		end
	end
	for inst, bb in visuals do
		if not wanted[inst] then
			bb:Destroy()
			visuals[inst] = nil
		end
	end
	for model, hl in highlights do
		if not wanted[model] then
			hl:Destroy()
			highlights[model] = nil
		end
	end
end

local function nearestMonkey(maxDist)
	local root = rootOf()
	local camera = workspace.CurrentCamera
	if not root or not camera then
		return nil
	end
	local best, bestDist
	for model in monkeys do
		if model.Parent and isMonkey(model) then
			local part = partOf(model)
			if part then
				local dist = (part.Position - root.Position).Magnitude
				if dist <= maxDist and (not bestDist or dist < bestDist) then
					local look = camera.CFrame.LookVector
					local aim = part.Position - camera.CFrame.Position
					if aim.Magnitude > 0.1 then
						local angle = math.deg(math.acos(math.clamp(look:Dot(aim.Unit), -1, 1)))
						if not Config.Aim or angle <= (Config.AimFov or 80) or dist <= (Config.MeleeRange or 16) then
							best = part
							bestDist = dist
						end
					end
				end
			end
		else
			monkeys[model] = nil
		end
	end
	return best, bestDist
end

local function pressSkip()
	local pg = LocalPlayer:FindFirstChild("PlayerGui")
	if not pg then
		return
	end
	for _, d in pg:GetDescendants() do
		if (d:IsA("TextButton") or d:IsA("ImageButton")) and d.Visible and d.AbsoluteSize.X > 20 then
			local screen = d:FindFirstAncestorOfClass("ScreenGui")
			if screen and screen.Name ~= "BunnyHubMonkeys" and screen.Name ~= "BunnyHub_Loader" then
				local blob = string.lower(d.Name .. " " .. (d:IsA("TextButton") and d.Text or ""))
				for _, child in d:GetDescendants() do
					if child:IsA("TextLabel") or child:IsA("TextButton") then
						blob = blob .. " " .. string.lower(child.Text or "")
					end
				end
				if string.find(blob, "skip", 1, true) then
					if type(getconnections) == "function" then
						for _, signalName in { "MouseButton1Click", "Activated" } do
							local signal = d[signalName]
							if typeof(signal) == "RBXScriptSignal" then
								for _, conn in getconnections(signal) do
									pcall(function()
										conn:Fire()
									end)
								end
							end
						end
					end
					if type(firesignal) == "function" then
						pcall(function()
							firesignal(d.MouseButton1Click)
						end)
						pcall(function()
							firesignal(d.Activated)
						end)
					end
					return
				end
			end
		end
	end
end

local lastShot = 0
local lastAlt = 0
local lastKeys = 0
local lastSkip = 0

local function stepCombat()
	local now = os.clock()
	local target, dist = nearestMonkey(Config.AimRange or 180)
	local camera = workspace.CurrentCamera
	if Config.Aim and target and camera and not Config.Freecam then
		local goal = CFrame.lookAt(camera.CFrame.Position, target.Position)
		camera.CFrame = camera.CFrame:Lerp(goal, math.clamp(Config.AimSmooth or 0.35, 0.08, 1))
	end
	local close = dist and dist <= (Config.MeleeRange or 16)
	if ((Config.AutoPrimary and target) or (Config.Melee and close)) and now - lastShot > 0.09 then
		lastShot = now
		shoot()
	end
	if Config.AutoSecondary and target and now - lastAlt > 0.45 then
		lastAlt = now
		clickRight()
	end
	if now - lastKeys > 0.85 then
		lastKeys = now
		if Config.AutoWeapon then
			tapKey(Enum.KeyCode.V)
		end
		if Config.AutoHealth then
			tapKey(Enum.KeyCode.C)
		end
		if Config.AutoUlt then
			tapKey(Enum.KeyCode.Q)
		end
		if Config.AutoNade then
			tapKey(Enum.KeyCode.G)
		end
		if Config.AutoBarricade then
			tapKey(Enum.KeyCode.X)
		end
	end
	if Config.AutoSkip and now - lastSkip > 1.6 then
		lastSkip = now
		pressSkip()
	end
end

local function applyWorld()
	if Config.Bright then
		if not lightBackup then
			lightBackup = {
				Brightness = Lighting.Brightness,
				ClockTime = Lighting.ClockTime,
				FogEnd = Lighting.FogEnd,
				GlobalShadows = Lighting.GlobalShadows,
			}
		end
		Lighting.Brightness = 2
		Lighting.ClockTime = 14
		Lighting.FogEnd = 100000
		Lighting.GlobalShadows = false
	elseif lightBackup then
		Lighting.Brightness = lightBackup.Brightness
		Lighting.ClockTime = lightBackup.ClockTime
		Lighting.FogEnd = lightBackup.FogEnd
		Lighting.GlobalShadows = lightBackup.GlobalShadows
		lightBackup = nil
	end
	if Config.NoBlur then
		local function mute(parent)
			if not parent then
				return
			end
			for _, inst in parent:GetChildren() do
				if inst:IsA("BlurEffect") or inst:IsA("DepthOfFieldEffect") then
					inst.Enabled = false
				end
			end
		end
		mute(Lighting)
		mute(workspace.CurrentCamera)
	end
end

local function applyMove(dt)
	local root, hum, character = rootOf()
	if not character or not hum or not root then
		return
	end
	if Config.Noclip then
		for _, part in character:GetDescendants() do
			if part:IsA("BasePart") and part.CanCollide then
				noclipState[part] = true
				part.CanCollide = false
			end
		end
	else
		for part in noclipState do
			if part.Parent then
				part.CanCollide = true
			end
			noclipState[part] = nil
		end
	end
	local camera = workspace.CurrentCamera
	if camera and Config.FovOn then
		camera.FieldOfView = Config.Fov
	end
	if Config.JumpOn then
		hum.UseJumpPower = true
		hum.JumpPower = Config.JumpPower or 100
	end
end

local camState
local freeLook = false
local function updateFreecam(dt)
	local camera = workspace.CurrentCamera
	if not camera then
		return
	end
	if Config.Freecam then
		if camera.CameraType ~= Enum.CameraType.Scriptable then
			camState = camera.CameraType
			camera.CameraType = Enum.CameraType.Scriptable
		end
		local boost = UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) and 3 or 1
		local speed = 40 * boost * dt
		local move = Vector3.zero
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then
			move += camera.CFrame.LookVector
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then
			move -= camera.CFrame.LookVector
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then
			move -= camera.CFrame.RightVector
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then
			move += camera.CFrame.RightVector
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
			move += Vector3.yAxis
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
			move -= Vector3.yAxis
		end
		if move.Magnitude > 0 then
			camera.CFrame += move.Unit * speed
		end
	elseif camState then
		camera.CameraType = camState
		camState = nil
	end
end

track(RunService.RenderStepped:Connect(function()
	if not running or not Config.SpeedOn then
		return
	end
	local root, hum = rootOf()
	if not root or not hum then
		return
	end
	local dir = hum.MoveDirection
	if dir.Magnitude <= 0 then
		return
	end
	local mult = Config.Speed
	if type(mult) ~= "number" or mult > 3 or mult < 1.2 then
		mult = 1.5
	end
	local vel = root.AssemblyLinearVelocity
	root.AssemblyLinearVelocity = Vector3.new(dir.X * 16 * mult, vel.Y, dir.Z * 16 * mult)
end))

local function saveSettings()
	if not fsReady() then
		notify("Saving needs an executor with writefile")
		return
	end

		if type(makefolder) == "function" and type(isfolder) == "function" and not isfolder(STORE_FOLDER) then
			makefolder(STORE_FOLDER)
		end
		writefile(SETTINGS_FILE, HttpService:JSONEncode(Config))
	end)
	notify(ok and "Settings saved" or "Could not save")
end

local function loadSettings()
	if not fsReady() then
		return
	end
	local ok, raw = pcall(function()
		if type(isfile) == "function" and not isfile(SETTINGS_FILE) then
			return nil
		end
		return readfile(SETTINGS_FILE)
	end)
	if not ok or type(raw) ~= "string" or raw == "" then
		return
	end
	local parsed
	pcall(function()
		parsed = HttpService:JSONDecode(raw)
	end)
	if type(parsed) ~= "table" then
		return
	end
	for key, value in parsed do
		if Config[key] ~= nil and type(value) == type(Config[key]) then
			Config[key] = value
		end
	end
end

	local function blob(parent, x, y, w, h, color, round, rot)
		local f = Instance.new("Frame", parent)
		f.Size = UDim2.fromOffset(w, h)
		f.Position = UDim2.fromOffset(x, y)
		f.BackgroundColor3 = color
		f.BorderSizePixel = 0
		f.Rotation = rot or 0
		if round then
			Instance.new("UICorner", f).CornerRadius = UDim.new(1, 0)
		else
			Instance.new("UICorner", f).CornerRadius = UDim.new(0, 2)
		end
		return f
	end

	local function polar(parent, cx, cy, r, angle, w, h, color, round, extraTilt)
		local rad = math.rad(angle)
		local piece = Instance.new("Frame", parent)
		piece.Size = UDim2.fromOffset(w, h)
		piece.Position = UDim2.fromOffset(cx + math.cos(rad) * r - w / 2, cy + math.sin(rad) * r - h / 2)
		piece.BackgroundColor3 = color
		piece.BorderSizePixel = 0
		piece.Rotation = angle + (extraTilt or 0)
		if round then
			Instance.new("UICorner", piece).CornerRadius = UDim.new(1, 0)
		else
			Instance.new("UICorner", piece).CornerRadius = UDim.new(0, 2)
		end
		return piece
	end

	local function shade(parent, light, dark, rotation)
		local g = Instance.new("UIGradient", parent)
		g.Color = ColorSequence.new(light, dark)
		g.Rotation = rotation or 90
		return g
	end

	local function makeHeart(size, color)
		local root = Instance.new("Frame")
		root.Size = UDim2.fromOffset(size, size)
		root.BackgroundTransparency = 1
		local square = size * 0.7
		local point = blob(root, size * 0.5 - square / 2, size - square * 0.707 - square / 2, square, square, color, false, 45)
		local light = color:Lerp(Color3.new(1, 1, 1), 0.42)
		blob(root, size * 0.04, size * 0.1, size * 0.5, size * 0.5, light, true)
		blob(root, size * 0.46, size * 0.1, size * 0.5, size * 0.5, light, true)
		local glow = blob(root, size * 0.2, size * 0.2, size * 0.2, size * 0.13, Color3.new(1, 1, 1), true, -35)
		glow.BackgroundTransparency = 0.55
		shade(point, color:Lerp(Color3.new(1, 1, 1), 0.3), color:Lerp(Color3.new(0, 0, 0), 0.12))
		return root
	end

	local function clearDecor()

	local function buildDecor(mode)
		clearDecor()
		local spec = DECOR[mode]
		if not spec then
			return
		end
		for i = 1, spec.count do
			local depth = (i % 4) / 3
			local size = spec.size(depth)
			local holder = Instance.new("Frame", decorLayer)
			holder.Size = UDim2.fromOffset(size, size)
			holder.BackgroundTransparency = 1
			holder.BorderSizePixel = 0
			local group = Instance.new("CanvasGroup", holder)
			group.Size = UDim2.fromScale(1, 1)
			group.BackgroundTransparency = 1
			group.BorderSizePixel = 0
			group.GroupTransparency = 0
			spec.build(size, spec.color()).Parent = group
			decorParts[i] = {
				frame = holder,
				group = group,
				x = math.random(),
				y = math.random(),
				speed = spec.speed(depth),
				sway = spec.sway * (0.5 + math.random()),
				phase = math.random() * math.pi * 2,
				pulse = 0.6 + math.random() * 1.6,
				spin = spec.spin and spec.spin(depth) or 0,
				twinkle = spec.twinkle,
				base = 0.35 + (1 - depth) * 0.4,
			}
		end
	end

				Instance.new("UICorner", frame).CornerRadius = UDim.new(1, 0)
			else
				Instance.new("UICorner", frame).CornerRadius = UDim.new(1, -2)
			end

				rim.Color = Color3.fromRGB(255, 255, 255)
				rim.Thickness = 1.2
				rim.Transparency = 0.55

				frame = frame,
				x = math.random(),
				y = math.random(),
				speed = spec.speed(depth),
				vx = spec.vx and spec.vx(depth) or 0,

	local function slider(parent, labelText, min, max, get, set, filter, decimals)
		local wrap = Instance.new("Frame", parent)
		wrap.Size = UDim2.new(1, 0, 0, phone and 48 or 36)
		wrap.BackgroundTransparency = 1
		wrap:SetAttribute("Filter", string.lower(filter or labelText))
		local name = Instance.new("TextLabel", wrap)

		name.Size = UDim2.new(1, -48, 0, 16)

		name.TextSize = phone and TYPE.small or TYPE.body
		name.TextXAlignment = Enum.TextXAlignment.Left
		name.TextColor3 = TEXT
		name.Text = labelText
		local valueLabel = Instance.new("TextLabel", wrap)
		valueLabel.BackgroundTransparency = 1
		valueLabel.Size = UDim2.fromOffset(48, 16)
		valueLabel.Position = UDim2.new(1, -48, 0, 0)
		valueLabel.Font = Enum.Font.GothamBold
		valueLabel.TextSize = TYPE.body
		valueLabel.TextColor3 = PINK
		local hit = Instance.new("TextButton", wrap)
		hit.Size = UDim2.new(1, 0, 0, phone and 30 or 18)
		hit.Position = UDim2.fromOffset(0, 18)
		hit.BackgroundTransparency = 1
		hit.Text = ""
		hit.AutoButtonColor = false
		hit.Active = true
		local trackBar = Instance.new("Frame", hit)
		trackBar.Size = UDim2.new(1, 0, 0, phone and 10 or 8)
		trackBar.Position = UDim2.new(0, 0, 0.5, phone and -5 or -4)
		trackBar.BackgroundColor3 = Color3.fromRGB(48, 44, 54)
		trackBar.BorderSizePixel = 0
		corner(trackBar, 5)
		local fill = Instance.new("Frame", trackBar)
		fill.BackgroundColor3 = PINK
		fill.BorderSizePixel = 0
		corner(fill, 5)
		local knob = Instance.new("Frame", hit)
		knob.Size = UDim2.fromOffset(phone and 18 or 14, phone and 18 or 14)
		knob.AnchorPoint = Vector2.new(0.5, 0.5)
		knob.BackgroundColor3 = TEXT
		corner(knob, 9)
		local function paint(n)
			local alpha = math.clamp((n - min) / (max - min), 0, 1)
			fill.Size = UDim2.fromScale(alpha, 1)
			knob.Position = UDim2.new(alpha, 0, 0.5, 0)
			if decimals and decimals > 0 then
				valueLabel.Text = string.format("%." .. tostring(decimals) .. "f", n)
			else
				valueLabel.Text = tostring(math.floor(n + 0.5))
			end
			valueLabel.TextColor3 = PINK
		end
		local function fromX(x)
			local alpha = math.clamp((x - hit.AbsolutePosition.X) / math.max(hit.AbsoluteSize.X, 1), 0, 1)
			local n = min + (max - min) * alpha
			if decimals and decimals > 0 then
				local scale = 10 ^ decimals
				n = math.floor(n * scale + 0.5) / scale
			end
			set(n)
			paint(get())
		end
		local sliding = false
		local scroll
		local baseKnobSize = phone and 18 or 14
		hit.MouseEnter:Connect(function()
			if not sliding then
				tweenBounce(knob, 0.2, { Size = UDim2.fromOffset(baseKnobSize + 3, baseKnobSize + 3), BackgroundColor3 = PINK })
			end
		end)
		hit.MouseLeave:Connect(function()
			if not sliding then
				tweenBounce(knob, 0.2, { Size = UDim2.fromOffset(baseKnobSize, baseKnobSize), BackgroundColor3 = TEXT })
			end
		end)
		hit.InputBegan:Connect(function(input)
			if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
				return
			end
			sliding = true
			tweenBounce(knob, 0.15, { Size = UDim2.fromOffset(baseKnobSize + 5, baseKnobSize + 5), BackgroundColor3 = PINK })
			scroll = hit:FindFirstAncestorOfClass("ScrollingFrame")
			if scroll then
				scroll.ScrollingEnabled = false
			end
			fromX(input.Position.X)
		end)
		track(UserInputService.InputChanged:Connect(function(input)
			if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
				fromX(input.Position.X)
			end
		end))
		track(UserInputService.InputEnded:Connect(function(input)
			if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
				return
			end
			if sliding then
				tweenBounce(knob, 0.2, { Size = UDim2.fromOffset(baseKnobSize, baseKnobSize), BackgroundColor3 = TEXT })
				if scroll then
					scroll.ScrollingEnabled = true
				end
			end
			sliding = false
		end))
		paint(get())
		sliderPaints[#sliderPaints + 1] = function()
			paint(get())
		end
	end

	local function snapshot()
		local features = {}
		for key, value in Config do
			features[key] = value
		end
		local moves = {}
		for styleName, list in MOVES do
			local delays = {}
			for index, mv in list do
				delays[tostring(index)] = mv.Delay
			end
			moves[styleName] = delays
		end
		local styles = {}
		for _, style in STYLES do
			styles[style.Name] = style.On
		end
		return {
			features = features,
			moves = moves,
			styles = styles,
			theme = Ui.Theme,
			scale = Ui.Scale,
			radius = Ui.Radius,
			transparent = Ui.Transparent,
			dim = Ui.Dim,
			weather = Ui.Weather,
			weatherMode = Ui.WeatherMode,
			decor = Ui.Decor,
			glass = Ui.Glass,
			drag = Ui.Drag,
			density = Ui.Density,
			menuButton = Ui.MenuButton,
			hide = hideKey.Name,
		}
	end

	local function applyChrome()
		local theme = THEMES[Ui.Theme] or THEMES.Bunny
		INK = theme.ink
		SIDE = theme.side
		CARD = theme.card
		LINE = theme.line
		NAV = theme.nav
		PINK = theme.pink
		MUTED = theme.muted
		TEXT = theme.text
		local level = glassLevel(Ui.Glass)
		local alpha = Ui.Transparent and 0.28 or 0
		if level > 0 then
			alpha = math.max(alpha, level == 1 and 0.42 or 0.6)
		end
		local rimOn = level > 0 and not Ui.Transparent
		local glassT = rimOn and (level == 1 and 0.88 or 0.7) or 1
		mainGlass.Transparency = glassT
		sideGlass.Transparency = glassT
		contentGlass.Transparency = glassT
		local ct = TweenInfo.new(0.4, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
		TweenService:Create(main, ct, { BackgroundColor3 = INK, BackgroundTransparency = alpha }):Play()
		TweenService:Create(side, ct, { BackgroundColor3 = SIDE, BackgroundTransparency = alpha }):Play()
		TweenService:Create(content, ct, { BackgroundColor3 = theme.content, BackgroundTransparency = alpha }):Play()
		TweenService:Create(windowStroke, ct, { Color = PINK, Transparency = Ui.Theme == "Bunny" and 0.78 or 0.45 }):Play()
		windowWash.Color = ColorSequence.new(PINK:Lerp(INK, 0.78), PINK:Lerp(INK, 0.93))
		windowWash.Enabled = Ui.Theme ~= "Mono"
		TweenService:Create(menuButton, ct, { BackgroundColor3 = CARD, TextColor3 = PINK }):Play()
		local menuStroke = menuButton:FindFirstChildOfClass("UIStroke")
		if menuStroke then
			TweenService:Create(menuStroke, ct, { Color = LINE }):Play()
		end
		TweenService:Create(search, ct, { BackgroundColor3 = CARD, TextColor3 = TEXT, PlaceholderColor3 = MUTED }):Play()
		TweenService:Create(searchStroke, ct, { Color = LINE }):Play()
		TweenService:Create(closeBtn, ct, { BackgroundColor3 = CARD, TextColor3 = MUTED }):Play()
		TweenService:Create(pageTitle, ct, { TextColor3 = TEXT }):Play()
		TweenService:Create(statusLabel, ct, { TextColor3 = MUTED }):Play()
		TweenService:Create(brandName, ct, { TextColor3 = TEXT }):Play()
		TweenService:Create(brandVer, ct, { TextColor3 = PINK }):Play()
		TweenService:Create(logo, ct, { ImageColor3 = TEXT }):Play()
		avatarRing.Color = PINK
		ringStroke.Color = PINK
		TweenService:Create(userName, ct, { TextColor3 = TEXT }):Play()
		TweenService:Create(footer, ct, { TextColor3 = MUTED }):Play()
		TweenService:Create(dimmer, ct, { BackgroundTransparency = Ui.Dim and 0.45 or 1 }):Play()
		uiScale.Scale = Ui.Scale / 100
		syncWeather()
		syncDecor()
		snowLayer.Visible = (Ui.Weather ~= "Off" or Ui.Decor ~= "Off") and main.Visible
		decorLayer.Visible = Ui.Decor ~= "Off" and main.Visible
		if Ui.MenuButton == "Hidden" then
			menuButton.Visible = false
		elseif Ui.MenuButton == "Mobile" then
			menuButton.Visible = phone
		elseif Ui.MenuButton ~= "Hover" then
			menuButton.Visible = true
		end
		for _, inst in gui:GetDescendants() do
			if inst:GetAttribute("BunnyRole") == "card" and inst:IsA("Frame") then
				TweenService:Create(inst, ct, { BackgroundColor3 = CARD, BackgroundTransparency = alpha }):Play()
				local stroke = inst:FindFirstChildOfClass("UIStroke")
				if stroke then
					TweenService:Create(stroke, ct, { Color = LINE }):Play()
				end
			end
			if inst:IsA("UICorner") and inst.Parent and (inst.Parent == main or inst.Parent == side or inst.Parent == content or inst.Parent:GetAttribute("BunnyRole") == "card") then
				inst.CornerRadius = UDim.new(0, Ui.Radius)
			end
			if inst:IsA("UIListLayout") and inst.Parent and inst.Parent:IsA("ScrollingFrame") then
				inst.Padding = UDim.new(0, Ui.Density == "Compact" and 4 or 8)

			entry.paint()
		end

			paint()
		end
		for _, paint in dropdownPaints do
			paint()
		end
	end

	local function dropdown(parent, labelText, options, get, set, filter)
		local row = Instance.new("Frame", parent)
		row.Size = UDim2.new(1, 0, 0, 28)
		row.BackgroundTransparency = 1
		row:SetAttribute("Filter", string.lower(filter or labelText))
		local name = Instance.new("TextLabel", row)
		name.BackgroundTransparency = 1
		name.Size = UDim2.new(1, -130, 0, 28)
		name.Font = Enum.Font.GothamMedium
		name.TextSize = TYPE.body
		name.TextXAlignment = Enum.TextXAlignment.Left
		name.TextColor3 = TEXT
		name.Text = labelText
		local button = Instance.new("TextButton", row)
		button.Size = UDim2.fromOffset(122, 22)
		button.Position = UDim2.new(1, -122, 0, 3)
		button.BackgroundColor3 = Color3.fromRGB(22, 20, 26)
		button.Font = Enum.Font.Gotham
		button.TextSize = TYPE.small
		button.TextColor3 = TEXT
		button.Text = tostring(get()) .. "  v"
		button.AutoButtonColor = false
		corner(button, 6)
		button.MouseEnter:Connect(function()
			tween(button, 0.15, { BackgroundColor3 = Color3.fromRGB(32, 28, 38) })
		end)
		button.MouseLeave:Connect(function()
			tween(button, 0.15, { BackgroundColor3 = Color3.fromRGB(22, 20, 26) })
		end)
		local list = Instance.new("Frame", row)
		list.BackgroundColor3 = Color3.fromRGB(18, 16, 22)
		list.Position = UDim2.fromOffset(0, 28)
		list.Size = UDim2.new(1, 0, 0, 0)
		list.Visible = false
		list.ClipsDescendants = true
		list.ZIndex = 8
		corner(list, 6)
		Instance.new("UIStroke", list).Color = LINE
		local function choices()
			if type(options) == "function" then
				return options()
			end
			return options
		end
		local function closeList()
			local tInfo = TweenInfo.new(0.2, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
			local tList = TweenService:Create(list, tInfo, { Size = UDim2.new(1, 0, 0, 0) })
			local tRow = TweenService:Create(row, tInfo, { Size = UDim2.new(1, 0, 0, 28) })
			tList:Play()
			tRow:Play()
			tList.Completed:Connect(function()
				if list.Size.Y.Offset <= 0 then
					list.Visible = false
				end
			end)
		end
		local function openList()
			for _, child in list:GetChildren() do
				if child:IsA("TextButton") then
					child:Destroy()
				end
			end
			local items = choices()
			for index, option in items do
				local pick = Instance.new("TextButton", list)
				pick.Size = UDim2.new(1, -8, 0, 22)
				pick.Position = UDim2.fromOffset(4, 4 + (index - 1) * 24)
				pick.BackgroundTransparency = 1
				pick.Font = Enum.Font.Gotham
				pick.TextSize = TYPE.small
				pick.TextXAlignment = Enum.TextXAlignment.Left
				pick.TextColor3 = TEXT
				pick.Text = "   " .. tostring(option)
				pick.ZIndex = 9
				corner(pick, 4)
				pick.MouseEnter:Connect(function()
					tween(pick, 0.15, { BackgroundTransparency = 0.85, BackgroundColor3 = PINK, TextColor3 = PINK })
				end)
				pick.MouseLeave:Connect(function()
					tween(pick, 0.15, { BackgroundTransparency = 1, TextColor3 = TEXT })
				end)
				pick.MouseButton1Click:Connect(function()
					set(option)
					button.Text = tostring(option) .. "  v"
					closeList()
				end)
			end
			list.Visible = true
			local targetH = 8 + #items * 24
			local tInfo = TweenInfo.new(0.25, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
			TweenService:Create(list, tInfo, { Size = UDim2.new(1, 0, 0, targetH) }):Play()
			TweenService:Create(row, tInfo, { Size = UDim2.new(1, 0, 0, 32 + targetH) }):Play()
		end
		button.MouseButton1Click:Connect(function()
			if list.Visible and list.Size.Y.Offset > 0 then
				closeList()
			else
				openList()
			end
		end)
		dropdownPaints[#dropdownPaints + 1] = function()
			button.Text = tostring(get()) .. "  v"
		end
		return button
	end

	local function paintMeter(bars, hist, accent)
		local peak = 1
		for _, sample in hist do
			if sample > peak then
				peak = sample
			end
		end
		for index, bar in bars do
			local sample = hist[index]
			local height = sample and math.clamp(sample / peak, 0.15, 1) or 0.18
			local tInfo = TweenInfo.new(0.25, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
			TweenService:Create(bar, tInfo, { Size = UDim2.new(0, 4, height, 0) }):Play()
			if index == #bars and sample then
				TweenService:Create(bar, tInfo, { BackgroundColor3 = accent }):Play()
			else
				local alpha = index / #bars
				local faded = Color3.fromRGB(
					math.floor(40 + (accent.R * 255 - 40) * alpha * 0.45),
					math.floor(40 + (accent.G * 255 - 40) * alpha * 0.45),
					math.floor(48 + (accent.B * 255 - 48) * alpha * 0.45)
				)
				bar.BackgroundColor3 = faded
			end
		end
	end

	track(RunService.Heartbeat:Connect(function(dt)
		fpsSmooth = fpsSmooth * 0.85 + (1 / math.max(dt, 1 / 240)) * 0.15
		clock = clock + dt

			return
		end

	local function applyData(data)
		for key, value in data.features or {} do
			if Config[key] ~= nil and type(value) == type(Config[key]) then
				Config[key] = value
			end
		end
		if type(data.moves) == "table" then
			for styleName, delays in data.moves do
				local moves = MOVES[styleName]
				if moves and type(delays) == "table" then
					for index, delay in delays do
						local slot = tonumber(index) or index
						if moves[slot] and type(delay) == "number" then
							moves[slot].Delay = delay
						end
					end
				end
			end
		end
		if type(data.styles) == "table" then
			for _, style in STYLES do
				if type(data.styles[style.Name]) == "boolean" then
					style.On = data.styles[style.Name]
				end
			end
		end
		if not Config.AutoParry then
			pcall(releaseParryKey)
		end
		Ui.Theme = data.theme or Ui.Theme
		Ui.Scale = data.scale or Ui.Scale
		Ui.Radius = data.radius or Ui.Radius
		Ui.Transparent = data.transparent or false
		Ui.Dim = data.dim or false
		Ui.Weather = data.weather or "Off"
		Ui.WeatherMode = data.weatherMode or "Screen"
		Ui.Decor = data.decor or DEFAULT_DECOR
		local savedGlass = data.glass
		if type(savedGlass) == "boolean" then
			savedGlass = savedGlass and "Strong" or "Off"
		end
		Ui.Glass = savedGlass or DEFAULT_GLASS
		Ui.Drag = data.drag or "Fill"
		Ui.Density = data.density or "Default"
		Ui.MenuButton = data.menuButton or "Always"
		if data.hide and Enum.KeyCode[data.hide] then
			hideKey = Enum.KeyCode[data.hide]
		end
		applyChrome()
		paintAll()
	end

	search:GetPropertyChangedSignal("Text"):Connect(function()

		for name, pack in pageButtons do

	local dragging = false
	local dragStart, startPos
	local function beginDrag(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end
		dragging = true
		dragStart = UserInputService:GetMouseLocation()
		startPos = main.Position
	end
	title.InputBegan:Connect(beginDrag)
	pageTitle.InputBegan:Connect(beginDrag)

	local menuDragging = false
	local menuStart, menuPos, menuMoved
	menuButton.InputBegan:Connect(function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end
		menuDragging = true
		menuMoved = false
		menuStart = UserInputService:GetMouseLocation()
		menuPos = menuButton.Position
	end)
	menuButton.InputEnded:Connect(function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end
		if menuDragging and not menuMoved then
			setMenu(not main.Visible)
		end
		menuDragging = false
	end)
	track(UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end))
	track(RunService.RenderStepped:Connect(function(dt)
		if menuDragging and menuStart then
			local delta = UserInputService:GetMouseLocation() - menuStart
			if delta.Magnitude > 8 then
				menuMoved = true
			end
			menuButton.Position = UDim2.new(menuPos.X.Scale, menuPos.X.Offset + delta.X, menuPos.Y.Scale, menuPos.Y.Offset + delta.Y)
		end
		if dragging and dragStart then
			local delta = UserInputService:GetMouseLocation() - dragStart
			main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
			if Ui.Drag == "Outline" then
				main.BackgroundTransparency = 0.45
			end
		elseif Ui.Drag == "Outline" then
			main.BackgroundTransparency = Ui.Transparent and 0.28 or 0
		end
		if Ui.MenuButton == "Hover" then
			local mouse = UserInputService:GetMouseLocation()
			local spot = menuButton.AbsolutePosition
			local near = mouse.X < 70 or (mouse - spot).Magnitude < 80
			menuButton.Visible = near or not main.Visible
		end
		local layerOn = (Ui.Weather ~= "Off" or Ui.Decor ~= "Off") and main.Visible
		if layerOn ~= snowLayer.Visible then
			snowLayer.Visible = layerOn
		end
		if layerOn ~= decorLayer.Visible then
			decorLayer.Visible = layerOn
		end
		syncWeather()
		syncDecor()
		if layerOn then
			local viewW = view.X
			local viewH = view.Y
			if Ui.WeatherMode == "Menu" then
				snowLayer.Size = UDim2.fromOffset(main.AbsoluteSize.X, main.AbsoluteSize.Y)
				snowLayer.Position = UDim2.fromOffset(main.AbsolutePosition.X, main.AbsolutePosition.Y)
				snowLayer.ZIndex = main.ZIndex + 1
				decorLayer.Size = UDim2.fromOffset(main.AbsoluteSize.X, main.AbsoluteSize.Y)
				decorLayer.Position = UDim2.fromOffset(main.AbsolutePosition.X, main.AbsolutePosition.Y)
				decorLayer.ZIndex = main.ZIndex + 2
				viewW = main.AbsoluteSize.X
				viewH = main.AbsoluteSize.Y
			else
				snowLayer.Size = UDim2.fromScale(1, 1)
				snowLayer.Position = UDim2.fromOffset(0, 0)
				snowLayer.ZIndex = 1
				decorLayer.Size = UDim2.fromScale(1, 1)
				decorLayer.Position = UDim2.fromOffset(0, 0)
				decorLayer.ZIndex = 1
			end
			if snowLayer.AbsoluteSize.X > 10 and snowLayer.AbsoluteSize.Y > 10 then
				viewW = snowLayer.AbsoluteSize.X
				viewH = snowLayer.AbsoluteSize.Y
			end
			local width = math.max(viewW, 1)
			local height = math.max(viewH, 1)
			local decorOn = Ui.Decor ~= "Off"
			if decorOn then
				for _, part in decorParts do
					part.y = part.y + part.speed * dt
					part.phase = part.phase + dt * part.pulse
					if part.y > 1.08 then
						part.y = -0.08
						part.x = math.random()
						part.phase = math.random() * math.pi * 2
					elseif part.y < -0.08 then
						part.y = 1.08
						part.x = math.random()
					end
					if part.spin ~= 0 then
						part.frame.Rotation = (part.frame.Rotation + part.spin * dt) % 360
					end
					local drift = math.sin(part.phase) * part.sway
					part.frame.Position = UDim2.fromOffset((part.x + drift) * width, part.y * height)
					if part.twinkle then
						part.group.GroupTransparency = math.clamp(
							part.base + math.sin(part.phase * 2) * 0.28, 0, 1)
					end
				end
			end
			for _, flake in flakes do
				flake.y = flake.y + flake.speed * dt
				flake.x = flake.x + flake.vx * dt
				flake.phase = flake.phase + dt * flake.pulse
				if flake.spin ~= 0 then
					flake.frame.Rotation = (flake.frame.Rotation + flake.spin * dt) % 360
				end
				if flake.y > 1.04 then
					flake.y = -0.04
					flake.x = math.random()
					flake.phase = math.random() * math.pi * 2
				elseif flake.y < -0.04 then
					flake.y = 1.04
					flake.x = math.random()
					flake.phase = math.random() * math.pi * 2
				end
				local drift = math.sin(flake.phase) * flake.sway
				local px = (flake.x + drift) * width
				if px < -20 then
					flake.x = flake.x + 1.05
				elseif px > width + 20 then
					flake.x = flake.x - 1.05
				end
				flake.frame.Position = UDim2.fromOffset((flake.x + drift) * width, flake.y * height)
				if flake.twinkle then
					local swing = flake.strongTwinkle and 0.55 or 0.3
					flake.frame.BackgroundTransparency = math.clamp(
						flake.base + math.sin(flake.phase * 2) * swing, 0, 1)
				end
			end
		end
	end))
	track(UserInputService.InputBegan:Connect(function(input, processed)
		if input.UserInputType ~= Enum.UserInputType.Keyboard then
			return
		end
		if bindTarget then
			if bindTarget.hide then
				hideKey = input.KeyCode
				bindTarget = nil
				paintAll()
				notify("Hide key set")
				return
			end
			if input.KeyCode == Enum.KeyCode.Backspace then
				bindTarget.key = nil
				bindTarget = nil
				paintAll()
				notify("Hotkey cleared")
				return
			end
			bindTarget.key = input.KeyCode
			bindTarget = nil
			paintAll()
			notify("Key set")
			return
		end
		if input.KeyCode == hideKey then
			setMenu(not main.Visible)
			return
		end
		if processed or UserInputService:GetFocusedTextBox() then
			return
		end
		local entry = hotkeyMap[input.KeyCode]
		if entry then
			entry.flip()
		end
	end))

	paintAll()
	local autoload = readStore().autoload
	if autoload ~= "" and readStore().configs[autoload] then
		applyData(readStore().configs[autoload])
		notify("Autoload " .. autoload)
	end
	syncWeather()
	syncDecor()
	main.Visible = false
	menuButton.Text = "B"
	applyMouse(false)
	snowLayer.Visible = false
	decorLayer.Visible = Ui.Decor ~= "Off" and main.Visible
	dimmer.BackgroundTransparency = Ui.Dim and 0.45 or 1
	-- Public API, usable from other scripts:
	--   BunnyHub.Open()            -> true if it opened
	--   BunnyHub.Close()           -> true if it closed
	--   BunnyHub.SetPage("Runner") -> true if the page exists
	--   BunnyHub.SetTheme("Mint")  -> true if the theme exists
	--   BunnyHub.IsOpen()          -> boolean
	--   BunnyHub.GetConfig()       -> the live settings table
	local api = {}

	local function guarded(fn)
		return function(...)
			if not running then
				return false
			end
			return fn(...)
		end
	end

	api.Open = guarded(function()
		if main.Visible then
			return false
		end
		setMenu(true)
		return true
	end)

	api.Close = guarded(function()
		if not main.Visible then
			return false
		end
		setMenu(false)
		return true
	end)

	api.SetPage = guarded(function(name)
		if type(name) ~= "string" or not pageButtons[name] then
			return false
		end
		showPage(name)
		return true
	end)

	api.SetTheme = guarded(function(name)
		if type(name) ~= "string" or not THEMES[name] then
			return false
		end
		Ui.Theme = name
		applyChrome()
		return true
	end)

	api.SetWeather = guarded(function(name)
		if type(name) ~= "string" or (name ~= "Off" and not WEATHER[name]) then
			return false
		end
		Ui.Weather = name
		applyChrome()
		return true
	end)

	api.SetDecor = guarded(function(name)
		if type(name) ~= "string" or (name ~= "Off" and not DECOR[name]) then
			return false
		end
		Ui.Decor = name
		applyChrome()
		return true
	end)

	api.IsOpen = function()
		return running and main.Visible == true
	end

	api.GetConfig = function()
		return Ui
	end

	api.Stop = function()
		shutdown()
		return true
	end

	_G.BunnyHub = api

	showPage("Home")
	notify("Bunny Hub ready")
end


shutdown = function()
	running = false
	Config.SpeedOn = false
	Config.CFrameSpeed = false
	Config.JumpOn = false
	Config.Noclip = false
	Config.Fly = false
	Config.Bright = false
	Config.Freecam = false
	Config.Zoom = false
	pcall(applyMove, 0)
	pcall(applyWorld)
	pcall(clearVisuals)
	if camState then
		local camera = workspace.CurrentCamera
		if camera then
			pcall(function()
				camera.CameraType = camState
			end)
		end
		camState = nil
	end
	for _, conn in connections do
		pcall(function()
			conn:Disconnect()
		end)
	end
	if gui then
		gui:Destroy()
	end
end

_G.BunnyUIStop = shutdown

local lastScan = 0
local lastAuto = 0
track(RunService.Heartbeat:Connect(function(dt)
	if not running then
		return
	end
	local now = os.clock()
	if Config.Noclip or Config.FovOn or Config.JumpOn then
		pcall(applyMove, dt)
	end
	if Config.Bright or Config.NoBlur or lightBackup then
		pcall(applyWorld)
	end
	pcall(stepCombat)
	if Config.Freecam or camState then
		pcall(updateFreecam, dt)
	end
	if now - lastScan >= (Config.Scan or 0.5) then
		lastScan = now
		pcall(scanWorld)
	end
end))
