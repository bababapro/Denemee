local function GetExecutorName()
    if identifyexecutor then
        local ok, name = pcall(identifyexecutor)
        if ok and name and tostring(name) ~= "" then
            return tostring(name)
        end
    end
    if getexecutorname then
        local ok, name = pcall(getexecutorname)
        if ok and name and tostring(name) ~= "" then
            return tostring(name)
        end
    end
    if RunService:IsStudio() then
        return "Roblox Studio"
    end
    return "Unknown"
end

local function GetPlatformName()
    if UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled then
        return "Mobile"
    elseif UserInputService.GamepadEnabled and not UserInputService.KeyboardEnabled then
        return "Console"
    else
        return "PC / Windows"
    end
end

local function ValidateKey(inputKey, validKeys)
    for _, validKey in pairs(validKeys) do
        if inputKey == validKey then
            return true
        end
    end
    return false
end

local function Create(className, properties, parent)
    local instance = Instance.new(className)
    for property, value in pairs(properties) do
        instance[property] = value
    end
    if parent then
        instance.Parent = parent
    end
    return instance
end

local function GetGameIconUrl(placeId)
    if not placeId or tonumber(placeId) == 0 then
        return ""
    end
    return "rbxthumb://type=GameIcon&id=" .. tostring(placeId) .. "&w=150&h=150"
end

local function LaunchFireworkOnly(Stage, launchX, burstY, accentColor)
    if not Stage.Parent then return end

    local Rocket = Create("Frame", {
        Size = UDim2.new(0, 5, 0, 16),
        Position = UDim2.new(launchX, -2.5, 1.02, 0),
        BackgroundColor3 = Color3.fromRGB(250, 252, 255),
        BorderSizePixel = 0,
        ZIndex = 82
    }, Stage)
    Create("UICorner", {CornerRadius = UDim.new(1, 0)}, Rocket)
    Create("UIStroke", {Color = accentColor, Thickness = 1.5}, Rocket)

    task.spawn(function()
        while Rocket.Parent and Rocket.Position.Y.Scale > burstY + 0.03 do
            local trail = Create("Frame", {
                Size = UDim2.new(0, 4, 0, 4),
                Position = UDim2.new(Rocket.Position.X.Scale, 0, Rocket.Position.Y.Scale + 0.02, 0),
                BackgroundColor3 = accentColor,
                BackgroundTransparency = 0.25,
                BorderSizePixel = 0,
                ZIndex = 81
            }, Stage)
            Create("UICorner", {CornerRadius = UDim.new(1, 0)}, trail)

            TweenService:Create(trail, TweenInfo.new(0.6, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Size = UDim2.new(0, 1, 0, 1),
                BackgroundTransparency = 1
            }):Play()

            task.delay(0.65, function()
                if trail.Parent then trail:Destroy() end
            end)
            task.wait(0.08)
        end
    end)

    local riseTween = TweenService:Create(Rocket, TweenInfo.new(1.6, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Position = UDim2.new(launchX, -2.5, burstY, 0)
    })
    riseTween:Play()
    riseTween.Completed:Wait()

    if not Stage.Parent then return end
    if Rocket.Parent then Rocket:Destroy() end

    local sparkColors = {
        accentColor,
        Color3.fromRGB(255, 65, 65),
        Color3.fromRGB(255, 215, 80),
        Color3.fromRGB(90, 235, 160),
        Color3.fromRGB(245, 252, 255)
    }

    local sparkCount = 16
    for i = 1, sparkCount do
        local angle = ((i - 1) / sparkCount) * math.pi * 2
        local distX = math.cos(angle) * 0.13
        local distY = math.sin(angle) * 0.13
        local sColor = sparkColors[(i % #sparkColors) + 1]

        local Spark = Create("Frame", {
            Size = UDim2.new(0, 7, 0, 7),
            Position = UDim2.new(launchX, -3.5, burstY, -3.5),
            BackgroundColor3 = sColor,
            BorderSizePixel = 0,
            ZIndex = 83
        }, Stage)
        Create("UICorner", {CornerRadius = UDim.new(1, 0)}, Spark)

        TweenService:Create(Spark, TweenInfo.new(0.95, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
            Position = UDim2.new(math.clamp(launchX + distX, 0.04, 0.96), -3.5, math.clamp(burstY + distY, 0.05, 0.85), -3.5),
            Size = UDim2.new(0, 2, 0, 2),
            BackgroundTransparency = 0.9
        }):Play()

        task.delay(1.0, function()
            if Spark.Parent then Spark:Destroy() end
        end)
    end
end

            mDragging = true; mMoved = false; mDragStart = input.Position; mStartPos = MinimizedBtn.Position
            input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then mDragging = false end end)
        end
    end)
    MinimizedBtn.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            mDragInput = input
        end
    end)

        if input == mDragInput and mDragging then
            local delta = input.Position - mDragStart
            if delta.Magnitude > 4 then mMoved = true end
            MinimizedBtn.Position = UDim2.new(mStartPos.X.Scale, mStartPos.X.Offset + delta.X, mStartPos.Y.Scale, mStartPos.Y.Offset + delta.Y)
        end
    end)

        if inputKey == "" then
            setBannerStatus("Empty key", "Please enter a license key first", Color3.fromRGB(235, 75, 75))
            local origInputPos = InputFrame.Position
            for i = 1, 6 do
                InputFrame.Position = origInputPos + UDim2.new(0, (i % 2 == 0 and 8 or -8), 0, 0)
                task.wait(0.04)
            end
            InputFrame.Position = origInputPos
            return
        end

        isVerifying = true
        RedeemTitleLbl.Text = "Verifying..."
        setBannerStatus("Checking license", "Validating your key with Kryvex servers...", Color3.fromRGB(245, 185, 65))
        task.wait(0.75)

        if ValidateKey(inputKey, Keys) then
            RedeemTitleLbl.Text = "Verified!"
            StatusTitle.Text = "Access Granted"
            StatusSub.Text = "Welcome to Kryvex Hub! Loading interface..."
            StatusIndicatorDot.BackgroundColor3 = Color3.fromRGB(65, 225, 130)
            ShowNotification("Key verified! Loading Kryvex Hub...")
            task.wait(0.9)
            KeySystemGui:Destroy()
            MainScriptCallback(true)
        else
            RedeemTitleLbl.Text = "Redeem Key"
            setBannerStatus("Invalid key", "The key you entered is incorrect or expired", Color3.fromRGB(235, 75, 75))
            local origPos = KeyContainer.Position
            for i = 1, 8 do
                KeyContainer.Position = origPos + UDim2.new(0, (i % 2 == 0 and 12 or -12), 0, 0)
                task.wait(0.04)
            end
            KeyContainer.Position = origPos
        end
        isVerifying = false
    end)

    TweenService:Create(KeyContainer, TweenInfo.new(0.65, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {Size = KContainerSize}):Play()
end
