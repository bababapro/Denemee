-- boot -----------------------------------------------------------------------------------------
-- A second run has to take the first one down before Fluent is loaded again, or the earlier
-- window stays in CoreGui with its own input handlers. Every loop also carries a generation
-- number, so the earlier run's threads stop even if its library never reported itself unloaded.
if not game:IsLoaded() then
    game.Loaded:Wait()
end

loadstring("function LPH_NO_VIRTUALIZE(f) return f end;")()

local Boot = { Gen = 0, Env = (type(getgenv) == "function" and getgenv()) or _G }

-- `shared` is a Roblox global, but a sandbox or a stripped executor may not expose it. Resolve it
-- once into a local so nothing below has to care which it was.
local Shared = (type(shared) == "table" and shared)
    or (type(Boot.Env.shared) == "table" and Boot.Env.shared)
    or {}
Boot.Env.shared = Shared

do
    local Env = Boot.Env

    local Old = rawget(Env, "AxonForestGui")
    if type(Env.AxonForestUnload) == "function" then
        pcall(Env.AxonForestUnload)
    end
    if typeof(Old) == "Instance" then
        pcall(function()
            Old:Destroy()
        end)
    end
    Env.AxonForestGui = nil

    -- and anything still carrying our own tag, whatever the previous run managed to do before it
    -- died. Only guis this hub tagged are touched, never the game's or another script's.
    pcall(function()
        if type(setthreadidentity) == "function" then
            pcall(setthreadidentity, 8)
        end
        local Roots = {}
        if type(gethui) == "function" then
            local Ok, Hidden = pcall(gethui)
            if Ok and Hidden then
                table.insert(Roots, Hidden)
            end
        end
        local Ok, Core = pcall(game.GetService, game, "CoreGui")
        if Ok and Core then
            table.insert(Roots, Core)
        end
        local Player = game:GetService("Players").LocalPlayer
        local Gui = Player and Player:FindFirstChild("PlayerGui")
        if Gui then
            table.insert(Roots, Gui)
        end
        for _, Root in Roots do
            for _, Node in Root:GetChildren() do
                if Node:IsA("ScreenGui") and Node:GetAttribute("AxonForest") == true then
                    pcall(function()
                        Node:Destroy()
                    end)
                end
            end
        end
    end)

    Env.AxonForestGen = (tonumber(rawget(Env, "AxonForestGen")) or 0) + 1
    Boot.Gen = Env.AxonForestGen
    Shared.NightsInTheForest = true
    task.wait(0.15)
end

-- Fluent creates Instances, which needs the Plugin capability. The farm threads deliberately run
-- at identity 2 so the game's own reads behave, and a signal connected from one of those threads
-- runs its callback at that identity too, so anything that touches the UI raises first.
local function RaiseIdentity()
    if type(Compat.Identity) == "function" then
        pcall(Compat.Identity, 8)
    end
end

-- fireproximityprompt is not on every executor: drive the prompt through its own API instead,
-- and keep the real hold duration so a prompt is never completed faster than the client could.
Compat.FirePrompt = LPH_NO_VIRTUALIZE(function(Prompt, Instant)
    if typeof(Prompt) ~= "Instance" or not Prompt:IsA("ProximityPrompt") then
        return false
    end
    if type(fireproximityprompt) == "function" then
        return (pcall(fireproximityprompt, Prompt))
    end
    return (pcall(function()
        local Hold = Prompt.HoldDuration
        if Instant then
            Prompt.HoldDuration = 0
        end
        Prompt:InputHoldBegin()
        if Prompt.HoldDuration > 0 then
            task.wait(Prompt.HoldDuration + 0.05)
        end
        Prompt:InputHoldEnd()
        Prompt.HoldDuration = Hold
    end))
end)

-- isnetworkowner is executor specific; fall back to the higher level variant, then to false
Compat.NetworkOwner = LPH_NO_VIRTUALIZE(function(Part)
    if typeof(Part) ~= "Instance" then
        return false
    end
    if type(isnetworkowner) == "function" then
        local Ok, Owned = pcall(isnetworkowner, Part)
        return Ok and Owned or false
    end
    if type(GetNetworkOwner) == "function" then
        local Ok, Owner = pcall(GetNetworkOwner, Part)
        return Ok and Owner == LocalPlayer
    end
    return false
end)

-- skin -----------------------------------------------------------------------------------------
-- Every piece of UI the hub draws itself - the loader, the mobile handle, the stop card, the
-- health plate, the exploration meter, the esp tags - is built from this one palette and these
-- few builders. Nothing here fetches an asset id: panels, glyphs and the wordmark are drawn out
-- of Frames, so there is no image to load and no one else's branding on screen.
local Skin = {
    Ink = Color3.fromRGB(12, 13, 17),
    Panel = Color3.fromRGB(22, 24, 31),
    Raise = Color3.fromRGB(32, 35, 45),
    Edge = Color3.fromRGB(58, 63, 78),
    Text = Color3.fromRGB(232, 234, 240),
    Dim = Color3.fromRGB(138, 144, 162),
    Accent = Color3.fromRGB(255, 138, 30),
    Deep = Color3.fromRGB(176, 84, 12),
    Good = Color3.fromRGB(61, 220, 132),
    Warn = Color3.fromRGB(255, 77, 94),
    Cool = Color3.fromRGB(77, 163, 255),
    Face = Enum.Font.Gotham,
    Mid = Enum.Font.GothamMedium,
    Bold = Enum.Font.GothamBold,
}

-- one constructor for every node, so a panel is three lines instead of fifteen
Skin.New = LPH_NO_VIRTUALIZE(function(Class, Props, Parent)
    local Node = Instance.new(Class)
    for Key, Value in Props or {} do
        Node[Key] = Value
    end
    if Parent then
        Node.Parent = Parent
    end
    return Node
end)

Skin.Round = LPH_NO_VIRTUALIZE(function(Node, Radius)
    return Skin.New("UICorner", { CornerRadius = Radius or UDim.new(0, 10) }, Node)
end)

Skin.Stroke = LPH_NO_VIRTUALIZE(function(Node, Colour, Thickness, Clear)
    return Skin.New("UIStroke", {
        Color = Colour or Skin.Edge,
        Thickness = Thickness or 1,
        Transparency = Clear or 0.35,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, Node)
end)

Skin.Pad = LPH_NO_VIRTUALIZE(function(Node, X, Y)
    return Skin.New("UIPadding", {
        PaddingLeft = UDim.new(0, X or 10),
        PaddingRight = UDim.new(0, X or 10),
        PaddingTop = UDim.new(0, Y or X or 10),
        PaddingBottom = UDim.new(0, Y or X or 10),
    }, Node)
end)

-- a two-stop gradient, the only decoration used anywhere in the hub's own UI
Skin.Gradient = LPH_NO_VIRTUALIZE(function(Node, From, To, Rotation)
    return Skin.New("UIGradient", {
        Color = ColorSequence.new(From or Skin.Accent, To or Skin.Deep),

    end
    pcall(function()

            BackgroundTransparency = 1,
            BorderSizePixel = 0,
        }, Gui)

        -- amber blob, no asset id, nothing uploaded
        local Card = Skin.New("Frame", {
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromScale(0.5, 0.5),
            Size = UDim2.fromOffset(360, 150),
            BackgroundColor3 = Skin.Panel,
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
            ZIndex = 100,
        }, Scrim)
        Skin.Round(Card, UDim.new(0, 18))
        local Edge = Skin.Stroke(Card, Skin.Edge, 1, 1)

        -- a single accent chip on the left shoulder of the card, the hub's one flash of colour
        local Chip = Skin.New("Frame", {
            Position = UDim2.fromOffset(26, 26),
            Size = UDim2.fromOffset(10, 10),
            BackgroundColor3 = Skin.Accent,
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
            ZIndex = 102,
        }, Card)
        Skin.Round(Chip, UDim.new(0.5, 0))

        local Kicker = Skin.Label(Card, {
            Position = UDim2.fromOffset(46, 18),
            Size = UDim2.fromOffset(180, 26),
            Font = Skin.Mid,
            Text = "AXON HUB",
            TextColor3 = Skin.Dim,
            TextSize = 13,
            TextTransparency = 1,
            TextXAlignment = Enum.TextXAlignment.Left,
            ZIndex = 102,
        })

        local MarkHolder = Skin.New("Frame", {
            Position = UDim2.fromOffset(24, 50),
            Size = UDim2.fromOffset(312, 46),
            BackgroundTransparency = 1,
            ZIndex = 102,
        }, Card)
        local _, Letters = Skin.Wordmark(MarkHolder, "99 NIGHTS", 26, Skin.Text)

        -- the loading rule: a thin track that fills once, left to right
        local Track, Fill = Skin.Meter(Card, {
            Position = UDim2.fromOffset(24, 112),
            Size = UDim2.fromOffset(312, 4),
            BackgroundTransparency = 1,
            ZIndex = 102,
        })
        Fill.BackgroundTransparency = 1

        local function Fade(Object, Time, Goal)
            Skin.Tween(Object, Time, Goal)
        end

        Fade(Scrim, 0.45, { BackgroundTransparency = 0.35 })
        task.delay(0.2, function()
            Fade(Card, 0.45, { BackgroundTransparency = 0.05 })
            Fade(Edge, 0.45, { Transparency = 0.45 })
            Fade(Chip, 0.45, { BackgroundTransparency = 0 })
            Fade(Kicker, 0.5, { TextTransparency = 0 })
            Fade(Track, 0.5, { BackgroundTransparency = 0 })
            Fade(Fill, 0.5, { BackgroundTransparency = 0 })
            -- the caps arrive one after another, which reads as typed rather than popped
            for Index, Cap in Letters do
                task.delay(0.03 * Index, function()
                    Fade(Cap, 0.35, { TextTransparency = 0 })
                end)
            end
            Fade(Fill, (Duration or 1) + 0.4, { Size = UDim2.fromScale(1, 1) })

            task.wait(Duration or 1)
            Fade(Chip, 0.4, { BackgroundTransparency = 1 })
            Fade(Kicker, 0.4, { TextTransparency = 1 })
            Fade(Track, 0.4, { BackgroundTransparency = 1 })
            Fade(Fill, 0.4, { BackgroundTransparency = 1 })
            for _, Cap in Letters do
                Fade(Cap, 0.4, { TextTransparency = 1 })
            end
            Fade(Edge, 0.5, { Transparency = 1 })
            Fade(Card, 0.5, { BackgroundTransparency = 1, Size = UDim2.fromOffset(392, 150) })
            Fade(Scrim, 0.7, { BackgroundTransparency = 1 })
            task.wait(0.9)
            Gui:Destroy()
        end)
    end)
end

-- library --------------------------------------------------------------------------------------
-- Fluent 1.1.0 and its addons, pinned and loaded with retries. The adapter below is the only
-- place that knows it is Fluent: every feature builds controls through this element API, so the
-- window can be swapped without touching a single feature.
local Fluent, SaveManager, InterfaceManager

local function LoadTimed(Callback, Seconds)
    local Done, Value = nil, nil
    task.spawn(function()
        local Ok, Result = pcall(Callback)
        Value = Result
        Done = Ok
    end)
    local Deadline = os.clock() + (Seconds or 15)
    while Done == nil and os.clock() < Deadline do
        task.wait(0.05)
    end
    if Done == nil then
        return false, "timed out"
    end
    return Done, Value
end

do
    local function Fetch()
        Fluent = loadstring(game:HttpGet("https://github.com/dawid-scripts/Fluent/releases/download/1.1.0/main.lua", true))()
        SaveManager = loadstring(game:HttpGet("https://raw.githubusercontent.com/dawid-scripts/Fluent/78ba067d6f927fe45557c79da05853331b89480e/Addons/SaveManager.lua", true))()
        InterfaceManager = loadstring(game:HttpGet("https://raw.githubusercontent.com/dawid-scripts/Fluent/78ba067d6f927fe45557c79da05853331b89480e/Addons/InterfaceManager.lua", true))()
    end

    for _ = 1, 5 do
        local Ok = LoadTimed(Fetch, 15)
        if Ok and Fluent then
            break
        end
        task.wait(1)
    end
end

if not Fluent then

end

-- ui adapter -----------------------------------------------------------------------------------
-- Nothing in a shipped hub writes to the console: F9 is visible to anyone and print is hookable.
-- Faults land here, and the Misc tab's Activity list shows them.
local Library = {
    Faults = {},
    Unloaded = false,
    Options = {},
    Toggles = {},
    Flags = {},
    Signals = {},
    UnloadHooks = {},
    Count = 0,
}

function Library.Fault(Where, Problem)
    local Line = ("%s: %s"):format(tostring(Where), tostring(Problem))
    table.insert(Library.Faults, 1, Line)
    if #Library.Faults > 40 then
        table.remove(Library.Faults)
    end
    return Line
end

function Library:GiveSignal(Signal)
    table.insert(self.Signals, Signal)
    return Signal
end

function Library:Connect(Signal, Callback)
    return self:GiveSignal(Signal:Connect(Callback))
end

function Library:OnUnload(Callback)
    table.insert(self.UnloadHooks, Callback)
end

function Library:Notify(Title, Content, Duration)
    RaiseIdentity()
    return Fluent:Notify({
        Title = tostring(Title or "Axon Hub"),
        Content = Content and tostring(Content) or "",
        Duration = Duration or 5,
    })
end

function Library:Unload()
    if self.Unloaded then
        return
    end
    self.Unloaded = true
    for _, Hook in self.UnloadHooks do
        pcall(Hook)
    end
    for _, Signal in self.Signals do
        pcall(function()
            Signal:Disconnect()
        end)
    end
    table.clear(self.Signals)
    table.clear(self.UnloadHooks)
    pcall(function()
        Fluent:Destroy()
    end)
end

local function NextId(Prefix)
    Library.Count += 1
    return ("%s_%d"):format(Prefix, Library.Count)
end

-- a multi dropdown reads back as an array at runtime but a Default or a loaded config can hand
-- over a { [name] = true } map, so every list goes through here
-- Always a fresh array, never the caller's table. Fluent keeps whatever it is handed as the
-- element's own Values and Value, and for a multi dropdown it writes Value[name] = true straight
-- into it. Passing the same World table as both Values and Default therefore grew string keys on
-- that array, so the next list build read a boolean where a name should be ("Unable to assign
-- property Text") and the World table stayed corrupted for every other reader.
local function ValueList(Values)
    local Out = {}
    if type(Values) ~= "table" then
        return Out
    end
    if #Values > 0 then
        for _, Value in Values do
            if type(Value) == "string" then
                table.insert(Out, Value)
            end
        end
        return Out
    end
    for Key, On in pairs(Values) do
        if On ~= false and type(Key) == "string" then
            table.insert(Out, Key)
        end
    end
    return Out
end

-- A multi dropdown's default is a plain array of names. Nil has to become an empty table rather
-- than nil, and the result is always a copy, for the reason above.
local function MultiDefault(Default)
    if Default == nil then
        return {}
    end
    if type(Default) ~= "table" then
        return { Default }
    end
    return ValueList(Default)
end

-- Element wrapper. Flags mirror every value into Library.Flags so a feature can read state
-- without holding the element, and OnChanged keeps the immediate-call semantics features expect
-- from a late listener.
local function Wrap(Element, Host, Flag)
    local Self = { Element = Element, Host = Host, Flag = Flag, Listeners = {} }

    if Flag then
        Library.Flags[Flag] = Element.Value
    end

    local function Fire(Value)
        if Flag then
            Library.Flags[Flag] = Element.Value
        end
        for _, Listener in Self.Listeners do
            local Ok, Error = pcall(Listener, Value)
            if not Ok then
                Library.Fault(Flag, Error)
            end
        end
    end

    -- Fluent keeps exactly one change handler per control (Element.Changed), and the hub claims it
    -- here for EVERY control, whether or not a feature attaches a listener later. Binding it
    -- lazily on the first :OnChanged left every control that only passes a Callback - all sixteen
    -- scheduler toggles among them - with its Flags entry frozen at the default, so the job never
    -- saw the switch go on. Fluent also calls the handler once on registration, which just
    -- re-reads the default.
    if type(Element.OnChanged) == "function" then
        pcall(function()
            Element:OnChanged(Fire)
        end)
    end

    function Self:OnChanged(Callback)
        table.insert(self.Listeners, Callback)
        -- matches the old behaviour: Fluent invoked a freshly bound handler straight away
        local Ok, Error = pcall(Callback, Element.Value)
        if not Ok then
            Library.Fault(Flag, Error)
        end
        return self
    end

    function Self:SetValue(Value)
        RaiseIdentity()
        if Element.Type == "Colorpicker" and Element.SetValueRGB and typeof(Value) == "Color3" then
            Element:SetValueRGB(Value)
        elseif Element.SetValue then
            Element:SetValue(Value)
        end
        if Flag then
            Library.Flags[Flag] = Element.Value
        end
        return self
    end

    function Self:SetValues(Values)
        RaiseIdentity()
        if Element.SetValues then
            Element:SetValues(ValueList(Values))
        end
        return self
    end

    function Self:SetText(Text)
        RaiseIdentity()
        if Element.SetTitle then
            Element:SetTitle(tostring(Text))
        end
        return self
    end

    function Self:SetDesc(Text)
        RaiseIdentity()
        if Element.SetDesc then
            Element:SetDesc(tostring(Text))
        end
        return self
    end

    function Self:Toggle()
        return self:SetValue(not Element.Value)
    end

    function Self:AddColorPicker(Index, Config)
        return Host:AddColorPicker(Index, Config)
    end

    function Self:AddKeyPicker(Index, Config)
        return Host:AddKeyPicker(Index, Config, self)
    end

    return setmetatable(Self, {
        __index = function(_, Key)
            if Key == "Value" then
                return Element.Value
            end
            return Element[Key]
        end,
    })
end

-- Element host, shared by tabs and groupboxes.
local function Host(Container)
    local Self = { Container = Container }

    function Self:AddToggle(Flag, Config)
        Config = Config or {}
        Flag = Flag or NextId("Toggle")
        local Element = Wrap(Container:AddToggle(Flag, {
            Title = tostring(Config.Text or Flag),
            Description = Config.Tooltip or Config.Desc,
            Default = Config.Default and true or false,
            Callback = Config.Callback,
        }), self, Flag)
        Library.Options[Flag] = Element
        Library.Toggles[Flag] = Element
        return Element
    end

    function Self:AddButton(First, Second, Third)
        local Text, Callback, Tooltip
        if type(First) == "table" then
            Text, Callback, Tooltip = First.Text, First.Func or First.Callback, First.Tooltip
        else
            Text, Callback, Tooltip = First, Second, Third
        end
        return Wrap(Container:AddButton({
            Title = tostring(Text or "Button"),
            Description = Tooltip,
            Callback = Callback,
        }), self)
    end

    function Self:AddLabel(Text, Content)
        return Wrap(Container:AddParagraph({
            Title = tostring(Text or ""),
            Content = tostring(Content or ""),
        }), self)
    end

    function Self:AddDropdown(Flag, Config)
        Config = Config or {}
        Flag = Flag or NextId("Dropdown")
        local Multi = Config.Multi and true or false
        local Element = Wrap(Container:AddDropdown(Flag, {
            Title = tostring(Config.Text or Flag),
            Description = Config.Tooltip,
            Values = ValueList(Config.Values),
            Multi = Multi,
            Default = Multi and MultiDefault(Config.Default) or Config.Default,
            Callback = Config.Callback,
        }), self, Flag)
        Library.Options[Flag] = Element
        return Element
    end

    function Self:AddSlider(Flag, Config)
        Config = Config or {}
        Flag = Flag or NextId("Slider")
        local Element = Wrap(Container:AddSlider(Flag, {
            Title = tostring(Config.Text or Flag),
            Description = Config.Tooltip,
            Default = Config.Default or Config.Min or 0,
            Min = Config.Min or 0,
            Max = Config.Max or 100,
            Rounding = Config.Rounding,
            Callback = Config.Callback,
        }), self, Flag)
        Library.Options[Flag] = Element
        return Element
    end

    function Self:AddInput(Flag, Config)
        Config = Config or {}
        Flag = Flag or NextId("Input")
        local Element = Wrap(Container:AddInput(Flag, {
            Title = tostring(Config.Text or Flag),
            Description = Config.Tooltip,
            Default = Config.Default,
            Placeholder = Config.Placeholder,
            Numeric = Config.Numeric,
            Finished = Config.Finished,
            Callback = Config.Callback,
        }), self, Flag)
        Library.Options[Flag] = Element
        return Element
    end

    function Self:AddColorPicker(Flag, Config)
        Config = Config or {}
        Flag = Flag or NextId("Color")
        local Element = Wrap(Container:AddColorpicker(Flag, {
            Title = tostring(Config.Text or Flag),
            Default = Config.Default or Color3.new(1, 1, 1),
            Transparency = Config.Transparency,
            Callback = Config.Callback,
        }), self, Flag)
        Library.Options[Flag] = Element
        return Element
    end

    function Self:AddKeyPicker(Flag, Config, Sync)
        Config = Config or {}
        Flag = Flag or NextId("Key")
        local Default = Config.Default
        if Default == nil then
            Default = "None"
        elseif typeof(Default) == "EnumItem" then
            Default = Default.Name
        end
        local Element = Wrap(Container:AddKeybind(Flag, {
            Title = tostring(Config.Text or Flag),
            Mode = Config.Mode or "Toggle",
            Default = tostring(Default),
            Callback = function(State)
                if Sync and Config.SyncToggleState ~= false then
                    pcall(Sync.SetValue, Sync, State)
                end
                if Config.Callback then
                    Config.Callback(State)
                end
            end,
            ChangedCallback = Config.ChangedCallback,
        }), self, Flag)
        Library.Options[Flag] = Element
        return Element
    end

    -- Fluent has no standalone divider; kept so feature code reads the same either way
    function Self:AddDivider()
        return self
    end

    function Self:Section(Config)
        local Title = type(Config) == "table" and (Config.Title or Config.Text) or Config
        return self:AddLabel(Title)
    end

    return Self
end

local function Tab(Node)
    local Self = { Node = Node, Fluent = Node }
    local Direct = Host(Node)

    function Self:AddLeftGroupbox(Name)
        return Host(Node:AddSection(tostring(Name or "")))
    end
    function Self:AddRightGroupbox(Name)
        return Host(Node:AddSection(tostring(Name or "")))
    end
    function Self:Section(Config)
        local Title = type(Config) == "table" and Config.Title or Config
        return Host(Node:AddSection(tostring(Title or "")))
    end

    for _, Name in { "AddToggle", "AddButton", "AddLabel", "AddDropdown", "AddSlider", "AddInput",
        "AddColorPicker", "AddKeyPicker", "AddDivider" } do
        Self[Name] = function(_, ...)
            return Direct[Name](Direct, ...)
        end
    end

    return Self
end

-- util -----------------------------------------------------------------------------------------
local Util = {}

Util.Identity = LPH_NO_VIRTUALIZE(function(Level)
    if type(Compat.Identity) == "function" then
        pcall(Compat.Identity, Level)
    end
end)

Util.Number = LPH_NO_VIRTUALIZE(function(Value, Fallback)
    local Result = tonumber(Value)
    if Result ~= Result then
        return Fallback or 0
    end
    return Result or Fallback or 0
end)

Util.Commas = LPH_NO_VIRTUALIZE(function(Value)
    local Text = tostring(math.floor(Util.Number(Value)))
    local Done
    repeat
        Text, Done = Text:gsub("^(-?%d+)(%d%d%d)", "%1,%2")
    until Done == 0
    return Text
end)

Util.Clock = LPH_NO_VIRTUALIZE(function(Seconds)
    Seconds = math.max(0, math.floor(Util.Number(Seconds)))
    local Minutes = Seconds // 60
    if Minutes > 0 then
        return ("%dm %ds"):format(Minutes, Seconds % 60)
    end
    return ("%ds"):format(Seconds)
end)

Util.Cap = LPH_NO_VIRTUALIZE(function(Text, Length)
    Text = tostring(Text or "")
    if #Text <= Length then
        return Text
    end
    return Text:sub(1, Length - 1) .. "\u{2026}"
end)

-- a multi dropdown's live value is an array, but a loaded config can be a map: normalise both
Util.Set = LPH_NO_VIRTUALIZE(function(Value)
    local Out = {}
    if type(Value) == "table" then
        for Key, Entry in pairs(Value) do
            if type(Entry) == "string" then
                Out[Entry] = true
            elseif Entry == true then
                Out[tostring(Key)] = true
            end
        end
    elseif type(Value) == "string" then
        Out[Value] = true
    end
    return Out
end)

Util.List = LPH_NO_VIRTUALIZE(function(Value)
    local Out = {}
    for Name in pairs(Util.Set(Value)) do
        table.insert(Out, Name)
    end
    table.sort(Out)
    return Out
end)

Util.Count = LPH_NO_VIRTUALIZE(function(Value)
    local Total = 0
    if type(Value) == "table" then
        for _ in pairs(Value) do
            Total += 1
        end
    end
    return Total
end)

-- what a multi dropdown currently holds, as a set
Util.Picked = LPH_NO_VIRTUALIZE(function(Flag)
    return Util.Set(Flags[Flag])
end)

Util.Chose = LPH_NO_VIRTUALIZE(function(Flag, Name)
    return Util.Picked(Flag)[tostring(Name)] == true
end)

Util.WaitUntil = LPH_NO_VIRTUALIZE(function(Check, Timeout, Step)
    local Deadline = os.clock() + (Timeout or 10)
    while os.clock() < Deadline and Alive() do
        local Ok, Result = pcall(Check)
        if Ok and Result then
            return Result
        end
        task.wait(Step or 0.1)
    end
    return nil
end)

Util.Path = LPH_NO_VIRTUALIZE(function(Root, ...)
    local Node = Root
    for _, Name in { ... } do
        if not Node then
            return nil
        end
        Node = Node:FindFirstChild(Name)
    end
    return Node
end)

-- HasTag is cheap but throws on a destroyed instance, and some executors hand back an
-- Instance proxy without it at all
Util.Tagged = LPH_NO_VIRTUALIZE(function(Node, Tag)
    if typeof(Node) ~= "Instance" then
        return false
    end
    local Ok, Has = pcall(Node.HasTag, Node, Tag)
    return Ok and Has == true
end)

-- require with a deadline: a yielding module must not hang the boot thread
Util.Require = LPH_NO_VIRTUALIZE(function(Node, Timeout)
    if typeof(Node) ~= "Instance" or not Node:IsA("ModuleScript") then
        return nil
    end
    local Done, Result = false, nil
    local Thread = task.spawn(function()
        Util.Identity(2)
        local Ok, Value = pcall(require, Node)
        if Ok then
            Result = Value
        end
        Done = true
    end)
    local Deadline = os.clock() + (Timeout or 5)
    while not Done and os.clock() < Deadline do
        task.wait(0.05)
    end
    if not Done then
        pcall(task.cancel, Thread)
        return nil
    end
    return Result
end)

Util.Center = LPH_NO_VIRTUALIZE(function(Model)
    if typeof(Model) ~= "Instance" then
        return nil
    end
    local Sum, Count = Vector3.zero, 0
    for _, Part in Model:GetDescendants() do
        if Part:IsA("BasePart") then
            Sum += Part.Position
            Count += 1
        end
    end
    if Count == 0 then
        return nil
    end
    return Sum / Count
end)

-- a model's stand point, whichever of the two APIs this model supports
Util.Pivot = LPH_NO_VIRTUALIZE(function(Model)
    if typeof(Model) ~= "Instance" then
        return nil
    end
    if Model:IsA("BasePart") then
        return Model.CFrame
    end
    if not Model:IsA("Model") then
        return nil
    end
    if Model.PrimaryPart then
        local Ok, Frame = pcall(Model.GetPrimaryPartCFrame, Model)
        if Ok then
            return Frame
        end
        return Model.PrimaryPart.CFrame
    end
    local Ok, Frame = pcall(Model.GetPivot, Model)
    return Ok and Frame or nil
end)

Util.Batches = LPH_NO_VIRTUALIZE(function(List, Size)
    local Out, Batch = {}, {}
    for _, Entry in List do
        table.insert(Batch, Entry)
        if #Batch >= Size then
            table.insert(Out, Batch)
            Batch = {}
        end
    end
    if #Batch > 0 then
        table.insert(Out, Batch)
    end
    return Out
end)

Util.Guard = LPH_NO_VIRTUALIZE(function(Name, Callback, ...)
    local Ok, Error = pcall(Callback, ...)
    if not Ok then
        Library.Fault(Name, Error)
    end
    return Ok
end)

-- mobile toggle --------------------------------------------------------------------------------
-- Fluent only offers a keyboard shortcut to put its window away, which a phone has no way to
-- press. This is a draggable floating button that calls the same Minimize, so the hub can be
-- tucked away and brought back with a thumb. Its gui carries the hub's tag, so a re-execute
-- sweeps it up with everything else.
local Mobile = { Gui = nil, Button = nil, Connections = {}, Shown = nil }

local function MobileRoot()
    if type(gethui) == "function" then
        local Ok, Hidden = pcall(gethui)
        if Ok and Hidden then
            return Hidden
        end
    end
    local Ok, Core = pcall(game.GetService, game, "CoreGui")
    if Ok and Core then
        return Core
    end
    return LocalPlayer:FindFirstChild("PlayerGui")
end

Mobile.Away = LPH_NO_VIRTUALIZE(function()
    return Window ~= nil and Window.Minimized == true
end)

-- the window can also be hidden by the keybind, so the face is repainted from the window's own
-- state rather than from whatever the button last did
Mobile.Paint = LPH_NO_VIRTUALIZE(function()
    local Button = Mobile.Button
    if not Button then
        return
    end
    local Away = Mobile.Away()
    if Mobile.Shown == Away then
        return
    end
    Mobile.Shown = Away
    -- open and closed read off the spine and the glyph, not off the label: the accent spine runs
    -- the full height while the window is hidden, and the bars dim while it is on screen
    local Spine = Button:FindFirstChild("Spine")
    if Spine then
        Skin.Tween(Spine, 0.18, {
            Size = Away and UDim2.fromOffset(3, 34) or UDim2.fromOffset(3, 22),
            BackgroundColor3 = Away and Skin.Accent or Skin.Dim,
        })
    end
    local Glyph = Button:FindFirstChild("Glyph")
    for _, Bar in (Glyph and Glyph:GetChildren() or {}) do
        if Bar:IsA("Frame") then
            Skin.Tween(Bar, 0.18, { BackgroundTransparency = Away and 0.1 or 0.45 })
        end
    end
    Skin.Tween(Button, 0.18, {
        BackgroundTransparency = Away and 0.08 or 0.3,
    })
end)

Mobile.Toggle = LPH_NO_VIRTUALIZE(function()
    RaiseIdentity()
    if not Window then
        return
    end
    local Ok = pcall(function()
        Window:Minimize()
    end)
    if not Ok then
        -- a build without Minimize still has the root frame to hide
        pcall(function()
            local Root = Window.Root
            if Root then
                Root.Visible = not Root.Visible
            end
        end)
    end
    Mobile.Paint()
end)

Mobile.Destroy = LPH_NO_VIRTUALIZE(function()
    for _, Connection in Mobile.Connections do
        pcall(function()
            Connection:Disconnect()
        end)
    end
    table.clear(Mobile.Connections)
    if Mobile.Gui then
        pcall(function()
            Mobile.Gui:Destroy()
        end)
    end
    Mobile.Gui, Mobile.Button, Mobile.Shown = nil, nil, nil
end)

Mobile.Build = LPH_NO_VIRTUALIZE(function()
    if Mobile.Gui then
        return
    end
    RaiseIdentity()
    local Root = MobileRoot()
    if not Root then
        return
    end

    local Gui = Skin.Screen("AxonMobile", Root, 2147483647)

    -- a squared-off handle rather than a round "AXON" bubble: a dark tile, an accent spine down
    -- the left edge and three drawn bars for the glyph, so there is no text to clip on a phone
    local Button = Skin.New("TextButton", {
        Name = "Toggle",
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.fromScale(0.07, 0.5),
        Size = UDim2.fromOffset(52, 52),
        AutoButtonColor = false,
        BorderSizePixel = 0,
        Text = "",
        BackgroundColor3 = Skin.Panel,
        BackgroundTransparency = 0.08,
        ZIndex = 50,
    }, Gui)
    Skin.Round(Button, UDim.new(0, 15))
    Skin.Stroke(Button, Skin.Edge, 1, 0.3)

    local Spine = Skin.New("Frame", {
        Name = "Spine",
        AnchorPoint = Vector2.new(0, 0.5),
        Position = UDim2.new(0, 5, 0.5, 0),
        Size = UDim2.fromOffset(3, 22),
        BackgroundColor3 = Skin.Accent,
        BorderSizePixel = 0,
        ZIndex = 51,
    }, Button)
    Skin.Round(Spine, UDim.new(1, 0))

    local Glyph = Skin.New("Frame", {
        Name = "Glyph",
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 4, 0.5, 0),
        Size = UDim2.fromOffset(20, 14),
        BackgroundTransparency = 1,
        ZIndex = 51,
    }, Button)
    Skin.New("UIListLayout", {
        FillDirection = Enum.FillDirection.Vertical,
        HorizontalAlignment = Enum.HorizontalAlignment.Left,
        VerticalAlignment = Enum.VerticalAlignment.Center,
        Padding = UDim.new(0, 4),
    }, Glyph)
    for Index, Width in { 20, 14, 17 } do
        local Bar = Skin.New("Frame", {
            Size = UDim2.fromOffset(Width, 2),
            BackgroundColor3 = Skin.Text,
            BackgroundTransparency = 0.1,
            BorderSizePixel = 0,
            LayoutOrder = Index,
        }, Glyph)
        Skin.Round(Bar, UDim.new(1, 0))
    end

    Mobile.Gui, Mobile.Button = Gui, Button

    -- drag and tap share the one press: it only counts as a tap if the thumb barely moved, so
    -- parking the button somewhere never opens or closes the window by accident
    local Dragging, Moved, Start, Origin = false, false, nil, nil

    local function Began(Input)
        if Input.UserInputType ~= Enum.UserInputType.Touch
            and Input.UserInputType ~= Enum.UserInputType.MouseButton1 then
            return
        end
        Dragging, Moved = true, false
        Start, Origin = Input.Position, Button.Position
    end

    local function Changed(Input)
        if not Dragging or not Mobile.Button then
            return
        end
        if Input.UserInputType ~= Enum.UserInputType.Touch
            and Input.UserInputType ~= Enum.UserInputType.MouseMovement then
            return
        end
        local Delta = Input.Position - Start
        if math.abs(Delta.X) > 6 or math.abs(Delta.Y) > 6 then
            Moved = true
        end
        local Screen = Gui.AbsoluteSize
        local Half = Button.AbsoluteSize / 2
        local X = Origin.X.Scale * Screen.X + Origin.X.Offset + Delta.X
        local Y = Origin.Y.Scale * Screen.Y + Origin.Y.Offset + Delta.Y
        -- kept fully on screen, or a careless drag loses the only way back in
        Button.Position = UDim2.fromOffset(
            math.clamp(X, Half.X, math.max(Half.X, Screen.X - Half.X)),
            math.clamp(Y, Half.Y, math.max(Half.Y, Screen.Y - Half.Y)))
    end

    local function Ended(Input)
        if not Dragging then
            return
        end
        if Input.UserInputType ~= Enum.UserInputType.Touch
            and Input.UserInputType ~= Enum.UserInputType.MouseButton1 then
            return
        end
        Dragging = false
        if not Moved then
            Mobile.Toggle()
        end
    end

    table.insert(Mobile.Connections, Button.InputBegan:Connect(Began))
    table.insert(Mobile.Connections, UserInputService.InputChanged:Connect(Changed))
    table.insert(Mobile.Connections, UserInputService.InputEnded:Connect(Ended))

    Mobile.Shown = nil
    Mobile.Paint()
end)

Mobile.Apply = LPH_NO_VIRTUALIZE(function(On)
    if On == nil then
        On = Flags.MobileButton ~= false
    end
    if On then
        Mobile.Build() else Mobile.Destroy()
    end
end)

end

local function Track(Signal, Callback)
    local Connection = Signal:Connect(Callback)
    table.insert(Reg.Connections, Connection)
    return Connection
end

-- a tag feed: keeps `List` in step with CollectionService, with an optional filter
local function Tagged(Tag, List, Filter)
    local function Add(Node)
        if Filter and not Filter(Node) then
            return
        end
        if not table.find(List, Node) then
            table.insert(List, Node)
        end
    end
    local function Remove(Node)
        local Index = table.find(List, Node)
        if Index then
            table.remove(List, Index)
        end
    end
    for _, Node in CollectionService:GetTagged(Tag) do
        Add(Node)
    end
    Track(CollectionService:GetInstanceAddedSignal(Tag), Add)
    Track(CollectionService:GetInstanceRemovedSignal(Tag), Remove)
    return List
end

-- a folder feed: `List` holds the children, and each model also lands in its name bucket
local function Watch(Folder, List, OnAdd, OnRemove)
    local function Add(Node)
        if not table.find(List, Node) then
            table.insert(List, Node)
        end
        if Node:IsA("Model") then
            local Named = Reg.Named[Node.Name]
            if Named and not table.find(Named, Node) then
                table.insert(Named, Node)
            end
        end
        if OnAdd then
            Util.Guard("registry", OnAdd, Node)
        end
    end
    local function Remove(Node)
        local Index = table.find(List, Node)
        if Index then
            table.remove(List, Index)
        end
        local Named = Reg.Named[Node.Name]
        if Named then
            local At = table.find(Named, Node)
            if At then
                table.remove(Named, At)
            end
        end
        if OnRemove then
            Util.Guard("registry", OnRemove, Node)
        end
    end
    task.spawn(function()
        for _, Node in Folder:GetChildren() do
            Add(Node)
            RunService.Heartbeat:Wait()
        end
    end)
    Track(Folder.ChildAdded, Add)
    Track(Folder.ChildRemoved, Remove)
end

local function Prune(List, Node)
    local Index = table.find(List, Node)
    if Index then
        table.remove(List, Index)
    end
end

local function SortTree(Tree)
    if not (Tree:IsA("Model") and Tree.PrimaryPart) then
        return
    end
    local Lower = Tree.Name:lower()
    if Lower:find("treebig") or Lower:find("bigtree") then
        if not table.find(Reg.BigTrees, Tree) then
            table.insert(Reg.BigTrees, Tree)
        end
    elseif Tree.Name == "Snowy Small Tree" or Tree.Name == "Northern Pine" then
        if not table.find(Reg.SnowyTrees, Tree) then
            table.insert(Reg.SnowyTrees, Tree)
        end
        if Tree:GetAttribute("HasLights") and not table.find(Reg.LightTrees, Tree) then
            table.insert(Reg.LightTrees, Tree)
        end
    elseif not table.find(Reg.SmallTrees, Tree) then
        table.insert(Reg.SmallTrees, Tree)
    end
end

-- One sliding window for every request the hub makes. Drag spam and the bring engine push the
-- most traffic, so the window is sized to the client's own burst rather than the hub's appetite.
local Bucket = { Rate = 25, Burst = 12, Tokens = 12, Stamp = os.clock() }

Util.Take = LPH_NO_VIRTUALIZE(function()
    while Alive() do
        local Now = os.clock()
        Bucket.Tokens = math.min(Bucket.Burst, Bucket.Tokens + (Now - Bucket.Stamp) * Bucket.Rate)
        Bucket.Stamp = Now
        if Bucket.Tokens >= 1 then
            Bucket.Tokens -= 1
            return true
        end
        task.wait((1 - Bucket.Tokens) / Bucket.Rate)
    end
    return false
end)

local Net = { Folder = nil, Cache = {}, Missing = {} }

Net.Load = LPH_NO_VIRTUALIZE(function()
    Net.Folder = ReplicatedStorage:WaitForChild("RemoteEvents", 25)
    if not Net.Folder then
        Note("RemoteEvents never appeared")
    end
    return Net.Folder ~= nil
end)

Net.Get = LPH_NO_VIRTUALIZE(function(Name)
    if REMOTES[Name] == nil then
        State.Refused += 1
        Note("refused to touch " .. tostring(Name))
        return nil
    end
    local Cached = Net.Cache[Name]
    if Cached and Cached.Parent then
        return Cached
    end
    local Folder = Net.Folder or ReplicatedStorage:FindFirstChild("RemoteEvents")
    if not Folder then
        return nil
    end
    local Remote = Folder:FindFirstChild(Name)
    if not Remote then
        if not Net.Missing[Name] then
            Net.Missing[Name] = true
            Note(Name .. " is not in this build")
        end
        return nil
    end
    Net.Cache[Name] = Remote
    return Remote
end)

Net.Fire = LPH_NO_VIRTUALIZE(function(Name, ...)
    local Remote = Net.Get(Name)
    if not Remote then
        return false
    end
    if not Util.Take() then
        return false
    end
    local Ok, Error = pcall(Remote.FireServer, Remote, ...)
    if Ok then
        State.Sent += 1
    else
        Note(Name .. " failed: " .. Util.Cap(tostring(Error), 60))
    end
    return Ok
end)

-- a RemoteFunction with a deadline, so a hung server answer cannot park a job forever
Net.Invoke = LPH_NO_VIRTUALIZE(function(Name, ...)
    local Remote = Net.Get(Name)
    if not Remote then
        return nil, "missing"
    end
    if not Util.Take() then
        return nil, "unloaded"
    end

    local Arguments = table.pack(...)
    local Done, Value, Failure = false, nil, nil
    local Thread = task.spawn(function()
        Util.Identity(2)
        local Ok, Result = pcall(function()
            return table.pack(Remote:InvokeServer(table.unpack(Arguments, 1, Arguments.n)))
        end)
        if Ok then
            Value = Result else Failure = Result
        end
        Done = true
    end)

    local Deadline = os.clock() + 12
    while not Done and os.clock() < Deadline and Alive() do
        task.wait(0.05)
    end
    if not Done then
        pcall(task.cancel, Thread)
        return nil, "timeout"
    end
    if Failure then
        Note(Name .. " failed: " .. Util.Cap(tostring(Failure), 60))
        return nil, tostring(Failure)
    end
    State.Sent += 1
    if type(Value) == "table" then
        return table.unpack(Value, 1, Value.n)
    end
    return nil
end)

-- did the server answer with a refusal, and why
Net.Why = LPH_NO_VIRTUALIZE(function(Result)
    if type(Result) ~= "table" then
        return false, "no answer"
    end
    if Result.Success == true then
        return true, nil
    end
    return false, tostring(Result.Error or Result.Reason or "refused")
end)

-- movement -------------------------------------------------------------------------------------
-- Movement Mode decides how the hub travels. This game teleports the client itself all over the
-- place (chest looting, kid rescue, the map reveal spiral all PivotTo), so Teleport is the mode
-- every feature was built around and stays the default; Glide and Walk are there for anyone who
-- would rather not be seen blinking.
local Move = { Owner = nil, Priority = 0, Stamp = 0, Anchor = nil }

local MOVE_PRIORITY = { travel = 60, chest = 40, kids = 40, bring = 30, collect = 20, plant = 20 }
local MOVE_STALE = 20

-- The lock only arbitrates between jobs inside one cycle. A job abandoned part way through a
-- walk must not hold the body for the rest of the session, so a stale lock is taken off it.
Move.Take = LPH_NO_VIRTUALIZE(function(Name)
    local Held = Move.Owner
    if Held and Held ~= Name and os.clock() - Move.Stamp > MOVE_STALE then
        Held, Move.Owner, Move.Priority = nil, nil, 0
    end
    local Want = MOVE_PRIORITY[Name] or 10
    if Held and Held ~= Name and Move.Priority > Want then
        return false
    end
    Move.Owner, Move.Priority, Move.Stamp = Name, Want, os.clock()
    return true
end)

Move.Drop = LPH_NO_VIRTUALIZE(function(Name)
    if Name == nil or Move.Owner == Name then
        Move.Owner, Move.Priority, Move.Stamp = nil, 0, 0
    end
end)

Move.Stop = LPH_NO_VIRTUALIZE(function()
    local _, Humanoid, Root = Char.Parts()
    if Humanoid then
        pcall(function()
            Humanoid:Move(Vector3.zero, false)
        end)
    end
    if Root then
        pcall(function()
            Root.AssemblyLinearVelocity = Vector3.zero
        end)
    end
end)

Move.Teleport = LPH_NO_VIRTUALIZE(function(Target)
    local Model, _, Root = Char.Parts()
    if not (Model and Root) then
        return false
    end
    local Exact = typeof(Target) == "CFrame"
    local Ok = pcall(function()
        Model:PivotTo(Exact and Target or CFrame.new(Target))
        Root.AssemblyLinearVelocity = Vector3.zero
    end)
    return Ok
end)

Move.Glide = LPH_NO_VIRTUALIZE(function(Target, Timeout)
    local _, Humanoid, Root = Char.Parts()
    if not Root then
        return false
    end
    if typeof(Target) == "CFrame" then
        Target = Target.Position
    end
    local Gap = (Root.Position - Target).Magnitude
    local Seconds = math.clamp(Gap / 120, 0.2, Timeout or 8)
    if Humanoid then
        pcall(function()
            Humanoid.PlatformStand = true
        end)
    end
    local Tween = TweenService:Create(Root, TweenInfo.new(Seconds, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        CFrame = CFrame.new(Target),
    })
    Tween:Play()
    local Deadline = os.clock() + Seconds + 1
    while Tween.PlaybackState == Enum.PlaybackState.Playing and os.clock() < Deadline and Alive() do
        pcall(function()
            Root.AssemblyLinearVelocity = Vector3.zero
        end)
        task.wait(0.05)
    end
    pcall(function()
        Tween:Cancel()
    end)
    if Humanoid then
        pcall(function()
            Humanoid.PlatformStand = false
        end)
    end
    Move.Stop()
    return Char.Distance(Target) <= 10
end)

Move.Walk = LPH_NO_VIRTUALIZE(function(Target, Timeout)
    if typeof(Target) == "CFrame" then
        Target = Target.Position
    end
    local Deadline = os.clock() + (Timeout or 15)
    while os.clock() < Deadline and Alive() do
        local _, Humanoid, Root = Char.Parts()
        if not (Humanoid and Root) then
            return false
        end
        if (Root.Position - Target).Magnitude <= 8 then
            return true
        end
        pcall(function()
            Humanoid:MoveTo(Target)
        end)
        task.wait(0.25)
    end
    return Char.Distance(Target) <= 10
end)

Move.Mode = LPH_NO_VIRTUALIZE(function()
    return tostring(Flags.MoveMode or "Teleport")
end)

-- one travel call for every job: a hop already inside the arrive radius is done
Move.GoTo = LPH_NO_VIRTUALIZE(function(Target, Radius, Name)
    if typeof(Target) == "Instance" then
        Target = Util.Pivot(Target)
    end
    if typeof(Target) ~= "CFrame" and typeof(Target) ~= "Vector3" then
        return false
    end
    local Point = typeof(Target) == "CFrame" and Target.Position or Target
    Radius = Radius or 8
    if Char.Distance(Point) <= Radius then
        return true
    end
    if Name and not Move.Take(Name) then
        return false
    end

    local Mode, Ok = Move.Mode(), false
    if Mode == "Glide" then
        Ok = Move.Glide(Point)
    elseif Mode == "Walk" then
        Ok = Move.Walk(Point)
        -- a long walk on this map stalls on geometry; close the last stretch with a glide
        if not Ok and Char.Distance(Point) <= 160 then
            Ok = Move.Glide(Point)
        end
    else
        Ok = Move.Teleport(Target)
        if not Ok then
            Ok = Move.Glide(Point)
        end
    end

    if Name then
        Move.Drop(Name)
    end
    return Ok or Char.Distance(Point) <= Radius
end)

-- Remember where the player was standing before a job dragged them across the map, and put them
-- back when it is done. Every multi-stop routine brackets itself with these two.
Move.Save = LPH_NO_VIRTUALIZE(function()
    Move.Anchor = Char.Frame()
    return Move.Anchor
end)

Move.Return = LPH_NO_VIRTUALIZE(function(Frame)
    local Spot = Frame or Move.Anchor
    if not Spot then
        return false
    end
    Move.Anchor = nil
    return Move.Teleport(Spot + Vector3.new(0, 5, 0))
end)

-- tools ----------------------------------------------------------------------------------------
local Tool = { Client = nil, ClientTried = false }

-- PlayerScripts.Client is the game's own module registry: it auto-registers every ModuleScript
-- under it by name, so Client.InventoryHandler is the real inventory code. Equipping is entirely
-- client side in this game - there is no equip remote - so the only honest way to put a tool in
-- hand is to call the game's own RequestEquipItem.
Tool.Game = LPH_NO_VIRTUALIZE(function()
    if Tool.Client or Tool.ClientTried then
        return Tool.Client
    end
    Tool.ClientTried = true
    local Scripts = LocalPlayer:FindFirstChild("PlayerScripts")
    local Node = Scripts and Scripts:FindFirstChild("Client")
    Tool.Client = Node and Util.Require(Node, 8) or nil
    return Tool.Client
end)

-- The Client table is not a plain table: ConfigureLoading gives it an __index that requires a
-- module on first touch and caches it. A rawget only sees what the game happened to import
-- already, so every lookup goes through a guarded normal index.
Tool.Module = LPH_NO_VIRTUALIZE(function(Name)
    local Client = Tool.Game()
    if not Client then
        return nil
    end
    local Ok, Found = pcall(function()
        return Client[Name]
    end)
    if Ok and type(Found) == "table" then
        return Found
    end
    return nil
end)

Tool.Inventory = LPH_NO_VIRTUALIZE(function()
    return Tool.Module("InventoryHandler")
end)

Tool.Kind = LPH_NO_VIRTUALIZE(function(Item)
    if typeof(Item) ~= "Instance" then
        return "Unknown"
    end
    if Item:GetAttribute("WeaponDamage") ~= nil or Item:GetAttribute("WeaponResourceDamage") ~= nil then
        return "MeleeWeapon"
    end
    if Item:GetAttribute("Capacity") ~= nil then
        return "Sack"
    end
    return "Unknown"
end)

Tool.Equip = LPH_NO_VIRTUALIZE(function(Item)
    if typeof(Item) ~= "Instance" then
        return nil
    end
    -- the real equip, through the game's own handler
    local Handler = Tool.Inventory()
    if Handler and type(Handler.RequestEquipItem) == "function" then
        local Ok = pcall(Handler.RequestEquipItem, Item)
        if Ok then
            -- EquipItemHandle only tells the OTHER clients to draw the tool; it takes the item
            -- alone, never a "FireAllClients" string, which is what the hub used to send.
            Net.Fire("EquipItemHandle", Item)
            return Item
        end
    end
    Net.Fire("EquipItemHandle", Item)
    return Item
end)

Tool.Find = LPH_NO_VIRTUALIZE(function(Query, Exact)
    local Inventory = LocalPlayer:FindFirstChild("Inventory")
    if not Inventory then
        return nil
    end
    local Text = tostring(Query)
    for _, Item in Inventory:GetChildren() do
        if Exact then
            if Item.Name == Text then
                return Item
            end
        elseif Item.Name:find(Text, 1, true) then
            return Item
        end
    end
    return nil
end)

Tool.Sack = LPH_NO_VIRTUALIZE(function()
    return Tool.Find("Sack")
end)

-- the best item of a kind in the inventory: for a weapon that is the hardest hitting one
Tool.Best = LPH_NO_VIRTUALIZE(function(Kind)
    local Inventory = LocalPlayer:FindFirstChild("Inventory")
    if not Inventory then
        return nil
    end
    local Best, Damage = nil, -1
    for _, Item in Inventory:GetChildren() do
        if Tool.Kind(Item) == Kind then
            if Kind ~= "MeleeWeapon" then
                return Item
            end
            local Hit = Util.Number(Item:GetAttribute("WeaponDamage"))
            if Hit > Damage then
                Best, Damage = Item, Hit
            end
        end
    end
    return Best
end)

-- what is in hand, falling back to equipping the best of the wanted kind
Tool.Held = LPH_NO_VIRTUALIZE(function(Want)
    if not LocalPlayer.Character then
        return nil
    end
    local Handle = LocalPlayer.Character:FindFirstChild("ToolHandle")
    local Original = Handle and Handle:FindFirstChild("OriginalItem")
    local Item = Original and Original.Value or nil

    if Item then
        -- the kind attributes (WeaponDamage / Capacity) live on the item, never on the ToolHandle
        -- part, which is what Tool.Best reads them off too. Asking the handle always answered
        -- "Unknown", so a weapon already in hand was never recognised and the hub re-equipped it
        -- on every pass - a FireServer every second that cancelled the swing it had just started.
        local Kind = Tool.Kind(Item)
        if Want == nil or Kind == Want then
            return Item, Kind
        end
    end
    if not Want then
        return nil
    end

    local Pick = Tool.Best(Want)
    if not Pick then
        return nil
    end
    Tool.Equip(Pick)
    return Pick, Want
end)

-- the weapon loops ask once a second at most; equipping on every swing floods the remote
local Weapon = { Item = nil, Kind = nil, Stamp = 0 }

Tool.Weapon = LPH_NO_VIRTUALIZE(function()
    if os.clock() - Weapon.Stamp < 1 and Weapon.Item and Weapon.Item.Parent then
        return Weapon.Item, Weapon.Kind
    end
    Weapon.Stamp = os.clock()
    Weapon.Item, Weapon.Kind = Tool.Held("MeleeWeapon")
    return Weapon.Item, Weapon.Kind
end)

-- is a weapon genuinely in the character's hands right now? Require Hand Check keys off this;
-- with it off the hub equips the best one itself instead of waiting for the player to.

-- freecam --------------------------------------------------------------------------------------
-- The slower bring methods fly the character to each item, which drags the camera with it. Taking
-- the camera off the character keeps the screen usable while that happens.
local Freecam = { On = false, Connection = nil, Look = Vector2.zero }

Freecam.Enable = LPH_NO_VIRTUALIZE(function()
    if Freecam.On or Flags.UseFreecamForBring ~= true then
        return
    end
    local View = Workspace.CurrentCamera
    if not View then
        return
    end

    Freecam.On = true
    Freecam.Look = Vector2.zero
    View.CameraType = Enum.CameraType.Scriptable

    Freecam.Connection = RunService.RenderStepped:Connect(function(Delta)
        local Current = Workspace.CurrentCamera
        if not Current then
            return
        end
        local Wanted = Vector3.new(
            (UserInputService:IsKeyDown(Enum.KeyCode.D) and 1 or 0) - (UserInputService:IsKeyDown(Enum.KeyCode.A) and 1 or 0),
            (UserInputService:IsKeyDown(Enum.KeyCode.E) and 1 or 0) - (UserInputService:IsKeyDown(Enum.KeyCode.Q) and 1 or 0),
            (UserInputService:IsKeyDown(Enum.KeyCode.S) and 1 or 0) - (UserInputService:IsKeyDown(Enum.KeyCode.W) and 1 or 0)
        ) * 50

        local Delta2 = UserInputService:GetMouseDelta() * 0.1
        Freecam.Look -= Vector2.new(Delta2.X, Delta2.Y)
        local Yaw = math.rad(Freecam.Look.X)
        local Pitch = math.clamp(math.rad(Freecam.Look.Y), -math.pi / 2, math.pi / 2)
        local Rotation = CFrame.Angles(0, Yaw, 0) * CFrame.Angles(Pitch, 0, 0)

        Current.CFrame = CFrame.new(Current.CFrame.Position) * Rotation
        Current.CFrame += Rotation * Wanted * Delta
    end)

    UserInputService.MouseBehavior = Enum.MouseBehavior.LockCurrentPosition
end)

Freecam.Disable = LPH_NO_VIRTUALIZE(function()
    if not Freecam.On then
        return
    end
    Freecam.On = false
    if Freecam.Connection then
        pcall(function()
            Freecam.Connection:Disconnect()
        end)
        Freecam.Connection = nil
    end
    local View = Workspace.CurrentCamera
    if View then
        View.CameraType = Enum.CameraType.Custom
        local Model = LocalPlayer.Character
        local Humanoid = Model and Model:FindFirstChildOfClass("Humanoid")
        if Humanoid then
            View.CameraSubject = Humanoid
        end
    end
    UserInputService.MouseBehavior = Enum.MouseBehavior.Default
end)

-- bring ----------------------------------------------------------------------------------------
-- The one item mover. Everything that says "Bring ..." anywhere in the UI comes through here:
-- it collects candidates out of the name registry, asks the server for drag ownership, pivots
-- each item to the drop point and lets go.
local Bring = { Cancel = nil, Running = false, Done = 0, Total = 0, What = "" }

Bring.Method = LPH_NO_VIRTUALIZE(function()
    local Picked = tostring(Flags.BringMethodValue or "Ultra Fast")
    if Picked == "Ultra Fast" then
        return "ULTRA"
    end
    if Picked == "Fast" then
        return "FAST"
    end
    return "SLOW"
end)

Bring.Where = LPH_NO_VIRTUALIZE(function()
    local Choice = tostring(Flags.BringSettingsLocation or "Player")
    if Choice == "Workbench" then
        local Bench = Place.Bench()
        if Bench then
            return Bench + Vector3.new(0, Util.Number(Flags.BringHeight, 20), 0)
        end
    elseif Choice == "Fire" then
        local Fire = Place.Camp()
        if Fire then
            return Fire + Vector3.new(0, Util.Number(Flags.BringHeight, 20), 0)
        end
    end
    local Frame = Char.Frame()
    if Frame then
        return Frame + Vector3.new(0, 10, 0)
    end
    local View = Workspace.CurrentCamera
    return View and View.CFrame + Vector3.new(0, 10, 0) or nil
end)

Bring.UsePlayer = LPH_NO_VIRTUALIZE(function()
    local Element = Options.BringSettingsLocation
    if Element and Flags.BringSettingsLocation ~= "Player" then
        pcall(Element.SetValue, Element, "Player")
    end
end)

Bring.UseFire = LPH_NO_VIRTUALIZE(function()
    local Element = Options.BringSettingsLocation
    if Element and Flags.BringSettingsLocation ~= "Fire" then
        pcall(Element.SetValue, Element, "Fire")
    end
end)

-- a draggable candidate: in the world, with a part, and not locked by the server
local function Draggable(Item)
    return typeof(Item) == "Instance"
        and Item:IsA("Model")
        and Item.Parent ~= nil
        and Item.PrimaryPart ~= nil
        and Item:IsDescendantOf(Workspace)
        and Item:GetAttribute("LockedMessage") == nil
end

Bring.Gather = LPH_NO_VIRTUALIZE(function(Names, Fuel, NoLimit)
    local Cap = Util.Number(Flags.MaxPerItem, 100)
    if NoLimit or Flags.NoBringLimit == true then
        Cap = math.huge
    end

    local Found, Missing = {}, {}
    for _, Name in Names do
        local List = Reg.Named[tostring(Name)]
        if not List then
            table.insert(Missing, tostring(Name))
        else
            local Taken = 0
            for _, Item in List do
                if Taken >= Cap then
                    break
                end
                if Draggable(Item) then
                    Taken += 1
                    table.insert(Found, Item)
                end
            end
        end
    end

    if #Missing > 0 then
        Note("no registry bucket for " .. Util.Cap(table.concat(Missing, ", "), 60))
    end

    local Here = Char.Position()
    if Here and #Found < 200 then
        if Fuel then
            -- burn the best fuel first so the fire climbs fastest
            table.sort(Found, function(Left, Right)
                return Util.Number(Left:GetAttribute("BurnFuel")) > Util.Number(Right:GetAttribute("BurnFuel"))
            end)
        else
            table.sort(Found, function(Left, Right)
                return (Left.PrimaryPart.Position - Here).Magnitude < (Right.PrimaryPart.Position - Here).Magnitude
            end)
        end
    end
    if Cap < #Found then
        Found = table.move(Found, 1, Cap, 1, {})
    end
    return Found
end)

-- Ask for drag ownership and keep asking: the server hands network ownership over on its own
-- schedule, and one FireServer is often dropped while the model is still streaming in.
Bring.Claim = LPH_NO_VIRTUALIZE(function(Item, Repeats)
    Net.Fire("RequestStartDraggingItem", Item)
    -- already ours: the server handed network ownership straight over, so stop asking
    if Compat.NetworkOwner(Item.PrimaryPart) then
        return true
    end
    if not Repeats then
        return false
    end
    task.spawn(function()
        Util.Identity(2)
        for _ = 1, Repeats do
            task.wait(0.1)
            if not (Alive() and Draggable(Item)) then
                return
            end
            if Compat.NetworkOwner(Item.PrimaryPart) then
                return
            end
            Net.Fire("RequestStartDraggingItem", Item)
        end
    end)
    return false
end)

Bring.Release = LPH_NO_VIRTUALIZE(function(Item)
    if Flags.DontStopDrag == true then
        return
    end
    Net.Fire("StopDraggingItem", Item)
end)

Bring.Run = LPH_NO_VIRTUALIZE(function(What, Names, Options_)
    Options_ = Options_ or {}
    if Bring.Cancel then
        pcall(Bring.Cancel)
    end
    if type(Names) ~= "table" then
        Names = { Names }
    end
    if #Names == 0 then
        if not Options_.Quiet then
            Alert("nothing selected to bring", 2)
        end
        return function() end
    end

    local Target = Options_.Target or Bring.Where()
    if not Target then
        Alert("could not work out where to bring items", 4)
        return function() end
    end

    if Options_.Fuel and not Char.Alive() then
        Alert("cannot bring fuel while dead", 4)
        return function() end
    end

    local Items = Bring.Gather(Names, Options_.Fuel, Options_.NoLimit)
    if #Items == 0 then
        if not Options_.Quiet then
            Alert("no " .. tostring(What) .. " found - try exploring the map", 4)
        end
        return function() end
    end

    local Method = Bring.Method()
    local Spot = Char.Frame()
    local Running = true

    Bring.Running, Bring.Done, Bring.Total, Bring.What = true, 0, #Items, tostring(What)

    local function Finish(ByUser)
        if not Running then
            return
        end
        Running = false
        Bring.Running = false
        Freecam.Disable()
        local _, Humanoid = Char.Parts()
        if Humanoid then
            pcall(function()
                Humanoid.PlatformStand = false
            end)
        end
        if Method ~= "ULTRA" and Spot then
            Move.Teleport(Spot)
        end
        Move.Stop()
        if Bring.Cancel then
            Bring.Cancel = nil
        end
        Stop.Hide()
        if ByUser and Options_.OnCancel then
            Util.Guard("bring cancel", Options_.OnCancel)
        end
    end

    Bring.Cancel = function()
        Finish(false)
    end
    Stop.Show(function(ByUser)
        Finish(ByUser)
    end)

    if not Options_.Quiet then
        Alert(("bringing %d %s to %s"):format(#Items, tostring(What), tostring(Flags.BringSettingsLocation or "Player")), 4)
    end
    if Method ~= "ULTRA" then
        Freecam.Enable()
    end

    -- The slow methods orbit the item so the server sees the client standing next to it before
    -- the drag request; ULTRA skips the trip entirely and just asks.
    local ORBIT = {
        FAST = { 0, 90, 180, 270 },
        SLOW = { 0, 45, 90, 135, 180, 225, 270, 315 },
    }

    local function Carry(Item, Index, Count)
        if not (Running and Draggable(Item)) then
            return
        end

        local _, Humanoid, Root = Char.Parts()
        if Method ~= "ULTRA" then
            if not (Humanoid and Root) then
                return
            end
            pcall(function()
                Humanoid.PlatformStand = true
            end)
            local Frame = Util.Pivot(Item)
            if not Frame then
                return
            end
            pcall(function()
                Root.CFrame = CFrame.new(Frame.Position + Vector3.new(0, 10, 0))
            end)
            RunService.Heartbeat:Wait()
            for _, Angle in ORBIT[Method] or ORBIT.SLOW do
                if not Running then
                    break
                end
                local Radians = math.rad(Angle)
                pcall(function()
                    Root.CFrame = CFrame.new(Frame.Position + Vector3.new(math.cos(Radians) * 5, 10, math.sin(Radians) * 5))
                end)
                RunService.Heartbeat:Wait()
            end
        end

        if not Running then
            return
        end
        Bring.Claim(Item)
        RunService.Heartbeat:Wait()

        if Method ~= "ULTRA" and Root then
            pcall(function()
                Root.CFrame = CFrame.new(Target.Position + Vector3.new(0, 10, 0))
            end)
            RunService.Heartbeat:Wait()
        end

        pcall(function()
            -- items dropped on the player are fanned into a ring so they do not tower up
            if Method == "ULTRA" and not Options_.Target and Flags.BringSettingsLocation == "Player" then
                local Angle = Index / math.max(1, Count) * math.pi * 2
                Item:PivotTo(CFrame.new(Target.Position + Vector3.new(math.cos(Angle) * 7, 0, math.sin(Angle) * 7)))
            else
                Item:PivotTo(Target)
            end
        end)
        RunService.Heartbeat:Wait()

        if Method ~= "ULTRA" and Humanoid then
            pcall(function()
                Humanoid.PlatformStand = false
            end)
        end
        Bring.Release(Item)
        -- what the destination is FOR: the fire burns, the bench scraps. Moving an item next to
        -- them does nothing on its own, the remote still has to be sent.
        if Options_.After then
            RunService.Heartbeat:Wait()
            Util.Guard("bring after", Options_.After, Item)
        end
        Bring.Done += 1
        State.Brought += 1
    end

    task.spawn(function()
        Util.Identity(2)
        local Batches = Util.Batches(Items, Method == "ULTRA" and 10 or 1)
        for _, Batch in Batches do
            if not (Running and Alive()) then
                break
            end
            for Index, Item in Batch do
                if not (Running and Alive()) then
                    break
                end
                Util.Guard("bring", Carry, Item, Index, #Batch)
            end
            task.wait(Method == "ULTRA" and 0.03 or 0.05)
        end
        if not Options_.Quiet and Running then
            Alert(("brought %d of %d %s"):format(Bring.Done, Bring.Total, tostring(What)), 3)
        end
        Finish(false)
        if Options_.OnDone then
            Util.Guard("bring done", Options_.OnDone)
        end
    end)

    return Bring.Cancel
end)

-- collect --------------------------------------------------------------------------------------
-- Flowers, coin stacks and diamonds are all "stand on it and ask". The pickup call differs per
-- tag, so one resolver covers all three and every caller just hands over the model.
local Collect = { Busy = false, Done = 0, Total = 0, What = "" }

Collect.Pick = LPH_NO_VIRTUALIZE(function(Item)
    if typeof(Item) ~= "Instance" then
        return false
    end
    if Item:HasTag("Coins") then
        return Net.Invoke("RequestCollectCoints", Item) ~= nil
    end
    if Item:HasTag("Flower") then
        return Net.Invoke("RequestPickFlower", Item) ~= nil
    end
    if Item:GetAttribute("Diamonds") ~= nil then
        return Net.Fire("RequestTakeDiamonds", Item)
    end
    return false
end)

Collect.Diamond = LPH_NO_VIRTUALIZE(function(Item)
    if not Item then
        return false
    end
    return Net.Fire("RequestTakeDiamonds", Item)
end)

-- Walk the list nearest first and put the player back afterwards. The sweep is interruptible,
-- so a toggle switched off mid-run stops at the next item rather than at the end.
Collect.Sweep = LPH_NO_VIRTUALIZE(function(What, List, Keep)
    if Collect.Busy then
        Note(Collect.What .. " sweep already running")
        return 0
    end

    local Here = Char.Position()
    if not Here then
        return 0
    end

    local Sorted = {}
    for _, Item in List do
        if typeof(Item) == "Instance" and Item.Parent and Item.PrimaryPart then
            table.insert(Sorted, Item)
        end
    end
    if #Sorted == 0 then
        Note("no " .. tostring(What) .. " to collect")
        return 0
    end
    table.sort(Sorted, function(Left, Right)
        return (Left.PrimaryPart.Position - Here).Magnitude < (Right.PrimaryPart.Position - Here).Magnitude
    end)

    Collect.Busy, Collect.Done, Collect.Total, Collect.What = true, 0, #Sorted, tostring(What)

    local Spot = Move.Save()
    local Running = true
    local Handle = Stop.Show(function(ByUser)
        Running = false
        if ByUser and Keep then
            local Element = Toggles[Keep]
            if Element then
                pcall(Element.SetValue, Element, false)
            end
        end
    end)

    Alert(("collecting %d %s"):format(#Sorted, tostring(What)), 2)

    local Taken, Failed = 0, 0
    for _, Item in Sorted do
        if not (Running and Alive()) then
            break
        end
        if Keep and Flags[Keep] ~= true then
            break
        end
        if Item.Parent and Item.PrimaryPart then
            local Frame = Util.Pivot(Item)
            if Frame then
                Move.Teleport(Frame + Vector3.new(0, 5, 0))
                task.wait(0.15)
                if Collect.Pick(Item) then
                    Taken += 1
                    State.Collected += 1
                else
                    Failed += 1
                end
                Collect.Done = Taken + Failed
                task.wait(0.1)
            end
        end
    end

    pcall(Handle.Clean)
    Move.Return(Spot)
    Collect.Busy = false
    Alert(("collected %d of %d %s"):format(Taken, #Sorted, tostring(What)), 3)
    return Taken
end)

-- a live feed for the auto-collect toggles: grab whatever spawns, then hop straight back
Collect.Feed = LPH_NO_VIRTUALIZE(function(What, Folder, Match, Keep)
    return Library:GiveSignal(Folder.ChildAdded:Connect(function(Node)
        if Flags[Keep] ~= true or not Alive() then
            return
        end
        if not Match(Node) then
            return
        end
        task.spawn(function()
            Util.Identity(2)
            task.wait(0.1)
            if not (Node.Parent and Node.PrimaryPart) then
                return
            end
            local Spot = Char.Frame()
            if not Spot then
                return
            end
            local Frame = Util.Pivot(Node)
            if not Frame then
                return
            end
            Move.Teleport(Frame + Vector3.new(0, 5, 0))
            task.wait(0.1)
            if Collect.Pick(Node) then
                State.Collected += 1
            end
            Move.Teleport(Spot)
        end)
    end))
end)

Chest.Open = LPH_NO_VIRTUALIZE(function(Model, Announce)

Chest.List = LPH_NO_VIRTUALIZE(function()

-- Open every chest where it stands, no travelling. Cheap, and the only safe option when the
-- player is somewhere they should not leave.
Chest.OpenAll = LPH_NO_VIRTUALIZE(function()
    local Opened = 0
    for _, Model in Chest.List() do
        if not Alive() then
            break
        end
        if Chest.Open(Model, true) then
            Opened += 1
        end
        task.wait(0.1)
    end
    Alert(("opened %d chests"):format(Opened), 3)
    return Opened
end)

-- Pull the drops out of an opened chest. Simple takes whatever is in the blast radius; Safe
-- raycasts out from the drop point so items behind walls are left alone.
Chest.Drops = LPH_NO_VIRTUALIZE(function(Model, Radius)
    local Out = {}
    local Drop = Model:FindFirstChild("ItemDrop")
    local Origin = (Drop and Drop:IsA("BasePart") and Drop.Position)
        or (Model.PrimaryPart and Model.PrimaryPart.Position)
    if not Origin then
        return Out
    end

    local Cap = Tool.BagCap()
    if Cap <= 0 then
        Cap = 10
    end

    local function Interactive(Node)
        local Kind = Node:GetAttribute("Interaction")
        return Kind == "Item" or Kind == "Tool"
    end

    local function Consider(Node)
        for _ = 1, 3 do
            if not Node or Node == Workspace then
                return
            end
            if Interactive(Node) then
                if not table.find(Out, Node) and #Out < Cap then
                    table.insert(Out, Node)
                end
                return
            end
            Node = Node.Parent
        end
    end

    if tostring(Flags.AutoChestMode or "Safe") == "Simple" then
        local Items = Workspace:FindFirstChild("Items")
        if Items then
            for _, Node in Items:GetChildren() do
                if Node:IsA("Model") and Node.PrimaryPart
                    and (Node.PrimaryPart.Position - Origin).Magnitude <= (Radius or 5) * 3 then
                    Consider(Node)
                end
            end
        end
        return Out
    end

    local Params = RaycastParams.new()
    Params.FilterType = Enum.RaycastFilterType.Exclude
    Params.FilterDescendantsInstances = { Model, LocalPlayer.Character }

    local Step = Radius or 5
    local Directions = {
        Vector3.new(1, 0, 0), Vector3.new(-1, 0, 0), Vector3.new(0, 0, 1), Vector3.new(0, 0, -1),
        Vector3.new(1, 0, 1).Unit, Vector3.new(-1, 0, 1).Unit,
        Vector3.new(1, 0, -1).Unit, Vector3.new(-1, 0, -1).Unit,
        Vector3.new(0, 1, 0), Vector3.new(0, -1, 0),
    }
    for Reach = Step, Step * 7, Step do
        for _, Direction in Directions do
            local Hit = Workspace:Raycast(Origin, Direction * Reach, Params)
            if Hit and Hit.Instance then
                Consider(Hit.Instance)
            end
        end
        if #Out >= Cap then
            break
        end
    end
    return Out
end)

-- Loot one chest into the sack, then dump the sack at the drop point. The sack is the only way
-- to carry more than one item, so a missing sack is a hard stop rather than a silent no-op.
Chest.Loot = LPH_NO_VIRTUALIZE(function(Model, Target, Radius, Delay)
    if not (Model and Model:IsA("Model") and Model.PrimaryPart) then
        return false
    end

    local Sack = Tool.Held("Sack")
    if not Sack then
        Alert("equip your sack before looting chests", 4)
        return false
    end

    local Frame = Util.Pivot(Model)
    if not Frame then
        return false
    end

    Move.Teleport(Frame + Vector3.new(0, 5, 0))
    task.wait(Delay or 0.1)

    if not Chest.Opened(Model) then
        Chest.Open(Model)
        task.wait(Delay or 0.1)
    end

    local Taken = 0
    for _, Item in Chest.Drops(Model, Radius) do
        if not Alive() or Tool.BagFull() then
            break
        end
        if Tool.Store(Item, Sack) then
            Taken += 1
        end
        task.wait(0.05)
    end

    if Taken > 0 then
        Move.Teleport(Target)
        task.wait(0.15)
        Tool.DropAll()
    end
    return Taken > 0
end)

Chest.Sweep = LPH_NO_VIRTUALIZE(function()
    if Chest.Busy then
        return 0
    end

    local Where = tostring(Flags.AutoChestLocation or "Player")
    local Target = (Where == "Workbench" and Place.Bench()) or Char.Frame()
    if not Target then
        return 0
    end

    local Chests = Chest.List()
    if #Chests == 0 then
        Note("no unopened chests in range")
        return 0
    end

    Chest.Busy, Chest.Done, Chest.Total = true, 0, #Chests
    local Spot = Move.Save()
    local Running = true
    local Handle = Stop.Show(function(ByUser)
        Running = false
        if ByUser then
            local Element = Toggles.AutoChestToggle
            if Element then
                pcall(Element.SetValue, Element, false)
            end
        end
    end)

    local Radius = Util.Number(Flags.AutoChestRadius, 5)
    local Delay = Util.Number(Flags.AutoChestDelay, 0.1)
    local Looted = 0

    for _, Model in Chests do
        if not (Running and Alive() and Flags.AutoChestToggle == true) then
            break
        end
        if Util.Guard("chest", Chest.Loot, Model, Target, Radius, Delay) then
            Looted += 1
        end
        Chest.Done += 1
        task.wait(Delay)
    end

    pcall(Handle.Clean)
    Move.Return(Spot)
    Chest.Busy = false
    Alert(("looted %d chests"):format(Looted), 3)
    return Looted
end)

-- Saplings and log walls both place around the camp circle, so one geometry helper serves both.
-- The infinite-sapling path reuses a single captured sapling instance: the server only validates
-- that the model is a sapling, never that this client still owns it.
local Plant = { Sapling = nil, Armed = false, Index = 1, Radius = 0, Visual = {} }

Plant.Center = LPH_NO_VIRTUALIZE(function()
    local Circle = Util.Path(Workspace, "Map", "Campground", "alecCircle")

        Alert("the camp circle (alecCircle) is not in this map", 4)

-- Where the Nth placement goes, for each of the three patterns. Pure maths, no side effects, so
-- the blueprint preview and the placer can share it exactly.
Plant.Points = LPH_NO_VIRTUALIZE(function(Count)
    local Center = Plant.Center()
    if not Center then
        return {}
    end

    local Pattern = tostring(Flags.PlantMethod or "Circle")
    local Start = Util.Number(Flags.InitialRadius, 130)
    local Step = math.max(0.5, Util.Number(Flags.AngleIncrement, 2))
    local Height = Util.Number(Flags.PlantHeight, 0)
    local Limit = Start + 100
    local Out = {}

    if Pattern == "One Place" then
        local Here = Char.Position() or Center
        for _ = 1, Count do
            table.insert(Out, Here + Vector3.new(0, Height, 0))
        end
        return Out
    end

    if Pattern == "Square" then
        -- Walk the perimeter of a growing square: bottom edge, right edge, top edge, left edge.
        local Radius = Start
        while #Out < Count and Radius <= Limit do
            local Side = Radius * 2
            local Per = math.max(2, math.floor(Side / math.max(1, Start * math.rad(Step))))
            local Edges = {
                function(T)
                    return Vector3.new(Center.X - Radius + T * Side, 0, Center.Z - Radius)
                end,
                function(T)
                    return Vector3.new(Center.X + Radius, 0, Center.Z - Radius + T * Side)
                end,
                function(T)
                    return Vector3.new(Center.X + Radius - T * Side, 0, Center.Z + Radius)
                end,
                function(T)
                    return Vector3.new(Center.X - Radius, 0, Center.Z + Radius - T * Side)
                end,
            }
            for _, Edge in Edges do
                for Index = 0, Per - 1 do
                    if #Out >= Count then
                        break
                    end
                    local Point = Edge(Index / Per)
                    table.insert(Out, Vector3.new(Point.X, Center.Y + Height, Point.Z))
                end
                if #Out >= Count then
                    break
                end
            end
            Radius += 10
        end
        return Out
    end

    -- Circle: an Archimedean spiral out from the camp, one turn pushing the radius out by 10
    local Turn = math.rad(Step)
    for Index = 0, Count * 4 do
        if #Out >= Count then
            break
        end
        local Angle = Index * Turn
        local Radius = Start + Angle / (math.pi * 2) * 10
        if Radius > Limit then
            break
        end
        table.insert(Out, Vector3.new(
            Center.X + math.cos(Angle) * Radius,
            Center.Y + Height,
            Center.Z + math.sin(Angle) * Radius
        ))
    end
    return Out
end)

local function PlantPing(Point, Size)
    local Part = Instance.new("Part")
    Part.Shape = Enum.PartType.Ball
    Part.Size = Vector3.one * (Size or 2.2)
    Part.Anchored = true
    Part.CanCollide = false
    Part.CanQuery = false
    Part.Material = Enum.Material.Neon
    Part.Color = Color3.fromRGB(0, 255, 0)

    Part.CFrame = CFrame.new(Point + Vector3.new(0, 1.5, 0))

        Transparency = 1,
    }):Play()
    task.delay(1.4, function() pcall(function()
        Part:Destroy()
    end) end)
end

-- combat ---------------------------------------------------------------------------------------
-- Every swing is one ToolDamageObject invoke: { target, tool, sessionId, originCFrame }. The
-- server authorises the hit, so the hub's job is to pick targets and to send one well formed
-- call per target per tick - never a burst at one thing.
local Combat = { Session = 0, Targets = {}, Trees = {}, Scanned = 0, TreeScanned = 0, Hop = {},
    Took = 0, Lost = 0, Far = 0, Near = 0 }

local function NextSession()
    Combat.Session += 1
    return Combat.Session .. "_" .. math.abs(LocalPlayer.UserId)
end

-- CombatUtility.ApplyArmourToDamage, verbatim: armour caps at 100 and scales out 80% of a hit.
local function AfterArmour(Damage, Armour)
    return Damage * (1 - math.clamp(Armour, 0, 100) / 100 * 0.8)
end

-- the client-side half of CombatUtility.GetTotalArmour
local function ArmourOf(Target)
    local Owner = Players:GetPlayerFromCharacter(Target)
    if Owner then
        local Stats = Owner:FindFirstChild("Stats")
        local Worn = Stats and Stats:GetAttribute("Armour")
        if Worn then
            return Util.Number(Worn)
        end
    end
    return Util.Number(Target:GetAttribute("Armour"), 0)
end

-- EnemyHandler.GetLocalHealthRegistered: sums the damage this client has registered and not yet
-- seen the server confirm
local function Registered(Registers)
    local Total = 0
    for Key, Value in Registers:GetAttributes() do
        if Key:sub(1, 6) == "Local_" and Registers:GetAttribute(Key:sub(7)) == nil then
            Total += Util.Number(Value)
        end
    end
    return Total
end

-- ToolModule.GetModelFromPart decides what a swing is even allowed to touch, and the server
-- holds to the same rules, so the hub asks the same questions before it spends a swing:
--   * a resource has to list the held tool by name in AllowTool_<ToolName>
--   * if it carries a ToolTier, the tool has to match it - unless the tool has an AxeLevel, in
--     which case level 3 opens everything
--   * an NPC has to be undead, untamed (chicks excepted) and tagged NPC
--   * a web ball is Happy's Scythe only; anything else needs the AttackableObject attribute
-- The returned kind is what picks the damage figure further down, exactly as the tool classes do.
-- Two questions, deliberately kept apart.
--
-- Kind() answers "what is this, and may it be hit at all". It is the only thing that can stop a
-- swing, and it only stops on what the game itself refuses: NotAttackable, NotDamageable, a
-- destroyed resource, a dead or tamed animal. Everything else is classified generously - if a
-- model carries Health or a HitRegisters holder it is treated as a target - because a swing that
-- the server throws away costs one rate-limited request, while a swing the hub refuses to take
-- costs the whole feature.
local function Kind(Target)
    if typeof(Target) ~= "Instance" then
        return nil
    end
    if Target:GetAttribute("NotAttackable") ~= nil or Target:GetAttribute("NotDamageable") ~= nil then
        return nil
    end

    if Target:GetAttribute("Resource") ~= nil then
        if Target:GetAttribute("Destroyed") ~= nil then
            return nil
        end
        return "Resource"
    end

    if Players:GetPlayerFromCharacter(Target) then
        return "Player"
    end

    if Target:GetAttribute("AttackableObject") ~= nil or Util.Tagged(Target, "WebBall") then
        return "Object"
    end

    -- an animal or a cultist: the game wants an NPC child and the NPC tag, but a model with a
    -- humanoid or a health attribute is one either way, and reading it any stricter than this is
    -- what makes a kill aura swing at nothing
    local Living = Target:FindFirstChild("NPC") ~= nil
        or Util.Tagged(Target, "NPC")
        or Target:FindFirstChildOfClass("Humanoid") ~= nil
        or Target:GetAttribute("Health") ~= nil
    if Living then
        if Target:GetAttribute("Dead") ~= nil then
            return nil
        end
        if Target.Name ~= "Chick" and Target:GetAttribute("Tamed") ~= nil then
            return nil
        end
        return "NPC"
    end
    return nil
end

-- Accepts() answers the narrower question "would this tool land on this target", from the rules
-- the game's own hitbox filter uses: a resource names the tools it takes in AllowTool_<ToolName>,
-- and a ToolTier on the target has to be met by the tool's ToolTier (or an AxeLevel of 3+). This
-- never blocks a swing on its own - it is how the hub picks which tool to reach for.
local function Accepts(Target, Item)
    if typeof(Target) ~= "Instance" or typeof(Item) ~= "Instance" then
        return false
    end
    if Kind(Target) == nil then
        return false
    end
    if Target:GetAttribute("Resource") == nil then
        -- only resources carry the tool list; anything else any weapon can hit
        return true
    end
    local ToolName = Item:GetAttribute("ToolName")
    if ToolName == nil or Target:GetAttribute("AllowTool_" .. tostring(ToolName)) == nil then
        return false
    end
    local Tier = Target:GetAttribute("ToolTier")
    if Tier ~= nil then
        local Level = Item:GetAttribute("AxeLevel")
        if Level ~= nil then
            return Util.Number(Level) >= 3
        end
        return Util.Number(Tier) <= Util.Number(Item:GetAttribute("ToolTier"), 1)
    end
    return true
end

Combat.Kind = Kind
Combat.Accepts = Accepts

-- Nothing stops a player holding a scythe in front of a tree, and the tool in hand is what
-- decides whether a swing lands at all. Find the inventory weapon that this target does accept,
-- strongest resource hit first, so the chop and mining loops can reach for the right one.
Combat.Fit = LPH_NO_VIRTUALIZE(function(Target)
    local Inventory = LocalPlayer:FindFirstChild("Inventory")
    if not Inventory or typeof(Target) ~= "Instance" then
        return nil
    end
    local Best, Score = nil, -1
    for _, Item in Inventory:GetChildren() do
        if Tool.Kind(Item) == "MeleeWeapon" and Accepts(Target, Item) then
            local Hit = Util.Number(Item:GetAttribute("WeaponResourceDamage"),
                Util.Number(Item:GetAttribute("WeaponDamage"), 10))
            if Hit > Score then
                Best, Score = Item, Hit
            end
        end
    end
    return Best
end)

-- how long to leave a tool swap alone before swinging again
local SwapHold = {}

-- Every tool class gates its own swing on `time() >= LastSwing + Cooldown`, where the cooldown is
-- the tool's `ToolCooldown` attribute or 0.5s. The server holds to the same figure, so swinging
-- on every heartbeat is not a faster chop - it is a thousand refused invokes a second, a rate
-- limiter stuck at empty and nothing felled. One pass per cooldown, and a pass may still hit
-- every target in range, because that is what one swing of the real hitbox does.
Combat.Swung = 0

-- One line that says whether the server is taking the hits, and from how far. "0 taken" with a
-- climbing refused count is a server-side refusal, not a dead feature, and the two distances say
-- whether range is what it is refusing on.
Combat.Score = LPH_NO_VIRTUALIZE(function()
    if Combat.Took == 0 and Combat.Lost == 0 then
        return "no swings yet"
    end
    local Text = ("%s taken, %s refused"):format(Util.Commas(Combat.Took), Util.Commas(Combat.Lost))
    if Combat.Took > 0 then
        Text ..= (", furthest taken %d studs"):format(Combat.Far)
    end
    if Combat.Lost > 0 and Combat.Near > 0 then
        Text ..= (", closest refused %d"):format(Combat.Near)
    end
    return Text
end)

Combat.Ready = LPH_NO_VIRTUALIZE(function(Item)
    local Wait = 0.5
    if typeof(Item) == "Instance" then
        Wait = Util.Number(Item:GetAttribute("ToolCooldown"), 0.5)
    end
    if Wait < 0.1 then
        Wait = 0.1
    end
    return (os.clock() - Combat.Swung) >= Wait, Wait
end)

-- the Feaster class hits harder while overfed; the tool classes fold this in before registering
local function FeasterBonus()
    if LocalPlayer:GetAttribute("Class") ~= "Feaster" then
        return 1
    end
    local Settings = Tool.Module("GlobalSettings")
    local Cap = Util.Number(Settings and Settings.MaxHunger, 200)
    if Util.Number(LocalPlayer:GetAttribute("Hunger")) <= Cap then
        return 1
    end
    return Util.Number(Settings and Settings.FeasterDamageMultiplier, 1.15)
end

-- The server range-checks every hit. When Auto TP to Targets is on and everything in the list is
-- out of reach, hop to one that is - but only one hop per pass, so the player is not thrown
-- around the map on every tick.
Combat.Approach = LPH_NO_VIRTUALIZE(function(Target, Candidates, Key)
    if Flags.AutoTPTargets ~= true then
        return
    end
    local Here = Char.Position()
    local Point = Target and Util.Pivot(Target)
    if not (Here and Point) then
        return
    end
    if (Here - Point.Position).Magnitude <= 200 then
        return
    end

    local Held = Combat.Hop[Key]
    if Held and Held.Parent and (Here - Util.Pivot(Held).Position).Magnitude <= 200 then
        return
    end

    for _, Candidate in Candidates do
        local Spot = Candidate ~= Target and Util.Pivot(Candidate)
        if Spot and (Here - Spot.Position).Magnitude <= 200 then
            Combat.Hop[Key] = Candidate
            return
        end
    end

    -- nothing nearby at all: step to the target itself, high enough not to land inside it
    Combat.Hop[Key] = Target
    Move.GoTo(Point + Vector3.new(0, 40, 0), 20, "collect")
end)

Combat.Hit = LPH_NO_VIRTUALIZE(function(Target, IsTree, Candidates)
    if typeof(Target) ~= "Instance" then
        return false
    end

    local Item = Tool.Weapon()
    if not Item then
        return false
    end

    local Shape = Kind(Target)
    if not Shape then
        return false
    end

    -- The tool in hand may be the wrong one for this target - a scythe in front of a tree, a
    -- starter axe in front of a high-tier one. Reach for a better one and swing on the next
    -- pass, because the game equips through its own handler and the swap is not instant. If
    -- nothing in the bag is a better fit, swing with what is held rather than stall: the server
    -- is the judge, and a refused hit costs one request.
    if not Accepts(Target, Item) then
        local Swap = Combat.Fit(Target)
        if Swap and Swap ~= Item and (os.clock() - (SwapHold[Swap] or 0)) > 1 then
            SwapHold[Swap] = os.clock()
            Tool.Equip(Swap)
            Weapon.Stamp = 0
            return false
        end
    end

    local Frame = Char.Frame()
    if not Frame then
        return false
    end

    Combat.Approach(Target, Candidates or {}, IsTree and "tree" or "char")

    local Here = Char.Position()
    local Point = Util.Pivot(Target)

    -- The server does not take the client's word for a swing. The game's own tools call
    -- EnemyHandler.ApplyLocalDamage first, which writes the damage onto the target's HitRegisters
    -- as an attribute named "Local_<session>", and only then invoke ToolDamageObject with that
    -- same session id so the server can look the figure up. A session id that was never
    -- registered is worth nothing, which is why the old call - a made-up id, and no killing-blow
    -- flag either - swung forever and never felled anything.
    -- which of the two damage figures applies is the target's kind, not whether it is a
    -- resource: a web ball or a punching bag takes the resource figure too, and both default
    -- to 10 when the tool carries neither attribute
    local Resource = Target:GetAttribute("Resource")
    local Damage
    if Shape == "Player" or Shape == "NPC" then
        Damage = Util.Number(Item:GetAttribute("WeaponDamage"), 10)
    else
        Damage = Util.Number(Item:GetAttribute("WeaponResourceDamage"), 10)
    end

    -- the ice bonus is a property of the tool, read against an ice target
    local Ice = Item:GetAttribute("IceDamageMultiplier")
    if Ice and Resource == "IceBlock" then
        Damage *= Util.Number(Ice, 1)
    end
    Damage *= FeasterBonus()
    Damage = AfterArmour(Damage, ArmourOf(Target))

    local Session = NextSession()
    local Registers = Target:FindFirstChild("HitRegisters")
    if Registers then
        pcall(function()
            Registers:SetAttribute("Local_" .. Session, Damage)
        end)
    end

    -- a resource only drops when the client says this swing is the one that finishes it
    local Killing = false
    if Shape == "Resource" and Registers then
        Killing = Util.Number(Target:GetAttribute("Health")) - Registered(Registers) <= 0
    end

    local Health = Util.Number(Target:GetAttribute("Health"), 50)
    local Result = Net.Invoke("ToolDamageObject", Target, Item, Session, Frame, Killing)

    -- Keep score of what the server actually does with a swing. Nothing else in the hub can tell
    -- a refused hit from an accepted one, and the distance it stops accepting at is the single
    -- number that decides whether an aura range of 200 is useful or is throwing requests away.
    local Reach = (Point and Here) and (Here - Point.Position).Magnitude or nil
    local Good = Result ~= nil and not (type(Result) == "table" and not Result.Success)
    if Good then
        Combat.Took += 1
        if Reach and Reach > Combat.Far then
            Combat.Far = Reach
        end
    else
        Combat.Lost += 1
        if Reach and (Combat.Near == 0 or Reach < Combat.Near) then
            Combat.Near = Reach
        end
    end

    -- the game rolls its local registration back whenever the server refuses, so a rejected swing
    -- cannot leave a phantom chunk missing from the health bar
    if not Good then
        if Registers then
            pcall(function()
                Registers:SetAttribute("Local_" .. Session, nil)
            end)
        end
        return false
    end

    State.Hits += 1
    if IsTree then
        State.Chopped += 1
    end

    -- the visualiser is for the slow jobs: trees, ice and ore, never a crowd of mobs
    local Visual = IsTree and Flags.AutoChopTreesVisualiser == true
    if not Visual then
        local Name = tostring(Target)
        Visual = Name == "IceBlock" or Name:find("Node") ~= nil
    end
    if Visual then
        Bar.Show(Target, Util.Number(Target:GetAttribute("Health"), Health))
    end
    return true
end)

-- Happy's structures append a suffix to their model names, so a selection of "Cultist" has to
-- still match "Cultist With Happy's Head"
local function BaseName(Name)
    if type(Name) ~= "string" then
        return Name
    end
    return (Name:gsub("%s+With%s+Happy's%s+.*$", ""))
end

-- One scan feeds every aura. It is cached for a second: the aura loops run on Heartbeat, and
-- re-walking workspace.Characters sixty times a second is what used to cost the frame rate.
Combat.Scan = LPH_NO_VIRTUALIZE(function(Radius)
    Radius = Util.Number(Radius)
    if Radius <= 0 then
        return {}
    end
    if os.clock() - Combat.Scanned < 1 then
        return Combat.Targets
    end
    Combat.Scanned = os.clock()

    local Out = {}
    local Here = Char.Position()
    if not Here then
        Combat.Targets = Out
        return Out
    end

    local function Consider(Model)
        local Point = Model.PrimaryPart and Util.Pivot(Model)
        if Point and (Here - Point.Position).Magnitude <= Radius then
            table.insert(Out, Model)
        end
    end

    local Wanted = Util.Picked("KillauraEntities")

    if Flags.KillAuraToggle == true then
        local Characters = Workspace:FindFirstChild("Characters")
        for _, Model in (Characters and Characters:GetChildren() or {}) do
            if Model ~= LocalPlayer.Character and Model:IsA("Model")
                and Model:GetAttribute("NotAttackable") == nil then
                local Name = Wanted[Model.Name] and Model.Name or BaseName(Model.Name)
                if Wanted[Name] or Name:find("Child") then
                    Consider(Model)
                end
            end
        end

        if Wanted.Turret then
            for _, Model in Reg.Turrets do
                if Model.Parent then
                    Consider(Model)
                end
            end
        end

        -- loose aliens are parked in Items, not Characters
        if Wanted.Alien then
            local Items = Workspace:FindFirstChild("Items")
            for _, Model in (Items and Items:GetChildren() or {}) do
                if Model.Name == "Alien" and Model:IsA("Model") then
                    Consider(Model)
                end
            end
        end
    end

    if Flags.IceAuraToggle == true then
        for _, Model in Reg.IceBlocks do
            local Block = Model.Parent and Model:FindFirstChild("IceBlock")
            if Block then
                table.insert(Out, Block)
            end
        end
    end

    if Flags.OreAuraToggle == true then
        for _, Crater in Reg.Craters do
            local Nodes = Crater.Parent and Crater:FindFirstChild("OreNodes")
            for _, Node in (Nodes and Nodes:GetChildren() or {}) do
                if Node.Name:find("Node") then
                    table.insert(Out, Node)
                end
            end
        end
    end

    if Flags.WebBallAuraToggle == true then
        for _, Model in Reg.WebBalls do
            if Model.Parent and Model:IsA("Model") then
                table.insert(Out, Model)
            end
        end
    end

    Combat.Targets = Out
    return Out
end)

Combat.ScanTrees = LPH_NO_VIRTUALIZE(function(Radius)
    Radius = Util.Number(Radius)
    if Radius <= 0 then
        return {}
    end
    if os.clock() - Combat.TreeScanned < 1 then
        return Combat.Trees
    end
    Combat.TreeScanned = os.clock()

    local Out = {}
    local Here = Char.Position()
    if not Here then
        Combat.Trees = Out
        return Out
    end

    local Wanted = Util.Picked("AutoChopTreeType")
    local BigWanted = Wanted["Big Tree"] == true
    local function Consider(Model)
        if not Model:IsA("Model") or not Model.PrimaryPart then
            return
        end
        -- a tree nothing in the bag can fell is still listed: the chop job reaches for a better
        -- tool itself, and hiding the tree here is how the whole feature went quiet
        if Model:GetAttribute("Destroyed") ~= nil then
            return
        end
        local Lower = Model.Name:lower()
        local Big = Lower:find("treebig") ~= nil or Lower:find("bigtree") ~= nil
        local Want = Wanted[Model.Name] == true
            or (Big and (BigWanted or Lower:find("corrupted") ~= nil))
        if not Want then
            return
        end
        local Point = Util.Pivot(Model)
        if Point and (Here - Point.Position).Magnitude <= Radius then
            table.insert(Out, Model)
        end
    end

    local Map = Workspace:FindFirstChild("Map")
    for _, Name in { "Foliage", "Landmarks" } do
        local Folder = Map and Map:FindFirstChild(Name)
        for _, Model in (Folder and Folder:GetChildren() or {}) do
            Consider(Model)
        end
    end

    Combat.Trees = Out
    return Out
end)

-- The torch stun is its own remote and takes the whole model. It is a RemoteFunction, so one
-- call per monster per pass and nothing more.
Combat.Stun = LPH_NO_VIRTUALIZE(function()
    local Here = Char.Position()
    if not Here then
        return 0
    end

    local Radius = Util.Number(Flags.AutoStunRange, 50)
    local Wanted = Util.Picked("AutoStunChoiceDropdown")
    local Stunned = 0

    for Name, List in Reg.Monsters do
        if Wanted[Name] then
            for _, Model in List do
                if not Alive() then
                    return Stunned
                end
                local Point = Model.Parent and Model.PrimaryPart and Util.Pivot(Model)
                if Point and (Here - Point.Position).Magnitude <= Radius then
                    if Net.Invoke("MonsterHitByTorch", Model) ~= nil then
                        Stunned += 1
                        State.Stunned += 1
                    end
                end
            end
        end
    end
    return Stunned
end)

-- the nearest edible of a chosen kind, from the ground registry
Live.FindFood = LPH_NO_VIRTUALIZE(function()
    local Here = Char.Position()
    local Wanted = Util.Picked("AutoFillHungerDropdown")
    local Best, Gap = nil, math.huge
    for Name in Wanted do
        for _, Item in Reg.Named[Name] or {} do
            if Draggable(Item) then
                local Point = Util.Pivot(Item)
                local Distance = (Here and Point) and (Here - Point.Position).Magnitude or 0
                if Distance < Gap then
                    Best, Gap = Item, Distance
                end
            end
        end
    end
    return Best
end)

Live.Eat = LPH_NO_VIRTUALIZE(function()
    if Live.Hunger() >= Util.Number(Flags.AutoEatHealth, 80) then
        return false
    end

    -- prefer something already carried over a trip across the map
    local Wanted = Util.Picked("AutoFillHungerDropdown")
    local Inventory = LocalPlayer:FindFirstChild("Inventory")
    for _, Item in (Inventory and Inventory:GetChildren() or {}) do
        if Wanted[Item.Name] then
            Note("eating " .. Item.Name)
            return Tool.Consume(Item)
        end
    end

    local Ground = Live.FindFood()
    if not Ground then
        return false
    end
    local Spot = Char.Frame()
    Bring.Claim(Ground)
    Move.GoTo(Util.Pivot(Ground), 6, "collect")
    task.wait(0.2)
    local Ok = Net.Invoke("RequestHotbarItem", Ground) ~= nil
    if Ok then
        task.wait(0.2)
        local Picked = Tool.Find(Ground.Name, true)
        if Picked then
            Tool.Consume(Picked)
        end
    end
    if Spot then
        Move.Teleport(Spot)
    end
    return Ok
end)

Live.Bandage = LPH_NO_VIRTUALIZE(function()
    local _, Humanoid = Char.Parts()

-- Hitbox growth is local only, and the originals are kept so unload puts every entity back.
local Hitbox = { Original = {}, Connection = nil }

Hitbox.Apply = LPH_NO_VIRTUALIZE(function(Model, Scale)
    if not (Model:IsA("Model") and Model.PrimaryPart) then
        return
    end
    pcall(function()
        if not Hitbox.Original[Model] then
            Hitbox.Original[Model] = Model.PrimaryPart.Size
        end
        Model.PrimaryPart.Transparency = 1
        Model.PrimaryPart.CanCollide = false
        Model.PrimaryPart.CanQuery = true
        Model.PrimaryPart.Size = Hitbox.Original[Model] * Scale
    end)
end)

Hitbox.All = LPH_NO_VIRTUALIZE(function(Scale)
    local Characters = Workspace:FindFirstChild("Characters")
    for _, Model in (Characters and Characters:GetChildren() or {}) do
        Hitbox.Apply(Model, Scale)
    end
end)

Hitbox.Restore = LPH_NO_VIRTUALIZE(function()
    for Model, Size in Hitbox.Original do
        pcall(function()
            if Model and Model.Parent and Model.PrimaryPart then
                Model.PrimaryPart.Size = Size
            end
        end)
    end
    table.clear(Hitbox.Original)
    if Hitbox.Connection then
        pcall(function()
            Hitbox.Connection:Disconnect()
        end)
        Hitbox.Connection = nil
    end
end)

-- Entity godmode: the two remotes below are how an NPC tells the server it hit you. Destroying
-- the client's copy means the call can never be made - nothing is sent, and a rejoin undoes it.
local Godmode = { Hip = nil }

Godmode.Invincible = LPH_NO_VIRTUALIZE(function()
    local Folder = Net.Folder or ReplicatedStorage:FindFirstChild("RemoteEvents")
    if not Folder then
        return false
    end
    local Removed = 0
    for _, Name in { "ClientTriggerNPCAttack", "NPCProjectileDamagePlayer" } do
        local Remote = Folder:FindFirstChild(Name)
        if Remote then
            pcall(function()
                Remote:Destroy()
            end)
            Removed += 1
        end
    end
    if Removed > 0 then
        Alert("entity damage can no longer reach you", 3)
    end
    return Removed > 0
end)

-- Standing a long way off the floor puts you out of melee reach of ground entities. Reverted on
-- unload and on respawn, since a stuck hip height makes the character unplayable.
Godmode.Hover = LPH_NO_VIRTUALIZE(function(On)
    local _, Humanoid = Char.Parts()
    if not Humanoid then
        return false
    end
    if On then
        if Godmode.Hip == nil then
            Godmode.Hip = Humanoid.HipHeight
        end
        Humanoid.HipHeight = Util.Number(Flags.HipHeightChanger, 20)
    elseif Godmode.Hip ~= nil then
        Humanoid.HipHeight = Godmode.Hip
        Godmode.Hip = nil
    end
    return true
end)

-- The map is revealed by telling the server which cells you have seen. UpdateMapCells accepts
-- the whole grid in one call, which is what the radar reads, so the spiral flight is only there
-- for the landmarks that need the client to physically pass near them.
local Explore = { Running = false, Percent = 0, Abort = false, Gui = nil, Bar = nil, Label = nil }

Explore.Cells = LPH_NO_VIRTUALIZE(function()
    local Cells = {}
    for X = 0, 69 do
        for Z = 0, 69 do
            table.insert(Cells, { CellId = X .. "_" .. Z })
        end
    end
    return Cells
end)

Explore.Reveal = LPH_NO_VIRTUALIZE(function()
    local Cells = Explore.Cells()
    Net.Fire("UpdateMapCells", Cells)

    -- the radar reads a flag per landmark, set from the Landmarks folder's own attribute names
    Util.Guard("radar", function()
        local Landmarks = Util.Path(Workspace, "Map", "Landmarks")
        local Shops = ReplicatedStorage:FindFirstChild("Shops")
        local Radar = Shops and Shops:FindFirstChild("Radar")
        if not (Landmarks and Radar) then
            return
        end
        for Key in Landmarks:GetAttributes() do
            local Name = tostring(Key)
            if Name:sub(1, 7) == "Static_" then
                Name = Name:sub(8)
            elseif Name:sub(1, 8) == "Dynamic_" then
                Name = Name:sub(9)
            end
            Name = Name:gsub(" ", "_")
            if Name ~= "" then
                Radar:SetAttribute(Name, true)
            end
        end
    end)

    -- the client's own handler, so the local map UI redraws without a rejoin
    Util.Guard("client map", function()
        local Scripts = LocalPlayer:FindFirstChild("PlayerScripts")
        local Client = Scripts and Util.Require(Scripts:FindFirstChild("Client"), 6)
        if type(Client) == "table" and type(Client.Events) == "table" and Client.Events.UpdateMapCells then
            Client.Events.UpdateMapCells:Fire(nil, Cells)
        end
    end)
end)

local function ExploreProgress(Percent)
    Explore.Percent = math.clamp(Percent, 0, 100)
    if Explore.Bar then
        Explore.Bar:TweenSize(UDim2.fromScale(Explore.Percent / 100, 1),
            Enum.EasingDirection.Out, Enum.EasingStyle.Sine, 0.2, true)
    end
    if Explore.Label then
        Explore.Label.Text = ("%d%%"):format(Explore.Percent)
    end
end

-- Fly a widening circle over the map so every landmark streams in. The character is moved by
-- short interpolated steps rather than one jump per point, which is what makes the landmarks
-- actually register instead of being skipped over.
Explore.Fly = LPH_NO_VIRTUALIZE(function(OnStep)
    local _, _, Root = Char.Parts()
    if not Root then
        Alert("you have to be alive to reveal the map", 3)
        return false
    end

    Explore.Abort = false
    local Center = Vector3.new(0, Root.Position.Y + 45, 0)
    local Turns = 16
    local Turn = 0

    local function Slide(Goal, Steps)
        local Start = Root.CFrame
        for Step = 1, Steps do
            if Explore.Abort or not Alive() then
                return false
            end
            local _, _, Live_ = Char.Parts()
            if not Live_ then
                return false
            end
            Root = Live_
            Root.CFrame = Start:Lerp(Goal, Step / Steps)
            RunService.Heartbeat:Wait()
        end
        return true
    end

    for Radius = 50, 800, 50 do
        if Explore.Abort or not Alive() then
            break
        end
        for Index = 1, 24 do
            if Explore.Abort or not Alive() then
                break
            end
            local Angle = Turn * math.pi + Index / 24 * math.pi * 2
            local Point = Center + Vector3.new(math.cos(Angle) * Radius, 0, math.sin(Angle) * Radius)
            if not Slide(CFrame.lookAt(Point, Point + Vector3.new(-math.sin(Angle), 0, math.cos(Angle))), 10) then
                break
            end
        end
        Turn += 1
        if OnStep then
            OnStep(Turn, Turns)
        end
        ExploreProgress(Turn / Turns * 100)
    end

    return not Explore.Abort
end)

Explore.Run = LPH_NO_VIRTUALIZE(function()
    if Explore.Running then
        Alert("the map reveal is already running", 2)
        return
    end
    Explore.Running = true
    Explore.Clean()
    Explore.Gui, Explore.Bar, Explore.Label = ExploreGui()

    local Spot = Move.Save()
    local Handle = Stop.Show(function()
        Explore.Abort = true
    end)

    Explore.Reveal()
    Explore.Fly()

    ExploreProgress(100)
    if Explore.Label then
        Explore.Label.Text = "Exploration complete"
    end
    if Explore.Bar then
        Explore.Bar.BackgroundColor3 = Color3.fromRGB(0, 255, 100)
    end

    pcall(Handle.Clean)
    Move.Return(Spot)
    Alert("revealed the map", 4)
    task.delay(1.5, Explore.Clean)
    Explore.Running = false
end)

-- day farm -------------------------------------------------------------------------------------
-- An eight step opening routine: reveal the map, rescue the four lost children, build the beds,
-- raise the lightning rods, place the crock pot, clear the big trees. Steps can be skipped from

Farm.Order = {

    { Key = "REPEAT_KIDS", Name = "Children Rescue (second pass)" },

Farm.StepNames = LPH_NO_VIRTUALIZE(function()
    local Out = {}

Farm.SetSkips = LPH_NO_VIRTUALIZE(function(Value)

    end

-- children -------------------------------------------------------------------------------------
-- Each child is tracked by the notice board, found in workspace.Map.Characters, carried in the
-- sack and dropped at the fire. A child that has not streamed in yet is hunted around its saved
-- location, since the server only spawns the model when a player is close enough.
local Kids = {}

Kids.Found = LPH_NO_VIRTUALIZE(function(Key)
    local Board = Util.Path(Workspace, "Map", "Campground", "NoticeBoard", "MissingKidTracker", tostring(Key))
    return Board and Board:GetAttribute("Found") == true
end)

Kids.Saved = LPH_NO_VIRTUALIZE(function(Key)
    local Folder = Util.Path(Workspace, "Map", "MissingKids")
    local Value = Folder and Folder:GetAttribute(tostring(Key))
    if typeof(Value) == "Vector3" then
        return CFrame.new(Value)
    end
    if typeof(Value) == "CFrame" then
        return Value
    end
    local Node = Folder and Folder:FindFirstChild(tostring(Key))
    return Node and Util.Pivot(Node) or nil
end)

Kids.Model = LPH_NO_VIRTUALIZE(function(Name)
    local Characters = Util.Path(Workspace, "Map", "Characters") or Workspace:FindFirstChild("Characters")
    local Model = Characters and Characters:FindFirstChild(Name)
    if Model then
        return Model
    end
    local Items = Workspace:FindFirstChild("Items")
    Model = Items and Items:FindFirstChild(Name)
    if Model then
        return Model
    end
    local Bag = LocalPlayer:FindFirstChild("ItemBag")
    return Bag and Bag:FindFirstChild(Name) or nil
end)

Kids.Count = LPH_NO_VIRTUALIZE(function()
    local Done = 0
    for Key in World.Kids do
        if Kids.Found(Key) then
            Done += 1
        end
    end
    return Done
end)

-- A child has a prompt when it is standing; a downed one has to go straight into the sack.
Kids.Take = LPH_NO_VIRTUALIZE(function(Model)
    if not Model then
        return false
    end
    local Bag = LocalPlayer:FindFirstChild("ItemBag")
    if Bag and Model.Parent == Bag then
        return true
    end

    local Sack = Tool.Held("Sack")
    if not Sack then
        return false
    end

    local Head = Model:FindFirstChild("Head")
    local Prompt = Head and Util.Path(Head, "ProximityAttachment", "ProximityInteraction")
    if Prompt and Prompt:IsA("ProximityPrompt") then
        Compat.FirePrompt(Prompt, Flags.NoInteractDelay == true)
        task.wait(0.4)
        if Bag and Bag:FindFirstChild(Model.Name) then
            return true
        end
    end
    return Tool.Store(Model, Sack)
end)

-- Hop around the saved spot until the server streams the child in. Capped, because a child that
-- another player already rescued never appears and the loop would otherwise never end.
Kids.Hunt = LPH_NO_VIRTUALIZE(function(Key, Name)
    local Base = Kids.Saved(Key)
    if not Base then
        return false
    end

    for Attempt = 1, 30 do
        if not Alive() then
            return false
        end
        Move.Teleport(CFrame.new(Base.Position + Vector3.new(math.random(-10, 10), 5, math.random(-10, 10))))
        task.wait(0.3)
        local Model = Kids.Model(Name)
        if Model and Model:FindFirstChild("HumanoidRootPart") then
            local Frame = Util.Pivot(Model)
            if Frame then
                Move.Teleport(Frame + Vector3.new(0, 5, 0))
                task.wait(0.4)
                return Kids.Take(Model)
            end
        end
        if Attempt % 10 == 0 then
            Stage("KIDS", ("hunting %s (%d/30)"):format(Name, Attempt))
        end
    end
    return false
end)

Kids.Rescue = LPH_NO_VIRTUALIZE(function(Second)
    local Key = Second and "REPEAT_KIDS" or "KIDS"
    Stage(Key, "checking the notice board", 0)
    if Farm.Skipping(Key) then
        Stage(Key, "skipped", 100)
        return 0
    end

    local Missing = {}
    for Field, Name in World.Kids do
        if not Kids.Found(Field) then
            table.insert(Missing, { Field = Field, Name = Name })
        end
    end
    State.Kids = Kids.Count()

    if #Missing == 0 then
        Stage(Key, "every child is already home", 100)
        return 0
    end

    local Rescued = 0
    for Index, Entry in Missing do
        if not Alive() then
            break
        end
        Stage(Key, "rescuing " .. Entry.Name, Index / #Missing * 100)
        local Model = Kids.Model(Entry.Name)
        if Model then
            local Bag = LocalPlayer:FindFirstChild("ItemBag")
            if not (Bag and Model.Parent == Bag) then
                local Frame = Util.Pivot(Model)
                if Frame then
                    Move.Teleport(Frame + Vector3.new(0, 5, 0))
                    task.wait(0.6)
                    Kids.Take(Kids.Model(Entry.Name) or Model)
                    task.wait(0.4)
                end
            end
            Rescued += 1
        elseif Kids.Hunt(Entry.Field, Entry.Name) then
            Rescued += 1
        end
    end

    Stage(Key, "carrying them back to camp", 100)
    local Camp_ = Place.Camp()
    if Camp_ then
        Move.Teleport(Camp_ + Vector3.new(0, 15, 0))
        task.wait(1.2)
    end
    Tool.DropAll(function(Item)
        return Item.Name:find("Lost Child") ~= nil
    end)

    State.Kids = Kids.Count()
    Stage(Key, ("returned %d children"):format(Rescued), 100)
    return Rescued
end)

-- Each wave has a touch trigger. Growing it and brushing past means the waves start while the
-- player is already back where they were, instead of standing in the spawn.
Hold.Waves = LPH_NO_VIRTUALIZE(function()
    local Functional = Hold.Model and Hold.Model:FindFirstChild("Functional")
    if not Functional then
        return 0
    end

    local Spot = Char.Frame()
    local Tripped = 0
    for _, Node in Functional:GetChildren() do
        if not Alive() then
            break
        end
        local Zone = Node.Name:find("EnemyWave") and Util.Path(Node, "Wave1", "TriggerZone")
        if Zone and Zone:IsA("BasePart") and Zone:FindFirstChildOfClass("TouchTransmitter") then
            pcall(function()
                Zone.Size = Vector3.new(1000, 2048, 2048)
            end)
            Move.Teleport(CFrame.new(Zone.Position) + Vector3.new(0, 10, 0))
            task.wait(0.3)
            if Spot then
                Move.Teleport(Spot + Vector3.new(0, 10, 0))
            end
            Tripped += 1
        end
    end
    return Tripped
end)

Hold.Tick = LPH_NO_VIRTUALIZE(function()
    if not Hold.Find() then
        Hold.Status = "looking for the stronghold"
        State.Stronghold = Hold.Status
        return false
    end

    local Functional = Hold.Model:FindFirstChild("Functional")
    Hold.Level = Functional and tostring(Functional:GetAttribute("Level") or "-") or "-"
    State.StrongholdLevel = Hold.Level

    local At = Hold.OpenTime()
    local Left = At and (At - Workspace:GetServerTimeNow()) or nil
    Hold.Opens = Left and (Left <= 0 and "open" or Util.Clock(Left)) or "-"
    State.StrongholdOpen = Hold.Opens

    local Items = Workspace:FindFirstChild("Items")
    local Diamond = Items and Items:FindFirstChild("Diamond")
    if Diamond then
        Collect.Diamond(Diamond)

    Hold.Chest()

    if Hold.IsOpen() then
        Hold.Status = "open - raiding"
        State.Stronghold = Hold.Status
        if not Hold.Armed then
            Hold.Armed = true
            Hold.Doors()
            task.wait(0.5)
            Hold.Waves()
            Hold.Teleport()
        end
        return true
    end

    Hold.Armed = false
    Hold.Status = "shut, opens in " .. Hold.Opens
    State.Stronghold = Hold.Status
    return false
end)

        Hold.Running = false
        State.Stronghold = ""
    end)
end)

-- minigame patches -----------------------------------------------------------------------------
-- The taming flute and the fishing rod both keep their difficulty in a client module, so these
-- are purely local edits. Both are reverted on unload, and when the executor has neither
-- modifyconstant nor debug.getupvalue the feature says so instead of silently doing nothing.
local Patch = { Tuning = {}, Constants = {}, Bars = {} }

Patch.Supported = LPH_NO_VIRTUALIZE(function()
    return type(modifyconstant) == "function"
        or (type(debug) == "table" and type(debug.getupvalue) == "function")
end)

-- Replace fields inside the tuning table a minigame holds as an upvalue. The original values are
-- recorded per field so unload can put every one of them back.
Patch.Fields = LPH_NO_VIRTUALIZE(function(Key, Module, Method, Fields)
    if not (type(debug) == "table" and type(debug.getupvalue) == "function") then
        Alert("your executor cannot read module upvalues", 4)
        return false
    end

    local Loaded = Util.Require(Module, 6)
    local Entry = Loaded and Loaded[Method]
    if type(Entry) ~= "function" then
        Alert("could not find " .. tostring(Method) .. " in this build", 4)
        return false
    end

    local Ok, Tuning = pcall(debug.getupvalue, Entry, 1)
    if not (Ok and type(Tuning) == "table") then
        Alert("the tuning table moved in this game version", 4)
        return false
    end

    local Saved = {}
    for _, Group in pairs(Tuning) do
        if type(Group) == "table" then
            local Before = {}
            for Field, Value in Fields do
                Before[Field] = Group[Field]
                Group[Field] = Value
            end
            table.insert(Saved, { Group = Group, Before = Before })
        end
    end

    Patch.Tuning[Key] = Saved
    return #Saved > 0
end)

Patch.Restore = LPH_NO_VIRTUALIZE(function(Key)
    local Saved = Patch.Tuning[Key]
    if not Saved then
        return
    end
    Patch.Tuning[Key] = nil
    for _, Entry in Saved do
        pcall(function()
            for Field, Value in Entry.Before do
                Entry.Group[Field] = Value
            end
        end)
    end
end)

-- Swap one constant inside a function. Used where the game compares two fields and flipping the
-- name read makes the comparison always pass.
Patch.Constant = LPH_NO_VIRTUALIZE(function(Key, Module, Pick, From, To)
    if type(modifyconstant) ~= "function" then
        return false, "no modifyconstant"
    end
    local Loaded = Util.Require(Module, 6)
    local Target = Loaded and Pick(Loaded)
    if type(Target) ~= "function" then
        return false, "not in this build"
    end
    local Ok, Error = pcall(modifyconstant, Target, From, To)
    if not Ok then
        return false, tostring(Error)
    end
    Patch.Constants[Key] = { Target = Target, From = From, To = To }
    return true
end)

Patch.Unconstant = LPH_NO_VIRTUALIZE(function(Key)
    local Entry = Patch.Constants[Key]
    if not Entry then
        return
    end
    Patch.Constants[Key] = nil
    pcall(modifyconstant, Entry.Target, Entry.To, Entry.From)
end)

-- Fallback for executors without modifyconstant: stretch the minigame's timing bar so the whole
-- track counts as a hit. Local, visual, and put back when the toggle goes off.
Patch.Bar = LPH_NO_VIRTUALIZE(function(Key, Path, On)
    local Gui = LocalPlayer:FindFirstChild("PlayerGui")
    local Bar = Gui and Util.Path(Gui, "Interface", Path, "TimingBar", "Bar")
    if not Bar then
        return false
    end
    if On then
        if not Patch.Bars[Key] then
            Patch.Bars[Key] = Bar.Size
        end
        Bar.Size = UDim2.new(Bar.Size.X.Scale, Bar.Size.X.Offset, 1, 100)
    elseif Patch.Bars[Key] then
        Bar.Size = Patch.Bars[Key]
        Patch.Bars[Key] = nil
    end
    return true
end)

Patch.RestoreAll = LPH_NO_VIRTUALIZE(function()
    for Key in Patch.Tuning do
        Patch.Restore(Key)
    end
    for Key in Patch.Constants do
        Patch.Unconstant(Key)
    end
    for Key in Patch.Bars do
        Patch.Bar(Key, "", false)
    end
end)

-- All local, all reversible: nothing here is sent and everything is put back on unload.
local Perf = { Saved = nil }

Perf.Low = LPH_NO_VIRTUALIZE(function(On)
    if On then
        if not Perf.Saved then
            local Terrain = Workspace:FindFirstChildOfClass("Terrain")
            Perf.Saved = {
                Quality = settings() and nil,
                Global = Lighting.GlobalShadows,
                Technology = nil,
                Terrain = Terrain and {
                    Part = Terrain,
                    WaterWaveSize = Terrain.WaterWaveSize,
                    WaterWaveSpeed = Terrain.WaterWaveSpeed,
                    WaterReflectance = Terrain.WaterReflectance,
                    WaterTransparency = Terrain.WaterTransparency,
                } or nil,
                Effects = {},
            }
            for _, Node in Lighting:GetChildren() do
                if Node:IsA("PostEffect") then
                    table.insert(Perf.Saved.Effects, { Part = Node, Enabled = Node.Enabled })
                end
            end
        end

        Lighting.GlobalShadows = false
        local Terrain = Workspace:FindFirstChildOfClass("Terrain")
        if Terrain then
            Terrain.WaterWaveSize = 0
            Terrain.WaterWaveSpeed = 0
            Terrain.WaterReflectance = 0
            Terrain.WaterTransparency = 0
        end
        for _, Node in Lighting:GetChildren() do
            if Node:IsA("PostEffect") then
                Node.Enabled = false
            end
        end
        pcall(function()
            RunService:Set3dRenderingEnabled(true)
        end)
        Alert("low graphics on", 3)
        return
    end

    local Saved = Perf.Saved
    if not Saved then
        return
    end
    Perf.Saved = nil
    pcall(function()
        Lighting.GlobalShadows = Saved.Global
    end)
    if Saved.Terrain and Saved.Terrain.Part then
        pcall(function()
            local Terrain = Saved.Terrain.Part
            Terrain.WaterWaveSize = Saved.Terrain.WaterWaveSize
            Terrain.WaterWaveSpeed = Saved.Terrain.WaterWaveSpeed
            Terrain.WaterReflectance = Saved.Terrain.WaterReflectance
            Terrain.WaterTransparency = Saved.Terrain.WaterTransparency
        end)
    end
    for _, Entry in Saved.Effects do
        pcall(function()
            Entry.Part.Enabled = Entry.Enabled
        end)
    end
    Alert("low graphics off", 3)
end)

-- esp ------------------------------------------------------------------------------------------
-- One Highlight plus one BillboardGui per tracked object, and a single shared painter that walks
-- the list on a throttle. A RenderStepped connection per entry is what makes a full item sweep
-- cost frames, so there is exactly one.
Esp.Groups = {}
Esp.Count = 0

local function EspGroup(Name)
    local Group = Esp.Groups[Name]
    if not Group then
        Group = {}
        Esp.Groups[Name] = Group
    end
    return Group
end

Esp.Add = LPH_NO_VIRTUALIZE(function(Group, Object, Text, Color, Entity)
    if typeof(Object) ~= "Instance" or not Object:IsDescendantOf(Workspace) then
        return nil
    end

    local List = EspGroup(Group)
    if List[Object] then
        return List[Object]
    end

    local Label = Esp.Alias[Text or Object.Name] or Text or Object.Name
    local Anchor = Object:IsA("Model") and (Object.PrimaryPart or Object:FindFirstChildWhichIsA("BasePart")) or Object
    if not Anchor then
        return nil
    end

    local Entry = { Object = Object, Text = Label, Color = Color or Color3.new(1, 1, 1), Group = Group }

    -- a fully transparent entity cannot carry a Highlight, so nudge it just off invisible and
    -- record what it was so unload can put it back
    if Entity and Object:IsA("Model") and Object.PrimaryPart and Object.PrimaryPart.Transparency == 1 then
        Entry.Transparency = Object.PrimaryPart.Transparency
        Object.PrimaryPart.Transparency = 0.99
    end

    local Glow = Instance.new("Highlight")
    Glow.Adornee = Object
    Glow.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    Glow.FillColor = Entry.Color
    Glow.OutlineColor = Entry.Color
    Glow.FillTransparency = Util.Number(Flags.ESPFillTransparency, 0.75)
    Glow.OutlineTransparency = Util.Number(Flags.ESPOutlineTransparency, 0)
    Glow.Enabled = Flags.ESPHighlight ~= false
    Glow.Parent = Object
    Entry.Glow = Glow

    local Board = Skin.New("BillboardGui", {
        Adornee = Anchor,
        AlwaysOnTop = true,
        ClipsDescendants = false,
        Size = UDim2.fromOffset(1, 1),
    }, Anchor)
    Entry.Board = Board

    -- the tag sits on its own chip with a coloured spine, so a name stays readable over snow,
    -- fire and a lit campground instead of relying on a text stroke
    local Chip = Skin.New("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.fromScale(0.5, 0.5),
        Size = UDim2.fromOffset(1, 1),
        BackgroundColor3 = Skin.Ink,
        BackgroundTransparency = 0.35,
        BorderSizePixel = 0,
    }, Board)
    Skin.Round(Chip, UDim.new(0, 6))
    Entry.Chip = Chip

    local Spine = Skin.New("Frame", {
        AnchorPoint = Vector2.new(0, 0.5),
        Position = UDim2.new(0, 0, 0.5, 0),
        Size = UDim2.new(0, 2, 1, -4),
        BackgroundColor3 = Entry.Color,
        BorderSizePixel = 0,
    }, Chip)
    Skin.Round(Spine, UDim.new(1, 0))
    Entry.Spine = Spine

    local Title = Skin.Label(Chip, {
        Size = UDim2.fromScale(1, 1),
        Font = Skin.Mid,
        Text = Label,
        TextColor3 = Entry.Color,
        TextSize = Util.Number(Flags.ESPTextSize, 22),
    })
    Entry.Title = Title

    List[Object] = Entry
    Esp.Count += 1
    return Entry
end)

Esp.Remove = LPH_NO_VIRTUALIZE(function(Entry)
    if not Entry then
        return
    end
    local List = Esp.Groups[Entry.Group]
    if List then
        List[Entry.Object] = nil
    end
    Esp.Count = math.max(0, Esp.Count - 1)
    pcall(function()
        Entry.Glow:Destroy()
    end)
    pcall(function()
        Entry.Board:Destroy()
    end)
    Entry.Chip, Entry.Spine = nil, nil
    if Entry.Transparency and Entry.Object and Entry.Object.PrimaryPart then
        pcall(function()
            Entry.Object.PrimaryPart.Transparency = Entry.Transparency
        end)
    end
end)

Esp.Clear = LPH_NO_VIRTUALIZE(function(Group)
    if Group then
        for _, Entry in EspGroup(Group) do
            Esp.Remove(Entry)
        end
        Esp.Groups[Group] = {}
        return
    end
    for Name in Esp.Groups do
        Esp.Clear(Name)
    end
end)

Esp.Recolour = LPH_NO_VIRTUALIZE(function(Group, Colour)
    for _, Entry in EspGroup(Group) do
        Entry.Color = Colour
        pcall(function()
            Entry.Glow.FillColor = Colour
            Entry.Glow.OutlineColor = Colour
            Entry.Title.TextColor3 = Colour
            if Entry.Spine then
                Entry.Spine.BackgroundColor3 = Colour
            end
        end)
    end
end)

        ("%s [%d]"):format(Player.DisplayName, math.floor(Humanoid.Health)),
        Flags.PlayerEspColor or Color3.new(1, 1, 1))
end

local function EspSource(Name)
    if Name == "Items" then
        return Reg.Items
    end
    if Name == "Landmarks" then
        return Reg.Landmarks
    end
    if Name == "Children" then
        return Reg.Children
    end
    if Name == "Characters" then
        local Folder = Workspace:FindFirstChild("Characters")
        return Folder and Folder:GetChildren() or {}
    end
    return {}
end

-- One sweep per enabled group: add what is wanted and in range, drop what is gone. Runs on its
-- own slow thread so a big map never stutters the render.
local function EspSweep()
    for _, Entry in ESP_GROUPS do
        local List = EspGroup(Entry.Group)
        if Flags[Entry.Flag] ~= true then
            if next(List) then
                Esp.Clear(Entry.Group)
            end
        else
            local Wanted = Util.Picked(Entry.Picker)
            local Colour = Flags[Entry.Colour] or Entry.Default
            for _, Source in Entry.Sources do
                for _, Object in EspSource(Source) do
                    if typeof(Object) == "Instance" and Wanted[Object.Name] then
                        Esp.Add(Entry.Group, Object, Object.Name, Colour, Entry.Entity)
                    end
                end
            end
            for Object, Tracked in List do
                if not (Object and Object.Parent and Object:IsDescendantOf(Workspace)) then
                    Esp.Remove(Tracked)
                end
            end
        end
    end

    local Players_ = EspGroup("Player")
    if Flags.PlayerESP ~= true then
        if next(Players_) then
            Esp.Clear("Player")
        end
    else
        for _, Player in Players:GetPlayers() do
            PlayerEsp(Player)
        end
        for Object, Tracked in Players_ do
            if not (Object and Object.Parent) then
                Esp.Remove(Tracked)
            end
        end
    end
end

Visual.Fullbright = LPH_NO_VIRTUALIZE(function(On)

        Lighting.Brightness = 2

        Lighting.GlobalShadows = false
        Lighting.OutdoorAmbient = Color3.fromRGB(128, 128, 128)
        Lighting.Ambient = Color3.fromRGB(255, 255, 255)
    end

    if On then
        if not Fullbright.Saved then
            Fullbright.Saved = {}
            for _, Field in { "Brightness", "ClockTime", "FogEnd", "GlobalShadows", "OutdoorAmbient", "Ambient" } do
                Fullbright.Saved[Field] = Lighting[Field]
            end
        end
        Force()
        if Fullbright.Connection then
            pcall(function()
                Fullbright.Connection:Disconnect()
            end)
        end
        -- the game resets Lighting on its own day cycle, so hold the values
        Fullbright.Connection = Lighting.Changed:Connect(function()
            if Flags.FullbrightToggle ~= true then
                return
            end
            Force()
        end)
        return
    end

    if Fullbright.Connection then
        pcall(function()
            Fullbright.Connection:Disconnect()
        end)
        Fullbright.Connection = nil
    end
    if Fullbright.Saved then
        for Field, Value in Fullbright.Saved do
            pcall(function()
                Lighting[Field] = Value
            end)
        end
        Fullbright.Saved = nil
    end
end)

-- Everything here is local and reversible. Walk speed records the character's own value so
-- switching off restores what the game gave you rather than a hardcoded number.
local Self = {
    Speed = nil, Noclip = {}, NoclipThread = nil, FlyConnection = nil,
    Fov = nil, FovThread = nil, SpeedThread = nil, WalkThread = nil,
}

Self.ApplySpeed = LPH_NO_VIRTUALIZE(function()
    local _, Humanoid = Char.Parts()
    if not Humanoid then
        return
    end
    Humanoid.WalkSpeed = Util.Number(Flags.SpeedSlider, 30)
end)

Self.SpeedOn = LPH_NO_VIRTUALIZE(function(On)
    if On then
        local _, Humanoid = Char.Parts()
        if Humanoid and Self.Speed == nil then
            Self.Speed = Humanoid.WalkSpeed
        end
        Self.ApplySpeed()
        -- the game resets WalkSpeed on its own, so hold the value on a slow loop
        Self.SpeedThread = task.spawn(function()
            Util.Identity(2)
            while Alive() and Flags.SpeedToggle == true do
                Self.ApplySpeed()
                task.wait(0.5)
            end
        end)
        return
    end

    local _, Humanoid = Char.Parts()
    if Humanoid and Self.Speed ~= nil then
        Humanoid.WalkSpeed = Self.Speed
    end
    Self.Speed = nil
end)

-- TP Walk steps the character forward every frame instead of letting the humanoid walk. It reads
-- its own slider, never the Walk Speed one, so the knob always matches what it does.
Self.WalkOn = LPH_NO_VIRTUALIZE(function(On)
    if not On then
        return
    end
    local Element = Toggles.SpeedToggle
    if Element and Flags.SpeedToggle == true then
        pcall(Element.SetValue, Element, false)
    end

    Self.WalkThread = task.spawn(function()
        Util.Identity(2)
        while Alive() and Flags.TPSpeedToggle == true do
            local Delta = RunService.Heartbeat:Wait()
            local Model, Humanoid = Char.Parts()
            if Model and Humanoid and Humanoid.MoveDirection.Magnitude > 0 then
                local Rate = Util.Number(Flags.TPSpeedSlider, 5)
                pcall(function()
                    Model:TranslateBy(Humanoid.MoveDirection * Rate * Delta * 10)
                end)
            end
        end
    end)
end)

Self.NoclipOn = LPH_NO_VIRTUALIZE(function(On)
    if On then
        Self.NoclipThread = task.spawn(function()
            Util.Identity(2)
            while Alive() and Flags.Noclip == true do
                RunService.Heartbeat:Wait()
                local Model = LocalPlayer.Character
                if Model then
                    for _, Part in Model:GetDescendants() do
                        if Part:IsA("BasePart") and Part.CanCollide then
                            Part.CanCollide = false
                            Self.Noclip[Part] = true
                        end
                    end
                end
            end
        end)
        return
    end

    for Part in Self.Noclip do
        pcall(function()
            if Part and Part.Parent and Part:IsA("BasePart") then
                Part.CanCollide = true
            end
        end)
    end
    table.clear(Self.Noclip)
end)

Self.SetNoclip = LPH_NO_VIRTUALIZE(function(Want)
    local Element = Toggles.Noclip
    if not Element then
        return
    end
    if Want == nil then
        Want = Flags.Noclip ~= true
    end
    if Flags.Noclip ~= Want then
        pcall(Element.SetValue, Element, Want)
    end
end)

-- Fly anchors the head and drives it from the camera: the humanoid is left platform standing, so
-- nothing about the character's own physics has to be faked.
Self.FlyOn = LPH_NO_VIRTUALIZE(function(On)
    local Model, Humanoid = Char.Parts()

    if not On then
        if Self.FlyConnection then
            pcall(function()
                Self.FlyConnection:Disconnect()
            end)
            Self.FlyConnection = nil
        end
        if Humanoid then
            pcall(function()
                Humanoid.PlatformStand = false
            end)
        end
        local Head = Model and Model:FindFirstChild("Head")
        if Head then
            pcall(function()
                Head.Anchored = false
            end)
        end
        return
    end

    if not (Model and Humanoid) then
        Alert("you have to be alive to fly", 3)
        local Element = Toggles.CFlyToggle
        if Element then
            pcall(Element.SetValue, Element, false)
        end
        return
    end

    local Head = Model:WaitForChild("Head", 5)
    if not Head then
        return
    end
    Humanoid.PlatformStand = true
    Head.Anchored = true

    if Self.FlyConnection then
        pcall(function()
            Self.FlyConnection:Disconnect()
        end)
    end
    Self.FlyConnection = RunService.Heartbeat:Connect(function(Delta)
        local Live_, Hum = Char.Parts()
        local View = Workspace.CurrentCamera
        if not (Live_ and Hum and View and Head.Parent) then
            return
        end
        local Wanted = Hum.MoveDirection * Util.Number(Flags.FlySpeedSlider, 10) * Delta * 10
        local Facing = CFrame.new(Head.Position, Head.Position + View.CFrame.LookVector)
        Head.CFrame = Facing * CFrame.new(Wanted.X, Wanted.Y, -Wanted.Z)
    end)
end)

Self.FovOn = LPH_NO_VIRTUALIZE(function(On)
    local View = Workspace.CurrentCamera
    if On then
        if Self.Fov == nil and View then
            Self.Fov = View.FieldOfView
        end
        Self.FovThread = task.spawn(function()
            Util.Identity(2)
            while Alive() and Flags.FOVToggle == true do
                local Current = Workspace.CurrentCamera
                -- the game owns the camera while auto-run is on; leave it alone then
                local AutoRun = LocalPlayer:FindFirstChild("AutoRun")
                if Current and not (AutoRun and AutoRun.Value) then
                    Current.FieldOfView = Util.Number(Flags.FOVSlider, 60)
                end
                task.wait(0.1)
            end
        end)
        return
    end
    if View and Self.Fov ~= nil then
        View.FieldOfView = Self.Fov
    end
    Self.Fov = nil
end)

Self.Restore = LPH_NO_VIRTUALIZE(function()
    Self.SpeedOn(false)
    Self.NoclipOn(false)
    Self.FlyOn(false)
    Self.FovOn(false)
end)

-- anti void ------------------------------------------------------------------------------------
-- A collision plate tracked under the player's feet. Falling off the map lands on it and gets a
-- shove upwards, instead of ending the run. Rebuilt on respawn and destroyed on unload.
local Void = { Part = nil, Touch = nil, Thread = nil, Respawn = nil, On = false }

Void.Clean = LPH_NO_VIRTUALIZE(function()
    Void.On = false
    for _, Key in { "Touch", "Respawn" } do
        if Void[Key] then
            pcall(function()
                Void[Key]:Disconnect()
            end)
            Void[Key] = nil
        end
    end
    if Void.Thread then
        pcall(task.cancel, Void.Thread)
        Void.Thread = nil
    end
    if Void.Part then
        pcall(function()
            Void.Part:Destroy()
        end)
        Void.Part = nil
    end
end)

Void.Build = LPH_NO_VIRTUALIZE(function()
    if Void.Part then
        pcall(function()
            Void.Part:Destroy()
        end)
    end

    local Part = Instance.new("Part")
    Part.Name = "AxonAntiVoid"
    Part.Size = Vector3.new(10000, 1, 10000)
    Part.Transparency = 0.5
    Part.Material = Enum.Material.ForceField
    Part.Color = Color3.fromRGB(0, 255, 255)
    Part.Anchored = true
    Part.CanCollide = true
    Part.CanQuery = false
    Part.Parent = Workspace
    Void.Part = Part

    local Next = 0
    Void.Touch = Part.Touched:Connect(function(Hit)
        if Hit.Parent ~= LocalPlayer.Character then
            return
        end
        local _, Humanoid, Root = Char.Parts()
        if not (Humanoid and Root) or os.clock() < Next then
            return
        end
        Next = os.clock() + 0.1
        Root.AssemblyLinearVelocity = Vector3.new(
            Root.AssemblyLinearVelocity.X, 100, Root.AssemblyLinearVelocity.Z)
    end)
end)

Void.Enable = LPH_NO_VIRTUALIZE(function(On)
    Void.Clean()
    if not On then
        return
    end

    Void.On = true
    Void.Build()

    Void.Thread = task.spawn(function()
        Util.Identity(2)
        while Void.On and Alive() do
            local _, _, Root = Char.Parts()
            if Root and Void.Part then
                local Params = RaycastParams.new()
                Params.RespectCanCollide = true
                Params.FilterDescendantsInstances = { LocalPlayer.Character, Void.Part }
                local Hit = Workspace:Raycast(Root.Position, Vector3.new(0, -1000, 0), Params)
                if Hit then
                    Void.Part.Position = Hit.Position - Vector3.new(0, 15, 0)
                end
            end
            task.wait(0.1)
        end
    end)

    Void.Respawn = Library:GiveSignal(LocalPlayer.CharacterAdded:Connect(function()
        if not Void.On then
            return
        end
        task.wait(0.6)
        Void.Build()
    end))
end)

    if Flags.AntiAfk == false then
        for _, Connection in AntiAfk.Disabled do
            pcall(function()
                Connection:Enable()
            end)
        end
        table.clear(AntiAfk.Disabled)
        return
    end

    -- disable rather than disconnect the game's own idle handler, so unload can hand it back
    local Ok, List = pcall(function()
        return getconnections(LocalPlayer.Idled)
    end)
    for _, Connection in (Ok and type(List) == "table" and List or {}) do
        if not table.find(AntiAfk.Disabled, Connection) then
            if pcall(function()
                Connection:Disable()
            end) then
                table.insert(AntiAfk.Disabled, Connection)
            end
        end
    end

    AntiAfk.Connection = LocalPlayer.Idled:Connect(function()
        if Flags.AntiAfk ~= false then
            AntiAfk.Pulse()
        end
    end)
end)

-- scheduler ------------------------------------------------------------------------------------
-- Three loops. The aura loop runs on Heartbeat so a swing is never starved by a claim. The mover
-- loop owns the character and runs one job at a time. The background loop is remote only and
-- walks its list in order, stopping at the first job that actually did something, so no single
-- job can hog the rate limiter.
local Auto = { Runs = 0, Fails = 0, Index = 1, Kick = 0, Seen = 0, Next = {} }

-- Idle backoff. A job whose pass found nothing rests for its interval rather than re-scanning the
-- world every tick; a pass that did something runs again immediately. Flipping any toggle
-- advances Auto.Kick, which clears every backoff so a freshly enabled job fires at once.
Auto.DefaultIdle = 1.5
Auto.Idle = {
    eat = 1, bandage = 1, stun = 0.5, pollinate = 2, shovel = 0.5, candy = 0.5, saplings = 1,
    chests = 2, flowers = 2, coins = 2, seedboxes = 3, bees = 3, presents = 3,
    campfire = 2, scrap = 3, gears = 3, stronghold = 1,
}

Auto.Wake = LPH_NO_VIRTUALIZE(function(On)
    if On then
        Auto.Kick = os.clock()
    end
end)

Auto.Nap = LPH_NO_VIRTUALIZE(function(Seconds)
    local Start, Kick = os.clock(), Auto.Kick
    while os.clock() - Start < Seconds and Auto.Kick == Kick and Alive() do
        task.wait(0.1)
    end
end)

local function Woke()
    if Auto.Kick ~= Auto.Seen then
        Auto.Seen = Auto.Kick
        table.clear(Auto.Next)
    end
end

-- Run a button's action off the UI thread at identity 2, so a yield or an error cannot take the
-- window down with it.
Auto.Button = LPH_NO_VIRTUALIZE(function(Name, Callback)
    return function()
        task.spawn(function()
            Util.Identity(2)
            State.Job = Name
            local Ok, Error = xpcall(Callback, function(Problem)
                return debug.traceback(tostring(Problem), 2)
            end)
            State.Job = ""
            State.Runs += 1
            if not Ok then
                Auto.Fails += 1
                Note(Name .. " failed: " .. Util.Cap(tostring(Error), 90))
            end
        end)
    end
end)

Auto.Cycle = LPH_NO_VIRTUALIZE(function()
    Woke()
    local Any, Did = false, false

    for _, Job in Auto.MoverJobs do
        if Flags[Job.Flag] == true then
            Any = true
            if os.clock() >= (Auto.Next[Job.Name] or 0) then
                if not Move.Anchor then
                    Move.Save()
                end
                State.Job = Job.Name
                local Ok, Result = pcall(Job.Run)
                State.Job = ""
                Move.Drop()
                if not Ok then
                    Auto.Fails += 1
                    Note(Job.Name .. " failed: " .. Util.Cap(tostring(Result), 90))
                    Auto.Next[Job.Name] = os.clock() + math.max(Auto.Idle[Job.Name] or Auto.DefaultIdle, 2)
                elseif Result then
                    Did = true
                    Auto.Runs += 1
                    Auto.Next[Job.Name] = 0
                else
                    Auto.Next[Job.Name] = os.clock() + (Auto.Idle[Job.Name] or Auto.DefaultIdle)
                end
            end
        end
    end

    -- nothing wants the body any more: put the player back where they were standing
    if not Any and Move.Anchor then
        Move.Return()
    end
    if not Did then
        task.wait(0.1)
    end
    return Did
end)

Auto.Background = LPH_NO_VIRTUALIZE(function()
    Woke()
    local Count = #Auto.BackgroundJobs
    for Step = 1, Count do
        local Index = ((Auto.Index + Step - 2) % Count) + 1
        local Job = Auto.BackgroundJobs[Index]
        if Flags[Job.Flag] == true and os.clock() >= (Auto.Next[Job.Name] or 0) then
            Auto.Index = Index + 1
            State.Job = Job.Name
            local Ok, Result = pcall(Job.Run)
            State.Job = ""
            if not Ok then
                Auto.Fails += 1
                Note(Job.Name .. " failed: " .. Util.Cap(tostring(Result), 90))
                Auto.Next[Job.Name] = os.clock() + math.max(Auto.Idle[Job.Name] or Auto.DefaultIdle, 2)
            elseif Result then
                Auto.Runs += 1
                Auto.Next[Job.Name] = 0
                return true
            else
                Auto.Next[Job.Name] = os.clock() + (Auto.Idle[Job.Name] or Auto.DefaultIdle)
            end
        end
    end
    Auto.Index = 1
    task.wait(0.15)
    return false
end)

    end
    Stop.Hide()

    Note("everything stopped")
end)

-- Each category gets its own picker, so "Bring Food" can mean three things rather than all of
-- them. The button only ever brings what its own dropdown holds.
local function BringSection(Title, Label, Flag, Values, Default, Options_)
    local Group = ItemsTab:AddLeftGroupbox(Title)
    Group:AddDropdown(Flag, {
        Text = Label .. " Choice",
        Values = Values,
        Multi = true,
        Default = Default or Values,
        Tooltip = "Only what is ticked here is brought.",
    })
    Group:AddButton("Bring " .. Label, Auto.Button("bring " .. Label:lower(), function()
        local Picked = Util.List(Flags[Flag])
        if #Picked == 0 then
            Alert("nothing ticked in the " .. Label .. " picker", 3)
            return
        end
        Bring.Run(Label:lower(), Picked, Options_)
    end))
    return Group
end

-- The sampler reads game state on its own thread at identity 2; the painter only writes the UI,
-- and only where the text actually changed. Touching game code on the render thread is what
-- breaks CoreGui writes, so the two never mix.
local function OnFlag(Flag)
    return Flags[Flag] == true
end

local function AnyFlag(List)
    for _, Flag in List do
        if Flags[Flag] == true then
            return true
        end
    end
    return false
end

-- teardown -------------------------------------------------------------------------------------
-- Registered before any feature can fail, so a half-booted run still leaves a working way to
-- take itself down. Everything the hub changed locally is put back here.
local function Teardown()
    Auto.Kick = os.clock()
    Explore.Abort = true
    Farm.Running = false
    Hold.Running = false

    pcall(Mobile.Destroy)
    pcall(Stop.Hide)
    if Bring.Cancel then
        pcall(Bring.Cancel)
    end
    pcall(Freecam.Disable)
    pcall(Move.Stop)
    pcall(Explore.Clean)

    -- character and camera
    pcall(Self.Restore)
    pcall(function()
        Godmode.Hover(false)
    end)
    pcall(Hitbox.Restore)
    pcall(function()
        Void.Clean()
    end)
    pcall(function()
        Circle.Show(false)
    end)
    pcall(function()
        Coords.Show(false)
    end)

    -- visuals
    pcall(function()
        Visual.Fullbright(false)
    end)
    pcall(Visual.ClearTheme)
    pcall(function()
        Visual.Aesthetic(false)
    end)
    pcall(function()
        Visual.Shader(false)
    end)
    pcall(function()
        Perf.Low(false)
    end)
    pcall(RevertLighting)

    -- module edits and esp
    pcall(Patch.RestoreAll)
    pcall(function()
        Esp.Clear()
    end)
    pcall(Bar.Clear)
    pcall(Plant.Blueprint)

    -- session
    for _, Connection in AntiAfk.Disabled do
        pcall(function()
            Connection:Enable()
        end)
    end
    table.clear(AntiAfk.Disabled)
    if AntiAfk.Connection then
        pcall(function()
            AntiAfk.Connection:Disconnect()
        end)
    end
    for _, Connection in Reg.Connections do
        pcall(function()
            Connection:Disconnect()
        end)
    end
    table.clear(Reg.Connections)

    Boot.Env.NightsInTheForest_loaded = false
    Shared.NightsInTheForest_Library = nil
    -- the gui handle is left in place on purpose: the next run reads it to destroy this window
end

Library:OnUnload(Teardown)

do
    local Gui = nil
    pcall(function()
        local Roots = {}
        if type(gethui) == "function" then
            local Ok, Hidden = pcall(gethui)
            if Ok and Hidden then
                table.insert(Roots, Hidden)
            end
        end
        local Ok, Core = pcall(game.GetService, game, "CoreGui")
        if Ok and Core then
            table.insert(Roots, Core)
        end
        local Player = LocalPlayer:FindFirstChild("PlayerGui")
        if Player then
            table.insert(Roots, Player)
        end
        for _, Root in Roots do
            local Kids_ = Root:GetChildren()
            for Index = #Kids_, 1, -1 do
                local Node = Kids_[Index]
                if Node:IsA("ScreenGui") and Node:GetAttribute("AxonForest") ~= true
                    and Node.Name ~= "AxonLoader" then
                    Gui = Node
                    break
                end
            end
            if Gui then
                break
            end
        end
    end)
    if Gui then
        pcall(function()
            Gui:SetAttribute("AxonForest", true)
        end)
        Boot.Env.AxonForestGui = Gui
    end
    Boot.Env.AxonForestUnload = function()
        pcall(Teardown)
        pcall(Library.Unload, Library)
        if Gui then
            pcall(function()
                Gui:Destroy()
            end)
        end
    end
end

-- fill the pickers that need the game's own data, once the registry has something in it

    local Pairs = {
    }
    for _, Entry in Pairs do
        local Element = Options[Entry[1]]
        if Element then
            local Ok, Labels = pcall(Entry[2])
            if Ok and type(Labels) == "table" and #Labels > 0 then
                pcall(Element.SetValues, Element, Labels)

        pcall(ChestElement.SetValues, ChestElement, Chests)
    end

-- loops ----------------------------------------------------------------------------------------
task.spawn(function()
    Util.Identity(2)

    -- sampler: reads the game, never writes the UI
    task.spawn(function()
        Util.Identity(2)
        while Alive() do
            Util.Guard("sampler", Sample)
            task.wait(0.4)
        end
    end)

    -- aura loop: on its own thread so a swing is never starved by a claim, and never the reverse
    task.spawn(function()
        Util.Identity(2)
        while Alive() do
            local Wanted = AnyFlag({ "KillAuraToggle", "IceAuraToggle", "OreAuraToggle", "WebBallAuraToggle" })
            if not Wanted then
                task.wait(0.5)
                continue
            end
            Util.Guard("aura", function()
                if not Char.Alive() then
                    return
                end
                -- hand check on: wait until the player is really holding a weapon. Off: let the
                -- hub equip the best one in the inventory and swing with that.
                if Flags.RequireTool == true then
                    if not Tool.InHand("MeleeWeapon") then
                        return
                    end
                elseif not Tool.Weapon() then
                    return
                end
                local Ready, Wait = Combat.Ready(Tool.Weapon())
                if not Ready then
                    task.wait(Wait / 4)
                    return
                end
                local Targets = Combat.Scan(Flags.KillAuraRange)
                if #Targets == 0 then
                    return
                end
                Combat.Swung = os.clock()
                for _, Target in Targets do
                    if not Alive() then
                        break
                    end
                    Combat.Hit(Target, false, Targets)
                end
            end)
            RunService.Heartbeat:Wait()
        end
    end)

    -- tree loop: same reasoning, and it uses its own range and its own target scan
    task.spawn(function()
        Util.Identity(2)
        while Alive() do
            if Flags.AutoChopTrees ~= true then
                task.wait(0.5)
                continue
            end
            Util.Guard("tree farm", function()
                if not Char.Alive() then
                    return
                end
                if Flags.RequireTool == true then
                    if not Tool.InHand("MeleeWeapon") then
                        return
                    end
                elseif not Tool.Weapon() then
                    return
                end
                local Ready, Wait = Combat.Ready(Tool.Weapon())
                if not Ready then
                    task.wait(Wait / 4)
                    return
                end
                local Range = Util.Number(Flags.AutoChopTreesRange, 200)
                if Flags.AutoTPTargets ~= true then
                    Range = math.min(Range, 250)
                end
                local Trees = Combat.ScanTrees(Range)
                if #Trees == 0 then
                    return
                end
                Combat.Swung = os.clock()
                for _, Tree in Trees do
                    if not Alive() then
                        break
                    end
                    Combat.Hit(Tree, true, Trees)
                end
            end)
            RunService.Heartbeat:Wait()
        end
    end)

    -- esp sweep: slow on purpose, the painter does the per-frame work
    task.spawn(function()
        Util.Identity(2)
        while Alive() do
            Util.Guard("esp", EspSweep)
            task.wait(1)
        end
    end)

    -- esp appearance: one shared pass instead of a connection per entry
    task.spawn(function()
        Util.Identity(2)
        while Alive() do
            local Fill = Util.Number(Flags.ESPFillTransparency, 0.75)
            local Outline = Util.Number(Flags.ESPOutlineTransparency, 0)
            local Size = Util.Number(Flags.ESPTextSize, 22)
            local Highlight = Flags.ESPHighlight ~= false
            local Distance = Flags.ESPDistance ~= false
            for _, Group in Esp.Groups do
                for Object, Entry in Group do
                    if not (Object and Object.Parent and Object:IsDescendantOf(Workspace)) then
                        Esp.Remove(Entry)
                    else
                        pcall(function()
                            Entry.Glow.Enabled = Highlight
                            Entry.Glow.FillTransparency = Fill
                            Entry.Glow.OutlineTransparency = Outline
                            Entry.Title.TextSize = Size
                            Entry.Title.Text = Distance
                                and ("%s\n[%d]"):format(Entry.Text, math.floor(Char.Distance(Object)))
                                or Entry.Text
                        end)
                    end
                end
            end
            task.wait(0.2)
        end
    end)

    -- background loop: remote only
    task.spawn(function()
        Util.Identity(2)
        while Alive() do
            if Auto.Wants(Auto.BackgroundJobs) then
                Auto.Background()
                task.wait(0.05)
            else
                Auto.Nap(0.5)
            end
        end
    end)

    -- picker refresh: the map keeps changing as you explore
    task.spawn(function()
        Util.Identity(2)
        task.wait(8)
        while Alive() do
            Util.Guard("choices", FillChoices)
            task.wait(30)
        end
    end)

    -- mover loop: owns the character, one job at a time
    while Alive() do
        if Auto.Wants(Auto.MoverJobs) then
            if not State.Busy then
                State.Busy = true
                Util.Guard("automation", Auto.Cycle)
                State.Busy = false
            end
            task.wait()
        else
            Auto.Nap(0.5)
        end
    end
end)
