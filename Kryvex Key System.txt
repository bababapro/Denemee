local function ValidateKey(inputKey, validKeys)
    for _, validKey in pairs(validKeys) do
        if inputKey == validKey then
            return true
        end
    end
    return false
end

local function Create(className, properties, parent)
    local instance = Instance.new(className)
    for property, value in pairs(properties) do
        instance[property] = value
    end
    if parent then
        instance.Parent = parent
    end
    return instance
end

        TextColor3 = Color3.fromRGB(200, 100, 100),
    }, KeyContainer)

    local JoinDiscordBtn = CreateParlayanButon("💬 JOIN DISCORD 💬", WhiteColor, Color3.fromRGB(120, 20, 20), 290, 45)
    local GetKeyBtn = CreateParlayanButon("🔑 GET KEY (Discord) 🔑", WhiteColor, Color3.fromRGB(120, 20, 20), 345, 45)

    local StatusLabel = Create("TextLabel", {
        Text = "",
        Size = UDim2.new(1, -20, 0, 30),
        Position = UDim2.new(0, 10, 0, 405),
        BackgroundTransparency = 1,
        Font = Enum.Font.GothamBold,
        TextSize = 13,
        TextColor3 = Color3.fromRGB(255, 100, 100),
        TextXAlignment = Enum.TextXAlignment.Center,
        TextWrapped = true,
    }, KeyContainer)

    -- === KÜÇÜLTÜLMÜŞ BUTON ===
    local MinimizedBtn = Create("TextButton", {
        Text = "KRYVEX",
        Font = Enum.Font.GothamBlack,
        TextSize = 14,
        TextColor3 = WhiteColor,
        BackgroundColor3 = RedColor,
        Size = UDim2.new(0, 90, 0, 90),
        Position = UDim2.new(0, 50, 0, 50),
        BorderSizePixel = 0,
        Visible = false,
        Active = true,
    }, BlurBackground)
    Create("UICorner", {CornerRadius = UDim.new(1, 0)}, MinimizedBtn)
    local MinBtnStroke = Create("UIStroke", {Color = SoftRed, Thickness = 2}, MinimizedBtn)
    
    -- Küçük Buton Sürükleme
    local mDragging, mDragInput, mDragStart, mStartPos
    MinimizedBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            mDragging = true
            mDragStart = input.Position
            mStartPos = MinimizedBtn.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then mDragging = false end
            end)
        end
    end)
    MinimizedBtn.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then mDragInput = input end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == mDragInput and mDragging then
            local delta = input.Position - mDragStart
            MinimizedBtn.Position = UDim2.new(mStartPos.X.Scale, mStartPos.X.Offset + delta.X, mStartPos.Y.Scale, mStartPos.Y.Offset + delta.Y)
        end
    end)

    -- === FONKSİYONLAR ===
    local function showStatus(text, color)
        StatusLabel.Text = text
        StatusLabel.TextColor3 = color or Color3.fromRGB(255, 100, 100)
        StatusLabel.TextTransparency = 0
        wait(2.5)
        TweenService:Create(StatusLabel, TweenInfo.new(0.5), {TextTransparency = 1}):Play()
    end

    JoinDiscordBtn.MouseButton1Click:Connect(function()
        if setclipboard then
            setclipboard(DiscordLink)
            showStatus("✅ Discord link copied!", Color3.fromRGB(100, 255, 100))
        end
    end)

    GetKeyBtn.MouseButton1Click:Connect(function()
        if setclipboard then
            setclipboard(DiscordLink)
            showStatus("✅ Go get your key from Discord!", Color3.fromRGB(100, 255, 100))
        end
    end)

    CloseButton.MouseButton1Click:Connect(function()
        TweenService:Create(KeyContainer, TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.In), {Size = UDim2.new(0,0,0,0)}):Play()
        wait(0.3)
        KeySystemGui:Destroy()
    end)

    MinimizeBtn.MouseButton1Click:Connect(function()
        KeyContainer.Visible = false
        MinimizedBtn.Visible = true
        MinimizedBtn.Size = UDim2.new(0, 0, 0, 0)
        TweenService:Create(MinimizedBtn, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, 90, 0, 90)
        }):Play()
    end)

    MinimizedBtn.MouseButton1Click:Connect(function()
        TweenService:Create(MinimizedBtn, TweenInfo.new(0.3, Enum.EasingStyle.Quad), {Size = UDim2.new(0, 0, 0, 0)}):Play()
        wait(0.3)
        MinimizedBtn.Visible = false
        KeyContainer.Visible = true
        KeyContainer.Size = UDim2.new(0, 0, 0, 0)
        TweenService:Create(KeyContainer, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = KContainerSize
        }):Play()
    end)

    VerifyBtn.MouseButton1Click:Connect(function()
        if isVerifying then return end
        
        local inputKey = KeyInput.Text
        if inputKey == "" then
            showStatus("⚠️ Please enter a key!", Color3.fromRGB(255, 200, 0))
            local origInputPos = InputFrame.Position
            for i = 1, 6 do
                InputFrame.Position = origInputPos + UDim2.new(0, (i % 2 == 0 and 8 or -8), 0, 0)
                wait(0.04)
            end
            InputFrame.Position = origInputPos
            return
        end

        isVerifying = true
        VerifyBtn.Text = "🔄 VERIFYING..."
        
        wait(0.6)

        if ValidateKey(inputKey, Keys) then
            showStatus("✅ Key Valid! Loading Hub...", Color3.fromRGB(100, 255, 100))
            VerifyBtn.BackgroundColor3 = Color3.fromRGB(0, 200, 0)
            VerifyBtn.Text = "✅ SUCCESS"
            
            local flash = Create("Frame", {Size = UDim2.new(1,0,1,0), BackgroundColor3 = WhiteColor, BackgroundTransparency = 0.8, ZIndex = 99}, BlurBackground)
            TweenService:Create(flash, TweenInfo.new(0.5), {BackgroundTransparency = 1}):Play()
            
            wait(1)
            KeySystemGui:Destroy()
            MainScriptCallback(true)
        else
            showStatus("❌ Invalid Key! Try again.", Color3.fromRGB(255, 50, 50))
            VerifyBtn.Text = "🚀 VERIFY KEY 🚀"
            
            local origPos = KeyContainer.Position
            for i = 1, 8 do
                KeyContainer.Position = origPos + UDim2.new(0, (i % 2 == 0 and 12 or -12), 0, 0)
                wait(0.04)
            end
            KeyContainer.Position = origPos
        end
        
        isVerifying = false
    end)

    KeyContainer.Size = UDim2.new(0, 0, 0, 0)
    TweenService:Create(KeyContainer, TweenInfo.new(0.5, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        Size = KContainerSize
    }):Play()
end
