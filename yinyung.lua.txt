local fullBrightState = {
    Enabled = false,
    Connection = nil,
    Original = nil,
}

local function captureFullBrightState()
    if fullBrightState.Original then return end
    local ok, snapshot = pcall(function()
        return {
            Brightness = Lighting.Brightness,
            Ambient = Lighting.Ambient,
            OutdoorAmbient = Lighting.OutdoorAmbient,
            ColorShift_Bottom = Lighting.ColorShift_Bottom,
            ColorShift_Top = Lighting.ColorShift_Top,
            GlobalShadows = Lighting.GlobalShadows,
            ClockTime = Lighting.ClockTime,
            ExposureCompensation = Lighting.ExposureCompensation,
            FogColor = Lighting.FogColor,
            FogEnd = Lighting.FogEnd,
            FogStart = Lighting.FogStart,
            EnvironmentDiffuseScale = Lighting.EnvironmentDiffuseScale,
            EnvironmentSpecularScale = Lighting.EnvironmentSpecularScale,
        }
    end)
    if ok and snapshot then fullBrightState.Original = snapshot end
end

local function applyFullBright()
    pcall(function()
        Lighting.GlobalShadows = false
        Lighting.Brightness = 1
        Lighting.Ambient = Color3.new(1, 1, 1)
        Lighting.OutdoorAmbient = Color3.new(1, 1, 1)
        Lighting.ColorShift_Bottom = Color3.new(1, 1, 1)
        Lighting.ColorShift_Top = Color3.new(1, 1, 1)
        Lighting.ClockTime = 12
        Lighting.ExposureCompensation = 0
        Lighting.FogColor = Color3.new(1, 1, 1)
        Lighting.FogEnd = 1e10
        Lighting.FogStart = 0
        Lighting.EnvironmentDiffuseScale = 1
        Lighting.EnvironmentSpecularScale = 1
    end)
end

local function restoreFullBrightState()
    if not fullBrightState.Original then return end
    pcall(function()
        Lighting.Brightness = fullBrightState.Original.Brightness
        Lighting.Ambient = fullBrightState.Original.Ambient
        Lighting.OutdoorAmbient = fullBrightState.Original.OutdoorAmbient
        Lighting.ColorShift_Bottom = fullBrightState.Original.ColorShift_Bottom
        Lighting.ColorShift_Top = fullBrightState.Original.ColorShift_Top
        Lighting.GlobalShadows = fullBrightState.Original.GlobalShadows
        Lighting.ClockTime = fullBrightState.Original.ClockTime
        Lighting.ExposureCompensation = fullBrightState.Original.ExposureCompensation
        Lighting.FogColor = fullBrightState.Original.FogColor
        Lighting.FogEnd = fullBrightState.Original.FogEnd
        Lighting.FogStart = fullBrightState.Original.FogStart
        Lighting.EnvironmentDiffuseScale = fullBrightState.Original.EnvironmentDiffuseScale
        Lighting.EnvironmentSpecularScale = fullBrightState.Original.EnvironmentSpecularScale
    end)
end

local function setFullBrightEnabled(state)
    local enabled = state and true or false
    fullBrightState.Enabled = enabled
    Config.EnableFullBright = enabled
    if enabled then
        captureFullBrightState()
        applyFullBright()
        if not fullBrightState.Connection then
            fullBrightState.Connection = RunService.Heartbeat:Connect(function()
                if not fullBrightState.Enabled then return end
                applyFullBright()
            end)
        end
    else
        if fullBrightState.Connection then
            fullBrightState.Connection:Disconnect()
            fullBrightState.Connection = nil
        end
        restoreFullBrightState()
    end
end

local SettingsFile = "KRYVEX_Settings.json"

local function SaveSettings()
    pcall(function()
        local data = {}
        for k, v in pairs(Config) do
            data[k] = v
        end
        data.FrontalJumpSpeed = getgenv().FrontalJumpSpeed
        data.BackJumpSpeed = getgenv().BackJumpSpeed
        writefile(SettingsFile, HttpService:JSONEncode(data))
    end)
end

local function LoadSettings()
    local ok, result = pcall(function()
        if isfile and isfile(SettingsFile) then
            local content = readfile(SettingsFile)
            if content and content ~= "" then
                return HttpService:JSONDecode(content)
            end
        end
        return nil
    end)
    if ok and result then
        for k, v in pairs(result) do
            if Config[k] ~= nil then
                Config[k] = v
            end
        end
        if result.FrontalJumpSpeed then getgenv().FrontalJumpSpeed = result.FrontalJumpSpeed end
        if result.BackJumpSpeed then getgenv().BackJumpSpeed = result.BackJumpSpeed end
    end
end

        removeESP()
        return
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local hrp = player.Character:FindFirstChild("HumanoidRootPart")
            if hrp and (hrp.Position - camera.CFrame.Position).Magnitude <= Config.KryvexESPDistance then
                createESP(player)
            end
        end
    end
end

--// Kryvex Aimbot (Basit)
local aimbotConnection = nil
local function getClosestTarget()
    local closest, closestDist = nil, Config.KryvexAimFOV
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local part = player.Character:FindFirstChild(Config.KryvexAimPart) or player.Character:FindFirstChild("HumanoidRootPart")
            if part then
                local screenPos, onScreen = camera:WorldToViewportPoint(part.Position)
                if onScreen then
                    local mousePos = UserInputService:GetMouseLocation()
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                    if dist < closestDist then
                        closest, closestDist = player, dist
                    end
                end
            end
        end
    end
    return closest
end

local function setAimbotEnabled(state)
    Config.EnableKryvexAimbot = state
    if aimbotConnection then aimbotConnection:Disconnect() aimbotConnection = nil end
    if not state then return end
    aimbotConnection = RunService.RenderStepped:Connect(function()
        if not Config.EnableKryvexAimbot then return end
        if not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then return end
        local target = getClosestTarget()
        if target and target.Character then
            local part = target.Character:FindFirstChild(Config.KryvexAimPart) or target.Character:FindFirstChild("HumanoidRootPart")
            if part then
                camera.CFrame = camera.CFrame:Lerp(CFrame.new(camera.CFrame.Position, part.Position), 1 / Config.KryvexAimSmooth)
            end
        end
    end)
end

--// Kryvex Hitbox Expander
local hitboxOriginals = {}
local function setHitboxExpander(state)
    Config.EnableKryvexHitbox = state
    if not state then
        for model, parts in pairs(hitboxOriginals) do
            for part, size in pairs(parts) do
                if part and part.Parent then part.Size = size end
            end
        end
        hitboxOriginals = {}
        return
    end
end

local function updateHitbox()
    if not Config.EnableKryvexHitbox then return end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local char = player.Character
            hitboxOriginals[char] = hitboxOriginals[char] or {}
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then
                    if not hitboxOriginals[char][part] then
                        hitboxOriginals[char][part] = part.Size
                    end
                    local size = hitboxOriginals[char][part]
                    part.Size = Vector3.new(size.X * Config.KryvexHitboxSize, size.Y * Config.KryvexHitboxSize, size.Z * Config.KryvexHitboxSize)
                end
            end
        end
    end
end

local function applyTeleportWalk()
    if not Config.EnableTeleportWalk and not Config.EnableTrimpTeleport then return end
    local char = LocalPlayer.Character
    if not char or not char.PrimaryPart then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end
    if hum.MoveDirection.Magnitude > 0 then
        pcall(function()
            char:TranslateBy(hum.MoveDirection * Config.TeleportMovementSpeed * 0.1)
        end)
    end
end

local function applyEnhancedJump()
    if not Config.EnableEnhancedJump then return end
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("Humanoid") then return end
    local hum = char.Humanoid
    if hum.Health <= 0 then return end
    if hum:GetState() ~= Enum.HumanoidStateType.Jumping then
        pcall(function()
            hum.JumpPower = Config.JumpHeight
            hum.UseJumpPower = true
        end)
    end
end

local function applyAutoJump()
    if not Config.AutoJump then return end
    local now = tick()
    if now < autoJumpNextTime then return end
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end
    if hum:GetState() == Enum.HumanoidStateType.Running then
        pcall(function()
            hum:ChangeState(Enum.HumanoidStateType.Jumping)
            local jumpsPerSecond = math.clamp(tonumber(Config.AutoJumpSpeed) or 1.5, 0.5, 8)
            autoJumpNextTime = now + (1 / jumpsPerSecond)
        end)
    else
        autoJumpNextTime = math.min(autoJumpNextTime, now)
    end
end

local function applyGravityModification()
    if not Config.EnableGravityMod then return end
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("Humanoid") or not char:FindFirstChild("HumanoidRootPart") then return end
    local hum = char.Humanoid
    local rootPart = char.HumanoidRootPart
    if hum.Health <= 0 then return end
    pcall(function()
        local velocity = rootPart.AssemblyLinearVelocity
        if velocity.Y < 0 then
            rootPart.AssemblyLinearVelocity = Vector3.new(
                velocity.X,
                velocity.Y * Config.GravityScale,
                velocity.Z
            )
        end
    end)
end

local function applyVoidEscape()
    if not Config.EnableVoidEscape then return end
    local char = LocalPlayer.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    if root.Position.Y > -350 then return end
    pcall(function()
        root.CFrame = CFrame.new(0, 9e9, 0)
    end)
end

local function applyAutoTrimp()
    if not Config.EnableAutoTrimp and not Config.EnableTrimpTeleport then return end
    local char = LocalPlayer.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local humanoid = char:FindFirstChild("Humanoid")
    if not root or not humanoid then return end
    if humanoid.Health <= 0 then return end
    local isAir = humanoid.FloorMaterial == Enum.Material.Air
    pcall(function()
        if isAir then
            local vel = root.AssemblyLinearVelocity
            if vel.Y < 0 then
                root.AssemblyLinearVelocity = Vector3.new(vel.X, vel.Y * Config.AutoTrimpFallSpeed, vel.Z)
            end
        end
    end)
end

--// AVATAR TAB
local AvatarTab = Window:Tab({
    Title = "Avatar",
    Icon = "user",
    Desc = "Avatar değiştir",
})

local AvatarSection = AvatarTab:Section({
    Title = "Avatar Change",
    Icon = "user",
})

local avatarInput = AvatarSection:Input({
    Title = "Username",
    Desc = "Avatarını değiştirmek istediğin oyuncunun adı",
    Icon = "user",
    Placeholder = "Username",
    Value = "",
    Callback = function(value) end,
})

AvatarSection:Button({
    Title = "Apply Avatar",
    Desc = "Avatarı uygula",
    Icon = "check",
    Callback = function()
        local username = avatarInput.Value
        if username == "" then
            Window:Notify({
                Title = "Hata",
                Content = "Kullanıcı adı boş olamaz",
                Icon = "x",
                Duration = 3,
            })
            return
        end
        local rigsFolder = workspace:FindFirstChild("Rigs")
        local visualRig = rigsFolder and rigsFolder:FindFirstChild(LocalPlayer.Name)
        local humanoid = visualRig and visualRig:FindFirstChildOfClass("Humanoid")
        if not humanoid then
            Window:Notify({
                Title = "Hata",
                Content = "Visual rig bulunamadı",
                Icon = "x",
                Duration = 3,
            })
            return
        end
        local okUser, userId = pcall(function()
            return Players:GetUserIdFromNameAsync(username)
        end)
        if not okUser then
            Window:Notify({
                Title = "Hata",
                Content = "Kullanıcı bulunamadı",
                Icon = "x",
                Duration = 3,
            })
            return
        end
        local okDesc, desc = pcall(function()
            return Players:GetHumanoidDescriptionFromUserId(userId)
        end)
        if not okDesc then
            Window:Notify({
                Title = "Hata",
                Content = "Avatar yüklenemedi",
                Icon = "x",
                Duration = 3,
            })
            return
        end
        local okApply = pcall(function()
            humanoid:ApplyDescriptionReset(desc)
        end)
        if okApply then
            Window:Notify({
                Title = "Başarılı",
                Content = username .. " avatarı uygulandı",
                Icon = "check",
                Duration = 3,
            })
        else
            Window:Notify({
                Title = "Hata",
                Content = "Avatar uygulanamadı",
                Icon = "x",
                Duration = 3,
            })
        end
    end,
})
