local function manageTool(toolName)
    task.spawn(function()
        while _G[toolName.."On"] do
            local character = LocalPlayer.Character
            if character then
                local tool = LocalPlayer.Backpack:FindFirstChild(toolName) or character:FindFirstChild(toolName)
                if tool then
                    if tool.Parent ~= character then tool.Parent = character end
                    local event = LocalPlayer:FindFirstChild("muscleEvent")
                    if event then
                        for i = 1, repsPerTick do
                            if not _G[toolName.."On"] then break end
                            event:FireServer("rep")
                        end
                    else
                        tool:Activate()
                    end
                end
            end
            task.wait(0.01)
        end
    end)
end

local parts = {}

local function manageParts(state)
    if state and #parts == 0 then
        local partSize = 2048
        local totalDistance = 50000
        local startPosition = Vector3.new(-2, -9.5, -2)
        local numberOfParts = math.ceil(totalDistance / partSize)
        for x = 0, numberOfParts - 1 do
            for z = 0, numberOfParts - 1 do
                local positions = {
                    startPosition + Vector3.new(x * partSize, 0, z * partSize),
                    startPosition + Vector3.new(-x * partSize, 0, z * partSize),
                    startPosition + Vector3.new(-x * partSize, 0, -z * partSize),
                    startPosition + Vector3.new(x * partSize, 0, -z * partSize)
                }
                for _, pos in ipairs(positions) do
                    local p = Instance.new("Part")
                    p.Size = Vector3.new(partSize, 1, partSize)
                    p.Position = pos
                    p.Anchored = true
                    p.Transparency = 1
                    p.CanCollide = true
                    p.Parent = workspace
                    table.insert(parts, p)
                end
            end
        end
    elseif not state then
        for _, p in ipairs(parts) do p.CanCollide = false end
    else
        for _, p in ipairs(parts) do p.CanCollide = true end
    end
end

local function createGymFarm(folder, toggleName, cframeValue, eventName)

                    local root = character and character:FindFirstChild("HumanoidRootPart")
                    if root then
                        if (root.Position - cframeValue.Position).Magnitude > 5 then
                            root.CFrame = cframeValue

                            pressEKey()
                        end
                        local event = LocalPlayer:FindFirstChild("muscleEvent")
                        if event then event:FireServer(eventName) end
                    end
                    task.wait(0.05)
                end
            end)
        end
    end)
end

local function LockPosition(CFramePosition)
    _G.lockRunning = true
    _G.LockedCFrame = CFramePosition
    
    if _G.lockConnection then
        _G.lockConnection:Disconnect()
    end
    
    _G.lockConnection = game:GetService("RunService").Heartbeat:Connect(function()
        local Character = game.Players.LocalPlayer.Character
        if Character and _G.lockRunning then
            local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")
            local Humanoid = Character:FindFirstChild("Humanoid")
            if HumanoidRootPart and Humanoid then
                HumanoidRootPart.CFrame = _G.LockedCFrame
                HumanoidRootPart.Velocity = Vector3.zero
                HumanoidRootPart.RotVelocity = Vector3.zero
                Humanoid.PlatformStand = true
            end
        end
    end)
end

local function UnlockPosition()
    _G.lockRunning = false
    if _G.lockConnection then

local function formatNumber(num)
    if num >= 1e18 then return string.format("%.2fQi", num/1e18) end
    if num >= 1e15 then return string.format("%.2fQa", num/1e15) end
    if num >= 1e12 then return string.format("%.2fT", num/1e12) end
    if num >= 1e9 then return string.format("%.2fB", num/1e9) end
    if num >= 1e6 then return string.format("%.2fM", num/1e6) end
    if num >= 1e3 then return string.format("%.2fK", num/1e3) end
    return string.format("%.0f", num)
end

local function findActiveBoss()
    local model, stats, health, maxHealth, hitbox
    pcall(function()
        for _, candidate in ipairs(CollectionService:GetTagged("BossEventBoss")) do
            if candidate:IsA("Model") and candidate:IsDescendantOf(workspace) then model = candidate; break end
        end
        if not model then
            local events = workspace:FindFirstChild("Events")
            local bossFolder = events and (events:FindFirstChild("BossEvent") or events:FindFirstChild("BossArena"))
            if bossFolder then
                for _, candidate in ipairs(bossFolder:GetDescendants()) do
                    if candidate:IsA("Model") then
                        local hasHitbox = candidate:FindFirstChild("BossDamageHitbox", true)
                        local hasTag = CollectionService:HasTag(candidate, "BossEventBoss")
                        if hasHitbox or hasTag then model = candidate; break end
                    end
                end
            end
        end
        if not model then return end
        stats = model:FindFirstChild("stats")
        health = stats and stats:FindFirstChild("Health")
        maxHealth = stats and stats:FindFirstChild("MaxHealth")
        hitbox = model:FindFirstChild("BossDamageHitbox", true)
        if not hitbox or not hitbox:IsA("BasePart") then model = nil; return end
    end)
    if not model then return nil end
    return model, health, maxHealth, hitbox
end

local function getBossHeadPosition(model, hitbox)
    local head = model:FindFirstChild("Head", true) or model:FindFirstChild("head", true)
    if head and head:IsA("BasePart") then return head.Position end
    if head and head:IsA("Bone") then return head.WorldPosition end
    return hitbox.Position + Vector3.new(0, hitbox.Size.Y * 0.35, 0)
end

local function evadeAttack(model, hitbox, root, humanoid)
    local sequence = model:GetAttribute("BossAttackSequence")
    local attackKind = model:GetAttribute("BossAttackKind")
    local attackCFrame = model:GetAttribute("BossAttackCFrame")
    local impactTime = model:GetAttribute("BossAttackImpactTime")
    if typeof(sequence) == "number" and sequence ~= Boss.Data.AttackSequence then
        Boss.Data.AttackSequence = sequence
        if typeof(attackKind) == "string" and typeof(attackCFrame) == "CFrame" and typeof(impactTime) == "number" then
            Boss.Data.EvadeUntil = os.clock() + 0.6
        end
    end
    if typeof(attackCFrame) ~= "CFrame" or os.clock() > Boss.Data.EvadeUntil then return end
    attackKind = string.lower(tostring(attackKind or ""))
    if string.find(attackKind, "punch", 1, true) then
        local center = getBossHeadPosition(model, hitbox)
        local direction = Vector3.new(attackCFrame.LookVector.X, 0, attackCFrame.LookVector.Z)
        if direction.Magnitude < 1 then direction = Vector3.new(hitbox.CFrame.LookVector.X, 0, hitbox.CFrame.LookVector.Z) end
        if direction.Magnitude < 1 then direction = Vector3.new(0, 0, 1) end
        local safePosition = center - direction.Unit * Boss.Data.PunchBehindDistance
        root.CFrame = CFrame.lookAt(safePosition, center)
        root.AssemblyLinearVelocity = Vector3.zero
        Boss.StatusLabel.Text = "Status: Dodging Punch"
        return "Punch"
    end
    if not string.find(attackKind, "stomp", 1, true) then return end
    if humanoid.Health / math.max(humanoid.MaxHealth, 1) <= Boss.Data.StompHealthThreshold then
        Boss.Data.DodgeStomps = true
    end
    if not Boss.Data.DodgeStomps then return end
    local direction = Vector3.new(root.Position.X - attackCFrame.Position.X, 0, root.Position.Z - attackCFrame.Position.Z)
    if direction.Magnitude < 1 then direction = Vector3.new(-attackCFrame.LookVector.X, 0, -attackCFrame.LookVector.Z) end
    if direction.Magnitude < 1 then direction = Vector3.new(1, 0, 0) end
    direction = direction.Unit
    local safePosition = attackCFrame.Position + direction * Boss.Data.EvadeDistance + Vector3.new(0, 18, 0)
    root.CFrame = CFrame.lookAt(safePosition, attackCFrame.Position)
    root.AssemblyLinearVelocity = Vector3.zero
    Boss.StatusLabel.Text = "Status: Dodging Stomp"
    return "Stomp"
end

local function attackBoss(model, health, maxHealth, hitbox)
    local character = LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid or humanoid.Health <= 0 then return end
    local evadeMode = evadeAttack(model, hitbox, root, humanoid)
    if evadeMode == "Stomp" then return end
    local punch = character:FindFirstChild("Punch") or LocalPlayer.Backpack:FindFirstChild("Punch")
    if punch and punch.Parent ~= character then pcall(function() humanoid:EquipTool(punch) end) end
    local leftHand = character:FindFirstChild("LeftHand")
    local rightHand = character:FindFirstChild("RightHand")
    local center = getBossHeadPosition(model, hitbox)
    if evadeMode ~= "Punch" then root.CFrame = CFrame.lookAt(center, center + hitbox.CFrame.LookVector) end
    root.AssemblyLinearVelocity = Vector3.zero
    pcall(function()
        if leftHand then firetouchinterest(hitbox, leftHand, 0); firetouchinterest(hitbox, leftHand, 1) end
        if rightHand then firetouchinterest(hitbox, rightHand, 0); firetouchinterest(hitbox, rightHand, 1) end
    end)
    if os.clock() - Boss.Data.LastPunch >= Boss.Data.PunchInterval then
        Boss.Data.LastPunch = os.clock()
        pcall(function()
            LocalPlayer.muscleEvent:FireServer("punch", "leftHand")
            LocalPlayer.muscleEvent:FireServer("punch", "rightHand")
            if punch then punch:Activate() end
        end)
    end
    Boss.BossLabel.Text = "Boss: " .. (workspace:GetAttribute("BossDisplayName") or model.Name)
    Boss.HealthLabel.Text = "Health: " .. formatNumber(health and health.Value or 0) .. " / " .. formatNumber(maxHealth and maxHealth.Value or 0)
end

local function collectChest(generation)
    Boss.StatusLabel.Text = "Status: Waiting for Chest"
    local startedAt = os.clock()
    local chest, prompt, part
    repeat
        pcall(function()
            for _, candidate in ipairs(CollectionService:GetTagged("BossEventChest")) do
                if candidate:IsDescendantOf(workspace) then chest = candidate; break end
            end
            if not chest then chest = workspace:FindFirstChild("BossChest", true) end
            if chest then
                prompt = chest:FindFirstChild("bossChestPrompt", true)
                part = prompt and prompt.Parent or (chest:IsA("Model") and (chest.PrimaryPart or chest:FindFirstChildWhichIsA("BasePart", true))) or chest
            end
        end)
        if chest and chest:GetAttribute("BossChestEmerging") ~= true then break end
        task.wait(0.1)
    until not Boss.Data.Running or Boss.Data.Generation ~= generation or os.clock() - startedAt >= 12
    if not chest or not part then return end
    Boss.StatusLabel.Text = "Status: Collecting Chest"
    startedAt = os.clock()
    repeat
        local character = LocalPlayer.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if root and chest.Parent and part.Parent then
            pcall(function()
                root.CFrame = CFrame.lookAt(part.Position + Vector3.new(0, 3, 4), part.Position)
                root.AssemblyLinearVelocity = Vector3.zero
            end)
            task.wait(0.15)
            if prompt and prompt.Parent then pcall(function() fireproximityprompt(prompt) end) end
        end
        task.wait(0.25)
    until not Boss.Data.Running or Boss.Data.Generation ~= generation or LocalPlayer:GetAttribute("BossChestPending") == false or not chest.Parent or os.clock() - startedAt >= 12
end

local function FormatNumberWithCommas(number)
    local formatted = tostring(number):reverse():gsub("(%d%d%d)", "%1,"):reverse()
    return formatted:gsub("^,", "")
end

local function FormatAbbreviated(number)
    local abbreviations = {"", "K", "M", "B", "T", "Qa", "Qi"}
    local abbreviationIndex = 1
    while number >= 1000 do number = number / 1000; abbreviationIndex = abbreviationIndex + 1 end
    return string.format("%.2f", number) .. abbreviations[abbreviationIndex]
end

local function FormatDisplay(value)
    return "[ " .. FormatNumberWithCommas(value) .. " | " .. FormatAbbreviated(value) .. " ]"
end
