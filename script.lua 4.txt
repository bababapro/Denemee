local function manageTool(toolName)
    task.spawn(function()
        while _G[toolName.."On"] do
            local player = LocalPlayer
            local character = player.Character
            if character then
                local tool = player.Backpack:FindFirstChild(toolName) or character:FindFirstChild(toolName)
                
                if tool then
                    if tool.Parent ~= character then
                        tool.Parent = character
                    end
                    
                    local event = player:FindFirstChild("muscleEvent")
                    if event then
                        for i = 1, repsPerTick do
                            if not _G[toolName.."On"] then break end
                            event:FireServer("rep")
                        end
                    else
                        tool:Activate()
                    end
                end
            end
            task.wait(0.01)
        end
    end)
end

local function manageParts(state)
    if state and #parts == 0 then
        local partSize = 2048
        local totalDistance = 50000
        local startPosition = Vector3.new(-2, -9.5, -2)
        local numberOfParts = math.ceil(totalDistance / partSize)
        for x = 0, numberOfParts - 1 do
            for z = 0, numberOfParts - 1 do
                local positions = {
                    startPosition + Vector3.new(x * partSize, 0, z * partSize),
                    startPosition + Vector3.new(-x * partSize, 0, z * partSize),
                    startPosition + Vector3.new(-x * partSize, 0, -z * partSize),
                    startPosition + Vector3.new(x * partSize, 0, -z * partSize)
                }
                for _, pos in ipairs(positions) do
                    local p = Instance.new("Part")
                    p.Size = Vector3.new(partSize, 1, partSize)
                    p.Position = pos
                    p.Anchored = true
                    p.Transparency = 1
                    p.CanCollide = true
                    p.Parent = workspace
                    table.insert(parts, p)
                end
            end
        end
    elseif not state then
        for _, p in ipairs(parts) do p.CanCollide = false end
    else
        for _, p in ipairs(parts) do p.CanCollide = true end
    end
end

-- Format fonksiyonları
local function formatNumber(num)
    if num >= 1e18 then return string.format("%.2fQi", num/1e18) end
    if num >= 1e15 then return string.format("%.2fQa", num/1e15) end
    if num >= 1e12 then return string.format("%.2fT", num/1e12) end
    if num >= 1e9 then return string.format("%.2fB", num/1e9) end
    if num >= 1e6 then return string.format("%.2fM", num/1e6) end
    if num >= 1e3 then return string.format("%.2fK", num/1e3) end
    return string.format("%.0f", num)
end

local function formatRate(ratePerSecond)
    return formatNumber(ratePerSecond) .. "/s | "
        .. formatNumber(ratePerSecond * 60) .. "/min | "
        .. formatNumber(ratePerSecond * 3600) .. "/h | "
        .. formatNumber(ratePerSecond * 86400) .. "/d"
end

-- Timer fonksiyonları
local function getElapsed()
    return FastFarm.totalElapsed + (FastFarm.timerStarted and os.clock() - FastFarm.startedAt or 0)
end

local function setTimerRunning(enabled)
    FastFarm.totalElapsed = getElapsed()
    FastFarm.startedAt = os.clock()
    FastFarm.timerStarted = enabled
end

local function updateRuntimeLabel()
    local elapsed = getElapsed()
    FastFarm.Runtime.Text = string.format("Runtime: %dd %dh %dm %ds",
        math.floor(elapsed / 86400), math.floor(elapsed % 86400 / 3600),
        math.floor(elapsed % 3600 / 60), math.floor(elapsed % 60))
end

local function recordStatGain(key, value)
    local previous = FastFarm.statSamples[key]
    FastFarm.statSamples[key] = value
    if FastFarm.running and FastFarm.timerStarted and previous then
        FastFarm.statGains[key] = FastFarm.statGains[key] + math.max(0, value - previous)
    end
end

local function updateStats()
    FastFarm.StrengthStats.Text = "Strength: " .. formatNumber(strengthStat.Value)
        .. " | Gained: " .. (FastFarm.running and not FastFarm.timerStarted
            and "Preparing" or formatNumber(FastFarm.statGains.Strength))
    FastFarm.DurabilityStats.Text = "Durability: " .. formatNumber(durabilityStat.Value)
        .. " | Gained: " .. (FastFarm.running and not FastFarm.timerStarted
            and "Preparing" or formatNumber(FastFarm.statGains.Durability))
    FastFarm.RebirthStats.Text = "Rebirths: " .. formatNumber(rebirthsStat.Value)
        .. " | Gained: " .. formatNumber(FastFarm.statGains.Rebirths)
end

-- Rep gönderme fonksiyonu
local function sendTrainingReps(state, rate, maxBurst, canSend)
    local now = os.clock()
    local elapsed = math.clamp(now - (state.LastAt or now), 0, 0.1)
    state.LastAt = now
    state.Tokens = math.min(maxBurst, (state.Tokens or 0) + rate * elapsed)
    local sent = 0
    while state.Tokens >= 1 and canSend() do
        LocalPlayer.muscleEvent:FireServer("rep")
        state.Tokens = state.Tokens - 1
        sent = sent + 1
    end
    return sent
end

-- Fast Rep Worker
FastFarm.workerActive = false

FastFarm.workerLoop = function()
    FastFarm.LastAt = os.clock()
    local function isRunning()
        return FastFarm.running
    end
    while FastFarm.running do
        -- Rebirth aktifse durdur
        if FastFarm.rebirthRunning then
            FastFarm.FastRepToggle:Set(false)
            break
        end
        
        sendTrainingReps(FastFarm, FastFarm.repSpeed, FastFarm.maxBurst, isRunning)
        
        -- Timer başlat
        if FastFarm.running and not FastFarm.timerStarted then
            FastFarm.statSamples.Strength = strengthStat.Value
            FastFarm.statSamples.Durability = durabilityStat.Value
            setTimerRunning(true)
            FastFarm.nextAverageElapsed = getElapsed() + 10
            updateStats()
        end
        
        RunService.Heartbeat:Wait()
    end
    FastFarm.workerActive = false
end

FastFarm.startWorker = function()
    if FastFarm.workerActive then return end
    FastFarm.workerActive = true
    task.spawn(FastFarm.workerLoop)
end

    updateStats()

local MAX_PET_SLOTS = 12
local TARGET_REP_SPEED = 100

local PETS = {
    ["Omega Overlord"] = {repSpeed = 20, strength = 0, durability = 0, rebirth = 0},
    ["Swift Samurai"] = {repSpeed = 15, strength = 0, durability = 0, rebirth = 0},
    ["Powercore Hound"] = {repSpeed = 0, strength = 25, durability = 15, rebirth = 0},
    ["Titanium Hydra"] = {repSpeed = 0, strength = 0, durability = 0, rebirth = 2},
    ["Tribal Overlord"] = {repSpeed = 0, strength = 0, durability = 0, rebirth = 2},
    ["Inferno Unicorn"] = {repSpeed = 0, strength = 25, durability = 25, rebirth = 0},
    ["Inferno Drake"] = {repSpeed = 40, strength = 0, durability = 0, rebirth = 0},
}

local function getOwnedConfiguredPets()
    local ownedPets = {}
    for _, folder in ipairs(LocalPlayer.petsFolder:GetChildren()) do
        if folder:IsA("Folder") then
            for _, pet in ipairs(folder:GetChildren()) do
                local config = PETS[pet.Name]
                if config then
                    table.insert(ownedPets, {
                        instance = pet,
                        name = pet.Name,
                        repSpeed = config.repSpeed or 0,
                        strength = config.strength or 0,
                        durability = config.durability or 0,
                        rebirth = config.rebirth or 0,
                    })
                end
            end
        end
    end
    return ownedPets
end

local function applyPetLoadout(selected, isCurrent)
    local remote = ReplicatedStorage.rEvents.equipPetEvent
    for _, folder in ipairs(LocalPlayer.petsFolder:GetChildren()) do
        if folder:IsA("Folder") then
            for _, pet in ipairs(folder:GetChildren()) do
                if isCurrent and not isCurrent() then return false end
                remote:FireServer("unequipPet", pet)
            end
        end
    end
    task.wait(0.1)
    for _, pet in ipairs(selected) do
        if isCurrent and not isCurrent() then return false end
        remote:FireServer("equipPet", pet)
    end
    task.wait(0.1)
    return not isCurrent or isCurrent()
end

local function equipBestFarmingPets(isCurrent)
    if isCurrent and not isCurrent() then return end
    local pets = getOwnedConfiguredPets()
    local selected = {}
    local equipped = {}
    local equippedCount = 0
    local currentRepSpeed = 0

    table.sort(pets, function(a, b) return a.repSpeed > b.repSpeed end)

    for _, pet in ipairs(pets) do
        if equippedCount >= MAX_PET_SLOTS or currentRepSpeed >= TARGET_REP_SPEED then break end
        if pet.repSpeed > 0 then
            table.insert(selected, pet.instance)
            equipped[pet.instance] = true
            equippedCount = equippedCount + 1
            currentRepSpeed = currentRepSpeed + pet.repSpeed
        end
    end

    local remainingPets = {}
    for _, pet in ipairs(pets) do
        if not equipped[pet.instance] then
            table.insert(remainingPets, pet)
        end
    end

    table.sort(remainingPets, function(a, b)
        return (a.strength + a.durability) > (b.strength + b.durability)
    end)

    for _, pet in ipairs(remainingPets) do
        if equippedCount >= MAX_PET_SLOTS then break end
        if pet.strength > 0 or pet.durability > 0 then
            table.insert(selected, pet.instance)
            equippedCount = equippedCount + 1
        end
    end

    applyPetLoadout(selected, isCurrent)
    return currentRepSpeed, equippedCount
end

local function equipBestRebirthPets(isCurrent)
    if isCurrent and not isCurrent() then return end
    local pets = getOwnedConfiguredPets()
    local selected = {}

    table.sort(pets, function(a, b) return a.rebirth > b.rebirth end)

    local equippedCount = 0
    for _, pet in ipairs(pets) do
        if equippedCount >= MAX_PET_SLOTS then break end
        if pet.rebirth > 0 then
            table.insert(selected, pet.instance)
            equippedCount = equippedCount + 1
        end
    end

    applyPetLoadout(selected, isCurrent)
    return equippedCount
end

FastFarm.rebirthWorkerActive = false
FastFarm.nextRequestAt = 0
FastFarm.cooldown = 6

FastFarm.isCurrent = function(generation)
    return FastFarm.rebirthRunning and FastFarm.generation == generation
end

FastFarm.performRebirth = function(generation)
    local cycle = {
        Target = 5000 + rebirthsStat.Value * 2550,
        Rebirths = rebirthsStat.Value,
        StartStrength = strengthStat.Value,
        Sent = 0,
        Tokens = 0,
        LastAt = os.clock(),
        StatusAt = -math.huge,
    }
    
    local function needsTraining()
        return FastFarm.isCurrent(generation)
            and rebirthsStat.Value == cycle.Rebirths
            and strengthStat.Value < cycle.Target
    end
    
    while needsTraining() do
        local now = os.clock()
        local character = LocalPlayer.Character
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        
        if humanoid and humanoid.Health > 0 then
            local sent = sendTrainingReps(cycle, FastFarm.repSpeed, FastFarm.maxBurst, function()
                return needsTraining()
            end)
            cycle.Sent = cycle.Sent + sent
            
            if now - cycle.StatusAt >= 0.25 then
                local progress = math.min(100, (strengthStat.Value / cycle.Target) * 100)
                FastFarm.Status.Text = string.format("Status: Training | %.1f%% | %s reps/s", 
                    progress, formatNumber(FastFarm.repSpeed))
                cycle.StatusAt = now
            end
        else
            cycle.Tokens = 0
            cycle.LastAt = now
            FastFarm.Status.Text = "Status: Waiting for Respawn"
        end
        
        RunService.Heartbeat:Wait()
    end
    
    if not FastFarm.isCurrent(generation) or rebirthsStat.Value ~= cycle.Rebirths then return end
    
    -- Rebirth petlerini equip et
    FastFarm.Status.Text = "Status: Switching to Rebirth Pets"
    equipBestRebirthPets(function() return FastFarm.isCurrent(generation) end)
    task.wait(0.25)
    
    -- Rebirth yap
    while FastFarm.isCurrent(generation) and rebirthsStat.Value == cycle.Rebirths do
        local remaining = FastFarm.nextRequestAt - os.clock()
        if remaining > 0 then
            FastFarm.Status.Text = "Status: Rebirth Cooldown | " .. string.format("%.1fs", remaining)
            task.wait(math.min(remaining, 0.1))
        else
            FastFarm.nextRequestAt = os.clock() + FastFarm.cooldown
            ReplicatedStorage.rEvents.rebirthRemote:InvokeServer("rebirthRequest")
        end
    end
end

FastFarm.startRebirthWorker = function()
    if FastFarm.rebirthWorkerActive then return end
    FastFarm.rebirthWorkerActive = true
    
    task.spawn(function()
        local generation = FastFarm.generation
        pcall(function()
            while FastFarm.isCurrent(generation) do
                equipBestFarmingPets(function() return FastFarm.isCurrent(generation) end)
                if not FastFarm.isCurrent(generation) then break end
                FastFarm.performRebirth(generation)
                task.wait()
            end
        end)
        FastFarm.rebirthWorkerActive = false
        if FastFarm.rebirthRunning then FastFarm.startRebirthWorker() end
    end)
end
